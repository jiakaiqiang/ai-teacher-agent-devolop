**apps/server/src/notifications/notices.controller.ts**

```ts
import { Controller, Get, Param, Post, Req, Res, Query, BadRequestException } from '@nestjs/common';
import { RequirePermission } from '../auth/permissions.decorator.js';
import type { AuthenticatedRequest } from '../auth/auth.types.js';
import type { Response } from 'express';
import { PrismaService } from '../prisma/prisma.service.js';

@RequirePermission('ticket:read')
@Controller('notices')
export class NoticesController {
  constructor(private readonly db:PrismaService) {}
  @Get()
  list(@Req() req:AuthenticatedRequest,@Query('before') raw?:string) {
    const before=raw ? Number(raw):undefined;
    if (before!==undefined && (!Number.isSafeInteger(before)||before<1)) throw new BadRequestException('通知游标无效');
    return this.db.notice.findMany({where:{recipientId:req.user!.id,id:before ? {lt:before}:undefined},orderBy:{id:'desc'},take:20});
  }
  @Post(':id/read')
  read(@Req() req:AuthenticatedRequest,@Param('id') id:string) {
    const key=Number(id);
    if (!Number.isSafeInteger(key)||key<1) throw new BadRequestException('通知编号无效');
    return this.db.notice.updateMany({where:{recipientId:req.user!.id,id:key,readAt:null},data:{readAt:new Date()}});
  }
  @Get('stream')
  async stream(@Req() req:AuthenticatedRequest,@Res() res:Response) {
    const recipientId=req.user!.id;
    let last=Number(req.headers['last-event-id'] ?? 0);
    if (!Number.isSafeInteger(last)||last<0) throw new BadRequestException('事件游标无效');
    res.set({'Content-Type':'text/event-stream','Cache-Control':'no-cache'});
    res.flushHeaders();
    let closed=false;
    res.on('close',()=>{closed=true});
    try {
      while (!closed) {
        const batch=await this.db.notice.findMany({where:{recipientId,id:{gt:last}},orderBy:{id:'asc'},take:50});
        for (const item of batch) {
          if (closed) break;
          res.write('id: '+item.id+'\ndata: '+JSON.stringify(item)+'\n\n');
          last=item.id;
        }
        if (batch.length<50 && !closed) await new Promise(resolve=>setTimeout(resolve,1000));
      }
    } finally {res.end()}
  }
}
```

**apps/server/test/notices.e2e-spec.ts**

```ts
import { PrismaService } from '../src/prisma/prisma.service.js';
import { bearer } from './login-fixtures.js';
import { Test } from '@nestjs/testing';
import type { INestApplication } from '@nestjs/common';
import request from 'supertest';
import { afterAll, beforeAll, expect, it } from 'vitest';
import { AppModule } from '../src/app.module.js';
import { configureApp } from '../src/app.setup.js';

let app: INestApplication;
let lead: string;
beforeAll(async () => {
  const mod = await Test.createTestingModule({ imports: [AppModule] }).compile();
  app = mod.createNestApplication();
  configureApp(app);
  await app.init();
  const db=app.get(PrismaService);
  const dept=await db.department.findUniqueOrThrow({where:{name:'网络部'}});
  lead=(await db.user.create({data:{username:'lead-'+crypto.randomUUID(),displayName:'主管',role:'department_lead',departmentId:dept.id,registrationStatus:'approved'}})).username;
});
afterAll(async () => { if (app && lead) await app.get(PrismaService).user.deleteMany({where:{username:lead}}); await app?.close(); });

it('通知按本人读取，重复标记已读只更新一次',async()=>{
  const db=app.get(PrismaService), user=await db.user.findUniqueOrThrow({where:{username:'lin'}});
  const row=await db.notice.create({data:{recipientId:user.id,ticketNo:'T-001',version:0,action:'exercise'}});
  const http=request(app.getHttpServer());
  try {
    const own=await http.get('/api/notices').set('Authorization',await bearer(app)).expect(200);
    expect(own.body.data.some((n:{id:number})=>n.id===row.id)).toBe(true);
    const path='/api/notices/'+row.id+'/read';
    for (const [name,count] of [['wu',0],['lin',1],['lin',0]] as const) {
      const response=await http.post(path).set('Authorization',await bearer(app,name)).expect(201);
      expect(response.body.data.count).toBe(count);
    }
  } finally {await db.notice.delete({where:{id:row.id}});}
});
```
