### 1.1 先定义观测点

计时器用同一进程的 performance.now。首文本到达与浏览器绘制、模型首 token 均有区别。本章 firstTextMs 记录消费者拿到第一个非空文本片段的时刻，meta与空串都不触发。

it('元数据与空文本不抢占首字', () => {

### 1.2 终态决定分母

每次请求只结算一次。completed、cancelled、failed分开计数；故障率以非取消的已结算请求为分母。成功耗时分位数只包含completed；失败时间另行保留。

### 1.3 分位数与空集

采用nearest-rank：排序后取ceil(p*n)-1。报告样本数，空集返回null。小样本p95接近最大值，不能声称生产SLO达标。

    const samples = (['completed', 'cancelled', 'failed'] as const)
      .map(outcome => new LatencyTrace(() => 0).finish(outcome));

      settled: 3, completed: 1, cancelled: 1, failed: 1, failureRate: 0.5,
      completedWithText: 0, completedFirstTextP95Ms: null,
    });
  });
  it('无样本的分位数不是零', () => {
    expect(p95([])).toBeNull();
    expect(summarizeLatency([]).failureRate).toBeNull();
    expect(summarizeLatency([new LatencyTrace(() => 0).finish('cancelled')]))
      .toMatchObject({ failureRate: null, completedTotalP95Ms: null });
  });
  it('终态只能结算一次且迟到文本无效', () => {
    let now = 0;
    const trace = new LatencyTrace(() => now);
    now = 20;
    const first = trace.finish('cancelled');
    now = 40; trace.observe('text', '迟到');
    expect(trace.finish('completed')).toBe(first);
    expect(first.codePoints).toBe(0);
    expect(Object.isFrozen(first)).toBe(true);
  });
  it('时钟回退与非法输入不会成为负延迟', () => {
    let now = 10;
    const trace = new LatencyTrace(() => now);
    now = 9;
    expect(() => trace.finish('completed')).toThrow('BAD_CLOCK');
    expect(() => new LatencyTrace(() => NaN)).toThrow('BAD_CLOCK');
    expect(() => p95([NaN])).toThrow('BAD_DURATION');
  });
  it('nearest rank不修改输入并只聚合成功耗时', () => {
    const values = Array.from({ length: 20 }, (_, i) => 20 - i);
    expect(p95(values)).toBe(19);
    expect(values[0]).toBe(20);
    const sample = { firstTextMs: 2, totalMs: 10, codePoints: 2 };
    expect(summarizeLatency([
      { ...sample, outcome: 'completed' },
      { ...sample, totalMs: 1000, outcome: 'failed' },
    ]).completedTotalP95Ms).toBe(10);
  });
});

**apps/server/src/retrieval/latency.ts**

```ts
export type Outcome = 'completed' | 'failed' | 'cancelled';
export type LatencySample = Readonly<{
  outcome: Outcome; firstTextMs: number | null; totalMs: number; codePoints: number;
}>;

// 一个请求一个实例；只保留数字，不记录业务正文。
export class LatencyTrace {
  private readonly started: number;
  private last: number;
  private first: number | null = null;
  private codePoints = 0;
  private result?: LatencySample;

  constructor(private readonly clock: () => number = () => performance.now()) {
    this.started = clock();
    if (!Number.isFinite(this.started)) throw new Error('BAD_CLOCK');
    this.last = this.started;
  }

  private elapsed() {
    const now = this.clock();
    if (!Number.isFinite(now) || now < this.last) throw new Error('BAD_CLOCK');
    this.last = now;
    return now - this.started;
  }

  observe(kind: 'meta' | 'text', text = '') {
    if (this.result) return;
    const elapsed = this.elapsed();
    if (kind !== 'text' || text.length === 0) return;
    if (this.first === null) this.first = elapsed;
    this.codePoints += Array.from(text).length;
  }

  finish(outcome: Outcome): LatencySample {
    if (this.result) return this.result;
    this.result = Object.freeze({
      outcome, firstTextMs: this.first, totalMs: this.elapsed(), codePoints: this.codePoints,
    });
    return this.result;
  }
}

export function p95(values: readonly number[]): number | null {
  if (values.some(n => !Number.isFinite(n) || n < 0)) throw new Error('BAD_DURATION');
  if (!values.length) return null;
  return [...values].sort((a, b) => a - b)[Math.ceil(values.length * 0.95) - 1]!;
}

// 本章只聚合已结算样本；主动取消不属于故障分母。
export function summarizeLatency(samples: readonly LatencySample[]) {
  const completed = samples.filter(s => s.outcome === 'completed');
  const failed = samples.filter(s => s.outcome === 'failed').length;
  const cancelled = samples.filter(s => s.outcome === 'cancelled').length;
  const eligible = completed.length + failed;
  return {
    settled: samples.length, completed: completed.length, failed, cancelled,
    failureRate: eligible ? failed / eligible : null,
    completedTotalP95Ms: p95(completed.map(s => s.totalMs)),
    completedFirstTextP95Ms: p95(completed.flatMap(s => s.firstTextMs === null ? [] : [s.firstTextMs])),
    completedWithText: completed.filter(s => s.firstTextMs !== null).length,
  };
}
```

**scripts/observe-proxy.mjs**

```text
import assert from 'node:assert/strict';
import { LatencyTrace, summarizeLatency } from '../apps/server/dist/retrieval/latency.js';

// 只对本章六帧夹具采样，不采集正文或凭证。
const kind = process.argv[2] ?? 'flow';
assert.ok(['flow', 'idle'].includes(kind));
const trace = new LatencyTrace();
let frames = 0;
let outcome = 'failed';
let reader;
try {
  const response = await fetch('http://127.0.0.1:8089/' + kind, { signal: AbortSignal.timeout(9000) });
  assert.equal(response.status, 200);
  reader = response.body.getReader();
  const decoder = new TextDecoder();
  let pending = '';
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    pending += decoder.decode(value, { stream: true });
    let boundary;
    while ((boundary = pending.indexOf('\n\n')) >= 0) {
      const frame = pending.slice(0, boundary);
      pending = pending.slice(boundary + 2);
      if (frame.startsWith('data: ')) {
        trace.observe('text', frame.slice(6));
        frames++;
      }
    }
  }
  if (frames === 6 && !pending) outcome = 'completed';
} catch {
  outcome = 'failed';
} finally {
  if (reader) { await reader.cancel().catch(() => {}); reader.releaseLock(); }
}
const sample = trace.finish(outcome);
console.log(JSON.stringify({ source: 'local-nginx-fixture', kind, frames, sample, summary: summarizeLatency([sample]) }));
assert.equal(outcome, kind === 'flow' ? 'completed' : 'failed');
assert.equal(frames, kind === 'flow' ? 6 : 1);
```
