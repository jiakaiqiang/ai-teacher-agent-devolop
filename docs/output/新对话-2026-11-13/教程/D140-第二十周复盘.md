```ts
import { describe, expect, it } from 'vitest';
import { countTokens, packHandover, type HandoverInput } from './handover-budget.js';

const base: HandoverInput = { policy: 'Use only ticket evidence.', task: { ticketNo: 'T-7', decision: 'hold', approval: 'waiting' },
  question: '当前进展？', history: [], documents: [], results: [] };

describe('输入和输出共享模型窗口', () => {
  it('必需块差一个 token 也不能挤占输出', () => {
    const exact = packHandover(base).guardTokens;
    expect(packHandover(base, exact + 16, 16).guardTokens).toBe(exact);
    expect(() => packHandover(base, exact + 15, 16)).toThrow('CONTEXT_REQUIRED_TOO_LARGE');
    expect(countTokens('Hello world!')).toBe(4);
  });

  it('可选块按字节上界裁剪且独立报告分词数', () => {
    const history = ['steady '.repeat(40)];
    const baseCost = packHandover(base).guardTokens;
    const plan = packHandover({ ...base, history }, baseCost + 30, 16);
    expect(plan.packed.history).toEqual([]);
    expect(plan.guardTokens).toBeGreaterThan(plan.inputTokens);
    expect(plan.guardTokens + plan.output).toBeLessThanOrEqual(baseCost + 30);
  });

  it('按区块限制历史同时保持最近轮次', () => {
    const history = Array.from({ length: 50 }, (_, i) => 'round ' + i + ' ' + 'disk check '.repeat(10));
    const result = packHandover({ ...base, history });
    expect(result.packed.history.length).toBeLessThan(history.length);
    expect(result.packed.history.at(-1)).toBe(history.at(-1));
    expect(result.packed.task.ticketNo).toBe('T-7');
  });
});
```

**apps/server/src/agent/handover-budget.ts**

```ts
import tokenizer from 'llama-tokenizer-js';

export const modelProfile = { name: 'tinyllama:latest', window: 2048, output: 64 } as const;
export const countTokens = (text: string) => tokenizer.encode(text).length;
// TinyLlama 的 raw 提示走 LLaMA 字节回退。按原始 UTF-8 字节另留 8 个位置给边界标记。
// 该上界仅用于放行判断；展示与成本对照仍保留分词器读数和模型回报。
function guardedTokens(prompt: string) {
  return Math.max(countTokens(prompt), Buffer.byteLength(prompt, 'utf8') + 8);
}
export interface HandoverInput {
  policy: string;
  task: { ticketNo: string; decision: string; approval: string };
  question: string;
  history: readonly string[];
  documents: readonly string[];
  results: readonly string[];
  summary?: string;
}

// 实验走 Ollama raw 模式。计数和发送使用同一份字符串，模型端不会再套模板。
export function rawPrompt(input: HandoverInput) {
  const { policy, ...material } = input;
  return policy + '\nDATA ' + JSON.stringify(material) + '\nAnswer briefly:';
}

function suffixWithin(items: readonly string[], budget: number) {
  const selected: string[] = [];
  for (const item of [...items].reverse()) {
    if (countTokens(JSON.stringify([item, ...selected])) > budget) break;
    selected.unshift(item);
  }
  return selected;
}

export function packHandover(input: HandoverInput, window: number = modelProfile.window, output: number = modelProfile.output) {
  if (!Number.isSafeInteger(window) || !Number.isSafeInteger(output) || output < 1 || window > modelProfile.window || output >= window) {
    throw new Error('CONTEXT_BUDGET_INVALID');
  }
  const packed = { ...input, history: suffixWithin(input.history, 400),
    documents: suffixWithin(input.documents, 256), results: suffixWithin(input.results, 384) };
  // 区块上限约束单一来源，再按实际序列化总长挤出低优先级材料。
  for (const items of [packed.history, packed.documents, packed.results]) {
    while (items.length && guardedTokens(rawPrompt(packed)) + output > window) items.shift();
  }
  const prompt = rawPrompt(packed);
  const inputTokens = countTokens(prompt);
  const guardTokens = guardedTokens(prompt);
  if (guardTokens + output > window) throw new Error('CONTEXT_REQUIRED_TOO_LARGE');
  return { prompt, inputTokens, guardTokens, output, packed };
}
```

```ts
import { describe, expect, it } from 'vitest';
import { manualForModel, manualPage } from './manual-page.js';

describe('手册按引用读取', () => {
  it('两页拼回脱敏正文且不切断表情', () => {
    const text = '盘'.repeat(239) + '📦结束';
    const first = manualPage('m', text);
    expect(first.next).toBe(240);
    const second = manualPage('m', text, first.next!, first.revision);
    expect(second.next).toBeNull();
    expect(first.quote.text + second.quote.text).toBe(text);
  });

  it('长手册只发引用而短手册仍给出原句', () => {
    const page = (body: string) => ({ title: { text: '手册', clipped: false }, ...manualPage('m', body) });
    const long = page('磁'.repeat(241));
    expect(manualForModel(long)).toEqual({ sourceKey: 'manual:m', revision: long.revision, title: long.title, offset: 0 });
    expect(manualForModel(page('磁'.repeat(240)))).toHaveProperty('quote.text', '磁'.repeat(240));
  });

  it('过期版本与越界游标分别拒绝', () => {
    const ref = manualPage('m', '旧文');
    expect(() => manualPage('m', '新文', 0, ref.revision)).toThrow('MANUAL_VERSION_CHANGED');
    expect(() => manualPage('m', '旧文', 3, ref.revision)).toThrow('MANUAL_OFFSET_INVALID');
  });
});
```

**apps/server/src/agent/manual-page.ts**

```ts
import { createHash } from 'node:crypto';
import { ConflictException, BadRequestException } from '@nestjs/common';
import { redactForLog } from '../common/sensitive.js';

// 对脱敏后的正文按 Unicode 码点分页，引用指向固定版本，翻页不占模型上下文。
export function manualPage(id: string, body: string, offset = 0, expected?: string) {
  const text = redactForLog(body);
  const revision = createHash('sha256').update(text).digest('hex');
  if (expected !== undefined && expected !== revision) throw new ConflictException('MANUAL_VERSION_CHANGED');
  const points = Array.from(text);
  if (!Number.isSafeInteger(offset) || offset < 0 || offset > points.length) throw new BadRequestException('MANUAL_OFFSET_INVALID');
  const end = Math.min(offset + 240, points.length);
  return { sourceKey: 'manual:' + id, revision, quote: { text: points.slice(offset, end).join(''), clipped: end < points.length },
    next: end < points.length ? end : null };
}

// 长手册只把引用交给诊断模型。人工打开引用时仍从第一页校验权限和版本。
export function manualForModel(page: ReturnType<typeof manualPage> & { title: { text: string; clipped: boolean } }) {
  if (!page.quote.clipped) return page;
  return { sourceKey: page.sourceKey, revision: page.revision, title: page.title, offset: 0 };
}
```
