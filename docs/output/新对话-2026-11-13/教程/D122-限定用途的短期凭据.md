## 一、知识点与推导

校验主体、工单、用途、动作及有效期，任一不符均拒绝。

画出三个批准动作的对象。客户管理员批准本公司注册，运维部门负责人复核处理结果并审核通用知识。分别写出要锁定的资源、版本和有权操作的主体。

执行器认领运行后，按当前任务加载主题，为摘要及补查工具签发一个对象引用。用 WeakMap 在服务端保留权限范围，到期时重新申请新的运行，旧引用不延长。

只有签发时创建的对象引用能找到凭据记录，复制对象只会得到空壳。执行器在数据库事务里同时检查账号的实时权限、运行和工具名称，不把凭据放进提示词或 Trace。

```ts
import { checkGrant, issueGrant } from './diagnosis-grant.js';

describe('诊断工具的短期调用凭据', () => {
  it('最后一毫秒可用且到期立即失效', () => {
    const handle = issueGrant('甲', '运行一', ['ticket_summary'], 100);
    expect(() => checkGrant(handle, '甲', '运行一', 'ticket_summary', 30099)).not.toThrow();
    expect(() => checkGrant(handle, '甲', '运行一', 'ticket_summary', 30100)).toThrow('TOOL_GRANT_INVALID');
  });
  it('复制结果和其他用途都不能消费凭据', () => {
    const handle = issueGrant('甲', '运行一', ['ticket_progress'], 0);
    expect(JSON.stringify(handle)).toBe('{}');
    for (const [user, run] of [['乙', '运行一'], ['甲', '运行二']]) {
      expect(() => checkGrant(handle, user, run, 'ticket_progress', 1)).toThrow('TOOL_GRANT_INVALID');
    }
    expect(() => checkGrant({}, '甲', '运行一', 'ticket_progress', 1)).toThrow('TOOL_GRANT_INVALID');
    expect(() => checkGrant(handle, '甲', '运行一', 'ticket_category', 1)).toThrow('TOOL_GRANT_INVALID');
  });
});
```

```ts
import { CallError, type Name } from './tool-catalog.js';

type Bound = { userId: string; runId: string; names: ReadonlySet<Name>; expires: number };
// 引用只在本机本次运行里传递。序列化所得的空对象没有调用能力。
const grants = new WeakMap<object, Bound>();
export function issueGrant(userId: string, runId: string, names: readonly Name[], now = Date.now()) {
  const handle = Object.freeze({});
  grants.set(handle, { userId, runId, names: new Set(names), expires: now + 30_000 });
  return handle;
}
export function checkGrant(handle: object, userId: string, runId: string, name: Name, now = Date.now()) {
  const grant = grants.get(handle);
  if (!grant || now >= grant.expires || grant.userId !== userId || grant.runId !== runId || !grant.names.has(name)) {
    throw new CallError('TOOL_GRANT_INVALID');
  }
}
```

```ts
import { Injectable } from '@nestjs/common';
import { z } from 'zod';
import type { AuthUser } from '../auth/auth.types.js';
import { ModelGateway } from '../assistant/model-gateway.js';
import { parseToolReply } from '../assistant/tool-protocol.js';
import { UsageMeter } from '../assistant/usage-meter.js';
import { issueGrant } from './diagnosis-grant.js';
import { definitions, type Name } from './tool-catalog.js';
import { TicketToolsService } from './ticket-tools.service.js';
import { RunExecutorService } from './run-executor.service.js';

export const diagnosisSchema = z.object({ category: z.enum(['network', 'storage', 'power', 'unknown']) }).strict();
const checks = { network: '核对链路状态与丢包记录', storage: '核对磁盘容量与增长记录',
  power: '核对供电与电源告警', unknown: '补充发生时间和设备告警' };

@Injectable()
export class DiagnosisService {
  constructor(private readonly executor: RunExecutorService, private readonly tools: TicketToolsService,
    private readonly gateway: ModelGateway) {}

  async execute(user: AuthUser, id: string) {
    const evidence: unknown[] = [];
    let pending = 'ticket_summary';
    let expanded = false;
    let grant: object | undefined;
    return this.executor.execute(user, id, async (step, context) => {
      const topic = /历史|记录/.test(context.goal) ? 'history' : 'device';
      const allowedTools = definitions(topic);
      grant ??= issueGrant(user.id, id, ['ticket_summary', ...allowedTools.map(tool => tool.name as Name)]);
      const signal = AbortSignal.timeout(step.deadlineMs);
      if (pending) {
        const name = pending;
        pending = '';
        const args = name.endsWith('_page') ? { limit: 3, after: null } : {};
        const call = { name, version: 1, args };
        const [result] = await this.tools.batch(user, id, [call], signal, false, grant);
        evidence.push(result);
        const failure = !result?.ok && name === 'ticket_summary' ? 'TICKET_READ_FAILED' : undefined;
        return { phase: 'tool', summary: name, tokens: 0, failure,
          trace: { kind: 'tool', input: JSON.stringify(call), output: JSON.stringify(result) } };
      }
      const available = expanded ? [] : allowedTools;
      const names = available.map(t => t.name);
      const instruction = '根据工单证据选择 network（网络）、storage（磁盘）、power（供电）或 unknown（信息不足），返回 {"category":"类别"}。' +
        (names.length ? '也可返回 {"tool":"名称"} 补查一次。可选工具：' + available.map(t => t.name + '：' + t.description).join('；') : '已经补查过，只返回类别。');
      const input = JSON.stringify({ goal: context.goal, evidence });
      const meter = new UsageMeter();
      let output = '';
      try {
        const schema = expanded ? diagnosisSchema : z.union([diagnosisSchema, z.object({ tool: z.enum(names) }).strict()]);
        const raw = await this.gateway.complete([{ role: 'system', content: instruction }, { role: 'user', content: input }],
          signal, 'default', Math.min(step.deadlineMs, 5000), { type: 'json_schema',
            json_schema: { name: 'diagnosis', strict: true, schema: z.toJSONSchema(schema) } }, meter);
        const reply = parseToolReply(raw);
        if (reply.kind !== 'answer') throw new Error('DIAGNOSIS_INVALID');
        output = reply.text;
        const choice = schema.parse(JSON.parse(output));
        const usage = meter.summary();
        const trace = { kind: 'model' as const, input: instruction + '\n' + input, output,
          inputTokens: usage.inputTokens, outputTokens: usage.outputTokens };
        const tokens = usage.inputTokens + usage.outputTokens;
        if ('tool' in choice) {
          if (expanded || !names.includes(choice.tool)) throw new Error('DIAGNOSIS_TOOL_UNAVAILABLE');
          expanded = true; pending = choice.tool;
          return { phase: 'plan', summary: '选择补查工具', tokens, trace };
        }
        return { phase: 'answer', summary: '分诊与核查建议', answer: choice.category + ' · ' + checks[choice.category], tokens, trace };
      } catch {
        const usage = meter.summary();
        return { phase: 'answer', summary: '模型未给出有效分诊', tokens: usage.inputTokens + usage.outputTokens,
          failure: 'DIAGNOSIS_MODEL_FAILED', failureReason: 'model_failed',
          trace: { kind: 'model', input: instruction + '\n' + input, output, inputTokens: usage.inputTokens, outputTokens: usage.outputTokens } };
      }
    });
  }
}
```

**apps/server/src/agent/ticket-tools.service.ts**

```ts
import { BadRequestException, Injectable, NotFoundException } from '@nestjs/common';
import { createHash } from 'node:crypto';
import { z } from 'zod';
import { AuditService } from '../audit/audit.service.js';
import { readAccess } from '../assistant/read-tools.service.js';
import type { AuthUser } from '../auth/auth.types.js';
import { Prisma } from '../generated/prisma/client.js';
import { PrismaService } from '../prisma/prisma.service.js';
import { RunFailure } from './run-errors.js';
import { AgentRunsService } from './agent-runs.service.js';
import { CallError, catalog, parseCall } from './tool-catalog.js';
import { saveNote, listNotes } from './ticket-tool-note.js';
import { readTicketTool, visibleTicket } from './ticket-tool-read.js';
import { decideTool, type ToolScene } from './agent-permission.js';
import { checkGrant } from './diagnosis-grant.js';
import { ToolPolicy } from './tool-policy.js';

export const batchSchema = z.object({ calls: z.array(z.unknown()).min(1).max(3) }).strict();
function describe(input: unknown) {
  const value = z.object({ name: z.string().regex(/^[a-z_]{1,50}$/), version: z.number().int() }).safeParse(input);
  return value.success ? value.data : { name: 'invalid', version: 0 };
}
function errorCode(error: unknown) {
  if (error instanceof CallError) return error.code;
  if (error instanceof Error && error.message === 'TOOL_FORBIDDEN') return 'TOOL_FORBIDDEN';
  return 'TOOL_FAILED';
}
@Injectable()
export class TicketToolsService {
  private readonly policy = new ToolPolicy();
  constructor(private readonly db: PrismaService, private readonly runs: AgentRunsService, private readonly audit: AuditService) {}

  async owned(user: AuthUser, id: string) {
    try { return await this.runs.get(user, id); }
    catch (error) {
      if (error instanceof RunFailure && error.code.endsWith('NOT_FOUND')) throw new NotFoundException('运行不可见');
      if (error instanceof RunFailure && error.code === 'RUN_INPUT_INVALID') throw new BadRequestException('运行编号无效');
      throw error;
    }
  }
  async history(user: AuthUser, id: string, after?: string) {
    await this.owned(user, id);
    const page = await this.db.auditLog.findMany({
      where: { entityType: 'user', entityKey: user.id, actorId: user.id, action: 'agent.tool_call',
        before: { path: ['runId'], equals: id }, ...(after ? { id: { lt: after } } : {}) },
      select: { id: true, actorUsername: true, occurredAt: true, requestId: true, before: true, after: true },
      take: 21, orderBy: { id: 'desc' },
    });
    const items = page.slice(0, 20);
    return { items, next: page.length > 20 ? items.at(-1)!.id : null };
  }
  async batch(user: AuthUser, runId: string, inputs: unknown[], signal: AbortSignal, allowWrites = false, grant?: object) {
    const run = await this.owned(user, runId);
    batchSchema.parse({ calls: inputs });
    // 每个槽独立结束。写工具只接受单项调用，避免用并行顺序表达依赖。
    return Promise.all(inputs.map(input => this.invoke(user, run.id, run.ticketNo, input, signal, allowWrites ? 'workbench' : 'diagnosis', allowWrites && inputs.length === 1, grant)));
  }
  private async invoke(user: AuthUser, runId: string, ticketNo: string, input: unknown, signal: AbortSignal, scene: ToolScene, allowWrites: boolean, grant?: object) {
    const description = describe(input);
    const argsHash = createHash('sha256').update(JSON.stringify(input ?? null)).digest('hex');
    let parameters: Prisma.InputJsonObject = {};
    const base = { actorId: user.id, actorUsername: user.username, action: 'agent.tool_call',
      entityType: 'user' as const, entityKey: user.id };
    try {
      const call = parseCall(input);
      if (call.effect === 'write' && !allowWrites) throw new CallError('TOOL_WRITE_REQUIRES_SINGLE');
      // 审计展示可检索的非正文参数。笔记正文仅记录摘要和长度。
      const args = call.args as { limit?: number; after?: string | null; requestKey?: string; text?: string };
      parameters = args.requestKey ? { requestKey: args.requestKey, chars: args.text?.length ?? 0 } :
        args.limit ? { limit: args.limit, after: args.after ?? null } : {};
      const result = await this.policy.run(call.name, call.effect === 'read' || call.name === 'save_private_note', signal, async child => {
        try {
          return await this.db.$transaction(async tx => {
            child.throwIfAborted();
            await tx.$executeRaw`SET LOCAL statement_timeout = '1200ms'`;
            const access = await readAccess(tx, user.id);
            const bound = await tx.agentRun.findFirst({ where: { id: runId, userId: user.id, ticketNo }, select: { id: true } });
            if (!bound) throw new CallError('TOOL_FORBIDDEN');
            if (decideTool(access.user.permissions ?? [], scene, call.name) !== 'allow') throw new CallError('TOOL_FORBIDDEN');
            if (scene === 'diagnosis') checkGrant(grant ?? {}, user.id, runId, call.name);
            let value: unknown;
            if (call.name === 'save_private_note' || call.name === 'list_private_notes') {
              const target = await visibleTicket(tx, access.user, ticketNo);
              value = call.name === 'save_private_note' ? await saveNote(tx, user.id, target.id, call.args) :
                await listNotes(tx, user.id, target.id, call.args);
            } else value = await readTicketTool(tx, access.user, ticketNo, call.name, call.args);
            await this.audit.record(tx, { ...base, before: { runId, ...description, argsHash, parameters },
              after: { ok: true, effect: catalog[call.name].effect, resultBytes: Buffer.byteLength(JSON.stringify(value)) } });
            child.throwIfAborted();
            return value;
          }, { maxWait: 500, timeout: 1600, isolationLevel: 'RepeatableRead' });
        } catch (error) {
          if (error instanceof Prisma.PrismaClientKnownRequestError) {
            if (['P2024', 'P2034'].includes(error.code)) throw new CallError('TOOL_TEMPORARY');
            if (error.code === 'P2028') throw new CallError('TOOL_TIMEOUT');
          }
          throw error;
        }
      });
      return { ...description, ok: true as const, data: result.value, attempts: result.attempts };
    } catch (error) {
      const code = errorCode(error);
      try {
        await this.db.$transaction(tx => this.audit.record(tx, { ...base, before: { runId, ...description, argsHash, parameters },
          after: { ok: false, code } }), { maxWait: 500, timeout: 1000 });
      } catch { return { ...description, ok: false as const, code: 'TOOL_AUDIT_UNAVAILABLE' }; }
      return { ...description, ok: false as const, code };
    }
  }
}
```
