```ts
import type { UsageSummary } from './usage-meter.js';

// 演示价表，不是供应商报价，更不代表本机实际成本。
export const demoTariff = Object.freeze({
  id: 'demo-linear-v1', currency: 'CNY', inputMicroPerToken: 2n, outputMicroPerToken: 6n,
});

function validate(summary: UsageSummary) {
  const counts = [summary.inputTokens, summary.outputTokens];
  if (counts.some(v => !Number.isSafeInteger(v) || v < 0 || v > 4_000_000) ||
      !Number.isSafeInteger(summary.attempts) || summary.attempts < 0 || summary.attempts > 4 ||
      typeof summary.exact !== 'boolean' ||
      (summary.attempts === 0 && counts.some(v => v !== 0))) throw new Error('USAGE_INVALID');
}

function formatMicro(value: bigint) {
  return (value / 1_000_000n).toString() + '.' + (value % 1_000_000n).toString().padStart(6, '0');
}

export function quoteUsage(summary: UsageSummary) {
  validate(summary);
  const micro = BigInt(summary.inputTokens) * demoTariff.inputMicroPerToken +
    BigInt(summary.outputTokens) * demoTariff.outputMicroPerToken;
  const knownAmount = formatMicro(micro);
  return {
    tariffId: demoTariff.id, currency: demoTariff.currency, tariffKind: 'synthetic' as const,
    status: summary.exact ? 'complete-usage' as const : 'unknown' as const,
    amount: summary.exact ? knownAmount : null, knownAmount,
    attempts: summary.attempts,
  };
}

// embed只有输入token；向量维度不属于completion token。
export function embeddingUsage(raw: unknown): UsageSummary {
  const count = raw && typeof raw === 'object' && 'prompt_eval_count' in raw ? raw.prompt_eval_count : undefined;
  const exact = typeof count === 'number' && Number.isSafeInteger(count) && count >= 0 && count <= 1_000_000;
  return { inputTokens: exact ? count : 0, outputTokens: 0, exact, attempts: 1 };
}
```

**apps/server/src/retrieval/artifact-audit.ts**

```ts
import { createHash } from 'node:crypto';
import { z } from 'zod';

const artifact = z.object({
  id: z.string().regex(/^[a-z0-9-]{1,48}$/),
  root: z.enum(['verify', 'lesson']),
  path: z.string().min(1).max(240),
  sha256: z.string().regex(/^[a-f0-9]{64}$/),
  scope: z.enum(['reference-code', 'local-fixture', 'local-model', 'offline-eval']),
}).strict();
const manifest = z.array(artifact).min(1).max(32);
export type Artifact = z.infer<typeof artifact>;

function allowed(item: Artifact) {
  const parts = item.path.split('/');
  if (item.path.includes('\\') || item.path.includes(':') ||
      parts.some(p => !p || p === '.' || p === '..' || p.startsWith('.env'))) return false;
  if (item.root === 'lesson' && /^\.cache\/验证\/新版验收\/D[\w\u4e00-\u9fff-]+\.(log|json)$/.test(item.path)) return true;
  if (item.root === 'lesson') return /^\.cache\/验证\/D\d+\/[\w-]+\.(log|json)$/.test(item.path);
  return /^(apps|packages|infra|scripts)\//.test(item.path) && /\.(ts|json|mjs|yml|conf)$/.test(item.path);
}

// 先完整检查路径，再读内容；只输出摘要，不暴露证据正文。
export function auditArtifacts(input: unknown, read: (item: Artifact) => Uint8Array) {
  const entries = manifest.parse(input);
  if (new Set(entries.map(e => e.id)).size !== entries.length) throw new Error('DUPLICATE_ID');
  if (!entries.every(allowed)) throw new Error('UNSAFE_PATH');
  return entries.map(entry => {
    const bytes = read(entry);
    if (bytes.byteLength > 4_000_000) throw new Error('ARTIFACT_TOO_LARGE');
    const sha256 = createHash('sha256').update(bytes).digest('hex');
    if (sha256 !== entry.sha256) throw new Error('ARTIFACT_CHANGED:' + entry.id);
    return { ...entry, bytes: bytes.byteLength };
  });
}
```

```ts
export type Outcome = 'completed' | 'failed' | 'cancelled';
export type LatencySample = Readonly<{
  outcome: Outcome; firstTextMs: number | null; totalMs: number; codePoints: number;
}>;

// 一个请求一个实例；只保留数字，不记录业务正文。
export class LatencyTrace {
  private readonly started: number;
  private last: number;
  private first: number | null = null;
  private codePoints = 0;
  private result?: LatencySample;

  constructor(private readonly clock: () => number = () => performance.now()) {
    this.started = clock();
    if (!Number.isFinite(this.started)) throw new Error('BAD_CLOCK');
    this.last = this.started;
  }

  private elapsed() {
    const now = this.clock();
    if (!Number.isFinite(now) || now < this.last) throw new Error('BAD_CLOCK');
    this.last = now;
    return now - this.started;
  }

  observe(kind: 'meta' | 'text', text = '') {
    if (this.result) return;
    const elapsed = this.elapsed();
    if (kind !== 'text' || text.length === 0) return;
    if (this.first === null) this.first = elapsed;
    this.codePoints += Array.from(text).length;
  }

  finish(outcome: Outcome): LatencySample {
    if (this.result) return this.result;
    this.result = Object.freeze({
      outcome, firstTextMs: this.first, totalMs: this.elapsed(), codePoints: this.codePoints,
    });
    return this.result;
  }
}

export function p95(values: readonly number[]): number | null {
  if (values.some(n => !Number.isFinite(n) || n < 0)) throw new Error('BAD_DURATION');
  if (!values.length) return null;
  return [...values].sort((a, b) => a - b)[Math.ceil(values.length * 0.95) - 1]!;
}

// 本章只聚合已结算样本；主动取消不属于故障分母。
export function summarizeLatency(samples: readonly LatencySample[]) {
  const completed = samples.filter(s => s.outcome === 'completed');
  const failed = samples.filter(s => s.outcome === 'failed').length;
  const cancelled = samples.filter(s => s.outcome === 'cancelled').length;
  const eligible = completed.length + failed;
  return {
    settled: samples.length, completed: completed.length, failed, cancelled,
    failureRate: eligible ? failed / eligible : null,
    completedTotalP95Ms: p95(completed.map(s => s.totalMs)),
    completedFirstTextP95Ms: p95(completed.flatMap(s => s.firstTextMs === null ? [] : [s.firstTextMs])),
    completedWithText: completed.filter(s => s.firstTextMs !== null).length,
  };
}
```

```ts
// 只比较完整 SSE 帧到达的相对时间；块数量不等于事件数量。
export function assessProxy(firstMs: number, totalMs: number, frames: number) {
  if (![firstMs,totalMs].every(Number.isFinite) || firstMs<0 || firstMs>totalMs) throw Error('BAD_CLOCK');
  if (frames!==6) throw Error('INCOMPLETE_STREAM');
  if (firstMs>=1500) throw Error('BUFFERED_FIRST_EVENT');
  if (totalMs<3000) throw Error('EARLY_END');
  return { firstMs,totalMs,frames };
}
```

```text
import assert from 'node:assert/strict';
import { LatencyTrace, summarizeLatency } from '../apps/server/dist/retrieval/latency.js';

// 只对本章六帧夹具采样，不采集正文或凭证。
const kind = process.argv[2] ?? 'flow';
assert.ok(['flow', 'idle'].includes(kind));
const trace = new LatencyTrace();
let frames = 0;
let outcome = 'failed';
let reader;
try {
  const response = await fetch('http://127.0.0.1:8089/' + kind, { signal: AbortSignal.timeout(9000) });
  assert.equal(response.status, 200);
  reader = response.body.getReader();
  const decoder = new TextDecoder();
  let pending = '';
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    pending += decoder.decode(value, { stream: true });
    let boundary;
    while ((boundary = pending.indexOf('\n\n')) >= 0) {
      const frame = pending.slice(0, boundary);
      pending = pending.slice(boundary + 2);
      if (frame.startsWith('data: ')) {
        trace.observe('text', frame.slice(6));
        frames++;
      }
    }
  }
  if (frames === 6 && !pending) outcome = 'completed';
} catch {
  outcome = 'failed';
} finally {
  if (reader) { await reader.cancel().catch(() => {}); reader.releaseLock(); }
}
```
