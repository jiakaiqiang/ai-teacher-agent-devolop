### 1.1 调用是建议，执行在宿主

模型输出工具名、JSON参数字符串和调用ID，不直接运行程序。工具结果以role:tool和对应tool_call_id回填，最终回答由下一次模型调用生成。协议依据[OpenAI Function Calling](https://developers.openai.com/api/docs/guides/function-calling)的Chat Completions示例；本章不混用Responses的call_id字段。

### 1.3 有限步骤是产品约束

第一轮允许工具，第二轮只总结；第二轮仍请求工具就报TOOL_STEP_LIMIT，不再执行。最多三个工具串行运行，单结果4096个UTF8字节，正文8192字符。工具结果和模型正文是不同边界，不能把字符上限写成token上限。

### 1.4 一个总预算和明确取消边界

10秒包住模型与工具，Promise.race让不合作回调也能结束本地等待；每次await后检查signal，阻止迟到结果继续发事件。finally移除定时器和监听。已读[Nest AI概览](https://docs.nestjs.com/ai/overview.md)，当前方案基于AI SDK7、要求Node22；本机Node20先复用W9网关，不新增该SDK。

```ts
import { parseToolReply, ToolFailure, type ToolCall, type ToolMessage } from './tool-protocol.js';

export type ToolProgress = { kind: 'tool-start'; call: ToolCall }
  | { kind: 'tool-result'; call: ToolCall; content: string };
export interface ToolLoopOptions {
  messages: ToolMessage[];
  complete: (messages: ToolMessage[], signal: AbortSignal, toolsEnabled: boolean) => Promise<unknown>;
  execute: (call: ToolCall, signal: AbortSignal) => Promise<unknown>;
  signal?: AbortSignal;
  budgetMs?: number;
  onProgress?: (event: ToolProgress) => void;
}

// 一轮最多三个只读工具，再给模型一次总结机会；不是无限Agent循环。
export async function runToolLoop(options: ToolLoopOptions) {
  const budget = options.budgetMs ?? 10000;
  if (!Number.isInteger(budget) || budget < 1 || budget > 10000) throw new ToolFailure('TOOL_BUDGET_INVALID');
  const controller = new AbortController();
  const cancel = () => controller.abort(new ToolFailure('TOOL_CANCELLED'));
  options.signal?.addEventListener('abort', cancel, { once: true });
  if (options.signal?.aborted) cancel();
  const timer = setTimeout(() => controller.abort(new ToolFailure('TOOL_TIMEOUT')), budget);
  const messages = structuredClone(options.messages);
  const appended: ToolMessage[] = [];
  let rejectAbort: () => void = () => {};
  const aborted = new Promise<never>((_, reject) => {
    rejectAbort = () => reject(controller.signal.reason);
    controller.signal.addEventListener('abort', rejectAbort, { once: true });
    if (controller.signal.aborted) rejectAbort();
  });
  const work = async () => {
    for (let step = 0; step < 2; step++) {
      controller.signal.throwIfAborted();
      const raw = await options.complete(messages, controller.signal, step === 0);
      controller.signal.throwIfAborted();
      const reply = parseToolReply(raw);
      if (reply.kind === 'answer') {
        appended.push({ role: 'assistant', content: reply.text });
        return { text: reply.text, appended };
      }
      if (step !== 0) throw new ToolFailure('TOOL_STEP_LIMIT');
      const assistant: ToolMessage = { role: 'assistant', content: reply.content, tool_calls: reply.calls };
      messages.push(assistant); appended.push(assistant);
      for (const call of reply.calls) {
        controller.signal.throwIfAborted();
        options.onProgress?.({ kind: 'tool-start', call });
        const result = await options.execute(call, controller.signal);
        controller.signal.throwIfAborted();
        const content = JSON.stringify(result);
        if (typeof content !== 'string' || Buffer.byteLength(content, 'utf8') > 4096) throw new ToolFailure('TOOL_RESULT_LIMIT');
        const message: ToolMessage = { role: 'tool', tool_call_id: call.id, content };
        messages.push(message); appended.push(message);
        options.onProgress?.({ kind: 'tool-result', call, content });
      }
    }
    throw new ToolFailure('TOOL_STEP_LIMIT');
  };
  try { return await Promise.race([work(), aborted]); }
  finally {
    clearTimeout(timer);
    options.signal?.removeEventListener('abort', cancel);
    controller.signal.removeEventListener('abort', rejectAbort);
  }
}
```

**apps/server/src/assistant/tool-protocol.ts**

```ts
import { z } from 'zod';

export const toolNameSchema = z.enum(['find_ticket', 'list_ticket_logs', 'list_ticket_history']);
export type ToolName = z.infer<typeof toolNameSchema>;
const callSchema = z.object({
  id: z.string().regex(/^[a-zA-Z0-9_-]{1,64}$/),
  type: z.literal('function'),
  function: z.object({ name: toolNameSchema, arguments: z.string().max(1024) }).strict(),
}).strict();
export type ToolCall = z.infer<typeof callSchema>;
export type ToolMessage =
  | { role: 'system' | 'user'; content: string }
  | { role: 'assistant'; content: string | null; tool_calls?: ToolCall[] }
  | { role: 'tool'; tool_call_id: string; content: string };
export class ToolFailure extends Error {
  constructor(readonly code: string) { super(code); }
}

// 供应商的外围元数据可以存在；只抽取经过验证的消息字段进入下一轮。
const envelope = z.object({ choices: z.array(z.object({
  index: z.literal(0), finish_reason: z.enum(['stop', 'length', 'tool_calls']),
  message: z.object({ content: z.string().max(8192).nullable().optional(),
    refusal: z.string().nullable().optional(), tool_calls: z.array(callSchema).max(3).optional() }),
})).length(1) });

export function parseToolReply(raw: unknown):
  { kind: 'answer'; text: string } | { kind: 'tools'; calls: ToolCall[]; content: string | null } {
  const parsed = envelope.safeParse(raw);
  if (!parsed.success) throw new ToolFailure('TOOL_PROTOCOL_ERROR');
  const choice = parsed.data.choices[0]!;
  if (choice.message.refusal) throw new ToolFailure('MODEL_REFUSED');
  if (choice.finish_reason === 'length') throw new ToolFailure('MODEL_TRUNCATED');
  const calls = choice.message.tool_calls ?? [];
  if (choice.finish_reason === 'stop') {
    if (calls.length || !choice.message.content?.trim()) throw new ToolFailure('TOOL_PROTOCOL_ERROR');
    return { kind: 'answer', text: choice.message.content };
  }
  if (!calls.length || new Set(calls.map(call => call.id)).size !== calls.length) {
    throw new ToolFailure('TOOL_PROTOCOL_ERROR');
  }
  return { kind: 'tools', calls, content: choice.message.content ?? null };
}
```
