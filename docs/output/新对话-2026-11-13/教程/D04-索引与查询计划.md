## 一、知识点与推导

B-tree 按列顺序排列。公司列表有 company_id 等值条件，后接 created_at、id 排序即可连续读取。部门待派单还有 status 的等值条件，因此用另一组复合索引。小表顺序扫描合理；实验使用两万张数据和两个不同的客户分布。N+1 与索引是两类问题，网络往返次数可以通过 query 事件计数。

## 二、动手开发

在尚未部署本课索引的 D03 库依次运行脚本 compare-worklists.sql：记录未建索引时的计划，然后对照临时索引。运行 pnpm exec tsx prisma/scripts/compare-queries.ts，比较逐单取材料和一次汇总。把正式索引加入 schema 和迁移，用 migrate deploy 应用。最后新增列表入口，使两个筛选分别返回 1、2 条。这里的查询参数用于索引教学，生产身份范围在 W3 由服务端提供。

```sql
INSERT INTO tickets (id,ticket_no,title,description,company_id,category_id,department_id,reporter_id,status,created_at)

 CASE WHEN n % 23 = 0 THEN 'pending_assignment' ELSE 'agent_processing' END,

FROM generate_series(1,20000) AS series(n)
JOIN tickets t ON t.ticket_no = CASE WHEN n % 41 = 0 THEN 'T-002' ELSE 'T-001' END;
ANALYZE tickets;
EXPLAIN (ANALYZE, BUFFERS)
```

**apps/server/prisma/scripts/compare-queries.ts**

```ts
import { PrismaClient } from '../../src/generated/prisma/client.js';
import { PrismaPg } from '@prisma/adapter-pg';
import assert from 'node:assert/strict';

const db = new PrismaClient({
  adapter: new PrismaPg({connectionString: process.env.DATABASE_URL}),
  log: [{ emit: 'event', level: 'query' }],
});
let count = 0;
db.$on('query', () => { count++; });

async function compare() {
  const rows = await db.ticket.findMany({select:{id:true, ticketNo:true},orderBy:{ticketNo:'asc'}});
  const slow = [];
  for (const row of rows) {
    const logs = await db.ticketMaterial.findMany({ where:{ ticketId:row.id } });
    slow.push(logs.length);
  }
  const first = count;
  count = 0;
  const batched = await db.ticket.findMany({select:{_count:{select:{materials:true}}},orderBy:{ticketNo:'asc'}});
  assert.deepEqual(slow,batched.map(row => row._count.materials));
  assert.ok(first > count);
  console.log({ tickets: rows.length, individual: first, batched: count });
}
compare().finally(() => db.$disconnect()).catch(error => {console.error(error);process.exitCode=1;});
```

**apps/server/src/tickets/tickets.service.ts**

```ts
import { Injectable, NotFoundException } from '@nestjs/common';
import type { Prisma } from '../generated/prisma/client.js';
import { PrismaService } from '../prisma/prisma.service.js';

const detailSelect = {
  ticketNo: true, title: true, description: true, status: true, createdAt: true,
  company: { select: { name: true } }, department: { select: { name: true } },
  category: { select: { name: true } }, reporter: { select: { displayName: true } },
  assignee: { select: { displayName: true } },
  materials: { select: { logText: true }, orderBy: { id: 'asc' } },
  events: { orderBy: [{ createdAt: 'asc' }, { id: 'asc' }],
    select: { action: true, toStatus: true, actor: { select: { displayName: true } } } },
} satisfies Prisma.TicketSelect;

export type TicketDetail = Prisma.TicketGetPayload<{ select: typeof detailSelect }>;

@Injectable()
export class TicketsService {
  constructor(private readonly prisma: PrismaService) {}
  async list(company?: string, department?: string) {
    const items = await this.prisma.ticket.findMany({
      where: { company: company ? { name: company } : undefined,
        department: department ? { name: department } : undefined },
      orderBy: [{ createdAt: 'desc' }, { id: 'desc' }], take: 20,
      select: { ticketNo: true, title: true, status: true, createdAt: true },
    });
    return { items };
  }

  async findDetail(ticketNo: string): Promise<TicketDetail> {
    const ticket = await this.prisma.ticket.findUnique({ where: { ticketNo }, select: detailSelect });
    if (!ticket) throw new NotFoundException('工单不存在');
    return ticket;
  }
}
```
