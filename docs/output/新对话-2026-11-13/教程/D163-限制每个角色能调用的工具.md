## 一、知识点与推导

每次调用在执行端校验角色白名单及当前主体，拒绝后保留审计。

两侧使用同一个数据集、工具授权和模型配置。逐题保存 mode、round、实际工具序列、业务状态、耗时和 token 数；按题号和轮次做配对。先核对违规，再讨论调用代价。

多角色组的分诊可读取摘要，诊断可读取告警计数，复核只看交接。单角色组有这两个读工具，比较时不会因为某一组额外获得写操作而失真。

角色来自运行调度器。grantRead 检查静态允许集合，再调用现有 issueGrant 绑定用户、运行和工具。原服务还要检查数据库中的有效权限。

**apps/server/src/agent/team/authority.spec.ts**

```ts
import { checkGrant } from '../diagnosis-grant.js';
import { grantRead } from './authority.js';

describe('角色授权', () => {
  it('分诊只能取摘要而复核不能读取工具', () => {
    expect(() => grantRead('triage', 'u', 'r', 'ticket_log_counts')).toThrow('没有该工具');
    expect(() => grantRead('review', 'u', 'r', 'ticket_summary')).toThrow('没有该工具');
    expect(() => grantRead('diagnosis', 'u', 'r', 'ticket_summary')).toThrow('没有该工具');
    expect(() => grantRead('diagnosis', 'u', 'r', 'ticket_log_counts')).not.toThrow();
  });
  it('单项授权仍绑定原用户和运行', () => {
    const handle = grantRead('single', 'u', 'r', 'ticket_summary');
    expect(() => checkGrant(handle, 'u', 'r', 'ticket_summary')).not.toThrow();
    expect(() => checkGrant(handle, 'u', 'other', 'ticket_summary')).toThrow();
    expect(() => checkGrant(handle, 'other', 'r', 'ticket_summary')).toThrow();
    expect(() => checkGrant(handle, 'u', 'r', 'ticket_log_counts')).toThrow();
  });
});
```

**apps/server/src/agent/team/authority.ts**

```ts
import { issueGrant } from '../diagnosis-grant.js';

export type Role = 'single' | 'triage' | 'diagnosis' | 'review';
export type ReadName = 'ticket_summary' | 'ticket_log_counts';
const allowed: Record<Role, readonly ReadName[]> = {
  single: ['ticket_summary', 'ticket_log_counts'], triage: ['ticket_summary'],
  diagnosis: ['ticket_log_counts'], review: [],
};

// 角色由服务端调度器选择；每次只签发本次读取所需的单项授权。
export function grantRead(role: Role, userId: string, runId: string, name: ReadName) {
  if (!allowed[role].includes(name)) throw new Error('角色没有该工具');
  return issueGrant(userId, runId, [name]);
}
```
