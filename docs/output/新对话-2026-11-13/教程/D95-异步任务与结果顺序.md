### 1.1 先建立全部任务，再按输入收集

TaskGroup中按输入创建任务并保存句柄；离开上下文后，正常完成的任务才用result读取。返回顺序由句柄列表决定，不能按完成先后重新排列。用Event让B先于A完成以检查关联。

**可运行示例：test_group_keeps_input_order**

来源：`python/bridge/tests/test_execution.py`。

```python
def test_group_keeps_input_order() -> None:
    async def scenario() -> None:
        second = asyncio.Event()
        done: list[str] = []

        async def worker(ticket: TicketSnapshot) -> str:
            if ticket.ticket_no == "A":
                await second.wait()
            done.append(ticket.ticket_no)
            second.set()
            return ticket.ticket_no

        assert await run_all(TICKETS, worker) == ("A", "B")
        assert done == ["B", "A"]

    asyncio.run(scenario())
```

### 1.2 整批失败的生命周期

run_all中某个普通异常会让TaskGroup取消其余未完成任务，等待清理后以ExceptionGroup报告。finally用于释放资源。这里“整批失败”只约束返回结果，不会回滚已经发生的外部写入。依据：[TaskGroup官方说明](https://docs.python.org/3.12/library/asyncio-task.html#task-groups)。

**可运行示例：test_group_failure_waits_for_sibling_cleanup**

来源：`python/bridge/tests/test_execution.py`。

```python
def test_group_failure_waits_for_sibling_cleanup() -> None:
    async def scenario() -> None:
        started = asyncio.Event()
        cleaned: list[str] = []

        async def worker(ticket: TicketSnapshot) -> str:
            if ticket.ticket_no == "B":
                await started.wait()
                raise ValueError("模拟失败")
            try:
                started.set()
                await asyncio.Event().wait()
                return "不可到达"
            finally:
                cleaned.append("A")

        with pytest.raises(ExceptionGroup) as caught:
            await run_all(TICKETS, worker)
        assert any(isinstance(error, ValueError) for error in caught.value.exceptions)
        assert cleaned == ["A"]

    asyncio.run(scenario())
```

```powershell
.venv/Scripts/python -m pytest -q tests/test_execution.py::test_group_failure_waits_for_sibling_cleanup
```

### 1.3 逐项结果与取消分开

run_partial只捕获Exception，普通异常转成error项，成功项保留。CancelledError继承BaseException，不把它吞成业务结果。错误报告仅保存类名，避免把异常正文直接返回。子任务自行取消时，TaskGroup不会因此取消其余任务；收集其result会传播取消。

**可运行示例：test_partial_retains_success_and_redacts_error_message**

来源：`python/bridge/tests/test_execution.py`。

```python
def test_partial_retains_success_and_redacts_error_message() -> None:
    async def worker(ticket: TicketSnapshot) -> str:
        if ticket.ticket_no == "B":
            raise ValueError("此正文不应进入报告")
        return "成功"

    result = asyncio.run(run_partial(TICKETS, worker))
    assert [(row.ticket_no, row.status) for row in result] == [("A", "ok"), ("B", "error")]
    assert result[0].text == "成功"
    assert result[1].error == "ValueError"
    assert result[1].text is None
```

## 四、本章完整源码

**python/bridge/ops_bridge/execution.py**

```python
"""区分整批失败与逐项保留结果；取消始终向上传播。"""
import asyncio
from dataclasses import dataclass
from typing import Literal

from ops_bridge.contracts import TicketWorker
from ops_bridge.models import TicketSnapshot


@dataclass(frozen=True, slots=True)
class ItemResult:
    ticket_no: str
    status: Literal["ok", "error"]
    text: str | None = None
    error: str | None = None


async def run_all(tickets: tuple[TicketSnapshot, ...], worker: TicketWorker) -> tuple[str, ...]:
    async with asyncio.TaskGroup() as group:
        tasks = [group.create_task(worker(ticket)) for ticket in tickets]
    return tuple(task.result() for task in tasks)


async def run_partial(
    tickets: tuple[TicketSnapshot, ...], worker: TicketWorker
) -> tuple[ItemResult, ...]:
    async def one(ticket: TicketSnapshot) -> ItemResult:
        try:
            text = await worker(ticket)
            return ItemResult(ticket.ticket_no, "ok", text=text)
        except Exception as error:
            # 只记录错误类名，不把可能含敏感数据的异常正文带入报告。
            return ItemResult(ticket.ticket_no, "error", error=type(error).__name__)

    async with asyncio.TaskGroup() as group:
        tasks = [group.create_task(one(ticket)) for ticket in tickets]
    return tuple(task.result() for task in tasks)
```

def test_parent_cancellation_cleans_children() -> None:
    async def scenario() -> None:
        started = asyncio.Event()
        cleaned: list[str] = []

        async def worker(ticket: TicketSnapshot) -> str:
            try:
                started.set()
                await asyncio.Event().wait()
                return "不可到达"

                cleaned.append(ticket.ticket_no)

            await task
