## 一、知识点与推导

按 createdAt 降序、id 降序得到全序。边界是时间更早，或时间相同且 id 更小。多取一行判断还有下一页，返回的游标取本页最后一行。页码是位置，插入新记录会移动位置。

**apps/server/src/common/cursor.ts**

```ts
import { BadRequestException } from '@nestjs/common';

// 游标记录“上一页最后一行”在排序里的位置：按 createdAt 倒序，时间相同按 id 倒序
export interface TicketCursor {
  createdAt: Date;
  id: string;
}

// 编码成 base64url：对前端是不透明的字符串，前端只负责原样传回，不解析也不拼接
export function encodeCursor(cursor: TicketCursor): string {
  const payload = JSON.stringify({ t: cursor.createdAt.toISOString(), id: cursor.id });
  return Buffer.from(payload, 'utf8').toString('base64url');
}

// 解码并校验；游标来自客户端，属于不可信输入，格式不对一律返回 400
export function decodeCursor(raw: string): TicketCursor {
  try {
    const parsed: unknown = JSON.parse(Buffer.from(raw, 'base64url').toString('utf8'));
    if (typeof parsed === 'object' && parsed !== null && 't' in parsed && 'id' in parsed) {
      const { t, id } = parsed as { t: unknown; id: unknown };
      const createdAt = typeof t === 'string' ? new Date(t) : new Date(NaN);
      const isUuid = typeof id === 'string' && /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(id);
      if (!Number.isNaN(createdAt.getTime()) && isUuid) {
        return { createdAt, id: id as string };
      }
    }
  } catch {
    // JSON 解析失败，落到下面统一报错
  }
  throw new BadRequestException('cursor 无效');
}
```

**apps/server/src/tickets/tickets.service.ts**

```ts
import { decodeCursor, encodeCursor } from '../common/cursor.js';
import type { ListTicketsQuery } from './dto/list-tickets.query.js';
import { Injectable, NotFoundException } from '@nestjs/common';
import type { Prisma } from '../generated/prisma/client.js';
import { PrismaService } from '../prisma/prisma.service.js';

const detailSelect = {
  ticketNo: true, title: true, description: true, status: true, createdAt: true,
  company: { select: { name: true } }, department: { select: { name: true } },
  category: { select: { name: true } }, reporter: { select: { displayName: true } },
  assignee: { select: { displayName: true } },
  materials: { select: { logText: true }, orderBy: { id: 'asc' } },
  events: { orderBy: [{ createdAt: 'asc' }, { id: 'asc' }],
    select: { action: true, toStatus: true, actor: { select: { displayName: true } } } },
} satisfies Prisma.TicketSelect;

export type TicketDetail = Prisma.TicketGetPayload<{ select: typeof detailSelect }>;

@Injectable()
export class TicketsService {
  constructor(private readonly prisma: PrismaService) {}
  async list({ company, department, cursor, limit }: ListTicketsQuery) {
    const after = cursor ? decodeCursor(cursor) : undefined;
    const rows = await this.prisma.ticket.findMany({
      where: {
        company: company ? { name: company } : undefined,
        department: department ? { name: department } : undefined,
        OR: after ? [
          { createdAt: { lt: after.createdAt } },
          { createdAt: after.createdAt, id: { lt: after.id } },
        ] : undefined,
      },
      orderBy: [{createdAt:'desc'}, {id:'desc'}], take: limit + 1,
      select: { id:true, ticketNo:true, title:true, status:true, createdAt:true },
    });
    const page = rows.slice(0,limit);
    return {
      items: page.map(({ id: _id, ...item }) => item),
      nextCursor: rows.length > limit ? encodeCursor(page[page.length - 1]) : null,
    };
  }

  async findDetail(ticketNo: string): Promise<TicketDetail> {
    const ticket = await this.prisma.ticket.findUnique({ where: { ticketNo }, select: detailSelect });
    if (!ticket) throw new NotFoundException('工单不存在');
    return ticket;
  }
}
```

it('客户和部门是两个独立的筛选维度', async () => {
  const customer = await request(app.getHttpServer()).get('/api/tickets').query({ company: '甲公司' }).expect(200);
  expect(customer.body.data.items.map((r: { ticketNo: string }) => r.ticketNo)).toEqual(['T-001']);
  const department = await request(app.getHttpServer()).get('/api/tickets').query({ department: '网络部' }).expect(200);
  expect(department.body.data.items).toHaveLength(2);

it('同一时间的两张工单能逐页读出并拒绝无效游标', async () => {
  const http = request(app.getHttpServer());
  const first = (await http.get('/api/tickets?limit=1').expect(200)).body.data;
  expect(first.items).toHaveLength(1);
  expect(first.nextCursor).toEqual(expect.any(String));
  const second = (await http.get('/api/tickets').query({ limit: 1, cursor: first.nextCursor }).expect(200)).body.data;
  expect(second.items).toHaveLength(1);
  expect(new Set([first.items[0].ticketNo, second.items[0].ticketNo]).size).toBe(2);
  expect(second.nextCursor).toBeNull();
  await http.get('/api/tickets?cursor=broken').expect(400);
});

**apps/server/prisma/sql/show-offset-shift.sql**

```sql
-- 在 D02 的两行种子上比较两种分页。同一事务结束时撤销插入。
BEGIN;
CREATE TEMP TABLE first_page ON COMMIT DROP AS
SELECT id,created_at,ticket_no FROM tickets ORDER BY created_at DESC,id DESC LIMIT 1;
INSERT INTO tickets (id,ticket_no,title,description,company_id,category_id,department_id,reporter_id,created_at)
SELECT gen_random_uuid(),'PAGE-NEW',title,description,company_id,category_id,department_id,reporter_id,
created_at+interval '1 minute' FROM tickets WHERE ticket_no='T-001';
SELECT 'offset',ticket_no FROM tickets ORDER BY created_at DESC,id DESC LIMIT 1 OFFSET 1;
SELECT 'cursor',t.ticket_no FROM tickets t,first_page f
WHERE (t.created_at,t.id)<(f.created_at,f.id) ORDER BY t.created_at DESC,t.id DESC LIMIT 1;
SELECT 'first',ticket_no FROM first_page;
ROLLBACK;
```
