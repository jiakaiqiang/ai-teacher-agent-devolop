**apps/server/prisma/scripts/rate-limit-demo.ts**

```ts
// 三种限流算法对比（D26）：同样“每秒 5 次”，在窗口边界前后各来 5 个请求，各放行几个
// 运行：pnpm exec tsx prisma/scripts/rate-limit-demo.ts（只连开发用的 Redis，键前缀 demo:，结束时清理）
import 'dotenv/config';
import { randomUUID } from 'node:crypto';
import { setTimeout as sleep } from 'node:timers/promises';
import { Redis } from 'ioredis';

const redis = new Redis(process.env.REDIS_URL!, { keyPrefix: 'demo:' });
const LIMIT = 5;
const WINDOW_MS = 1000;
const NOW = `local t = redis.call('time') local now = tonumber(t[1]) * 1000 + math.floor(tonumber(t[2]) / 1000)`;

// 固定窗口：按“第几秒”分桶计数，每个桶独立
const FIXED = `${NOW}
local key = KEYS[1] .. ':' .. math.floor(now / tonumber(ARGV[1]))
local n = redis.call('incr', key)
if n == 1 then redis.call('pexpire', key, ARGV[1]) end
return n <= tonumber(ARGV[2]) and 1 or 0`;

// 滑动窗口：和 RateLimitService.consume 同一个算法
const SLIDING = `${NOW}
redis.call('zremrangebyscore', KEYS[1], '-inf', now - tonumber(ARGV[1]))
if redis.call('zcard', KEYS[1]) < tonumber(ARGV[2]) then
  redis.call('zadd', KEYS[1], now, ARGV[3])
  redis.call('pexpire', KEYS[1], ARGV[1])
  return 1
end
return 0`;

// 令牌桶：桶里最多 capacity 个令牌，每毫秒补 rate 个；请求拿走一个令牌，没有令牌就拒绝
const BUCKET = `${NOW}
local capacity = tonumber(ARGV[1])
local rate = tonumber(ARGV[2])
local b = redis.call('hmget', KEYS[1], 'tokens', 'ts')
local tokens = tonumber(b[1]) or capacity
local ts = tonumber(b[2]) or now
tokens = math.min(capacity, tokens + (now - ts) * rate)
local allowed = 0
if tokens >= 1 then
  tokens = tokens - 1
  allowed = 1
end
redis.call('hset', KEYS[1], 'tokens', tokens, 'ts', now)
redis.call('pexpire', KEYS[1], math.ceil(capacity / rate))
return allowed`;

const algorithms = {
  固定窗口: () => redis.eval(FIXED, 1, 'fixed', WINDOW_MS, LIMIT),
  滑动窗口: () => redis.eval(SLIDING, 1, 'sliding', WINDOW_MS, LIMIT, randomUUID()),
  令牌桶: () => redis.eval(BUCKET, 1, 'bucket', LIMIT, LIMIT / WINDOW_MS),
};

// 等到某一秒的第 ms 毫秒（用 Redis 服务器的时间，和脚本里的时间一致）
async function waitUntilMsOfSecond(ms: number) {
  const [, micro] = await redis.time();
  const current = Math.floor(Number(micro) / 1000);
  await sleep((ms - current + 1000) % 1000);
}

async function burst(run: () => Promise<unknown>) {
  let ok = 0;
  for (let i = 0; i < 5; i++) if ((await run()) === 1) ok++;
  return ok;
}

try {
  for (const [name, run] of Object.entries(algorithms)) {
    await waitUntilMsOfSecond(850);
    const first = await burst(run);
    await waitUntilMsOfSecond(50); // 下一秒的第 50ms，离上一批只隔 200ms
    const second = await burst(run);
    console.log(`${name}：第 850ms 来 5 个放行 ${first} 个，200ms 后（跨过整秒）再来 5 个放行 ${second} 个，合计 ${first + second} 个`);
    await sleep(1500);
  }

  console.log('\n令牌桶的匀速请求：每 200ms 来 1 个，共 10 个');
  await redis.del('bucket');
  let ok = 0;
  for (let i = 0; i < 10; i++) {
    if ((await algorithms.令牌桶()) === 1) ok++;
    await sleep(200);
  }
  console.log(`  放行 ${ok} 个`);
} finally {
  const keys = (await redis.keys('demo:fixed:*')).map((k) => k.slice('demo:'.length));
  await redis.del('sliding', 'bucket', ...keys);
  redis.disconnect();
}
```

**apps/server/src/redis/rate-limit.service.ts**

```ts
import { Injectable, Logger } from '@nestjs/common';
import { randomUUID } from 'node:crypto';
import { RedisService } from './redis.service.js';

export interface RateLimitResult {
  allowed: boolean;
  // 被拒绝时，多少秒后可以再试（向上取整，至少 1）
  retryAfterSeconds: number;
}

// 计数加一；第一次计数时设置过期时间。两步放在一个脚本里：INCR 之后进程崩了，也不会留下永不过期的计数
const COUNT_SCRIPT = `
local n = redis.call('incr', KEYS[1])
if n == 1 then redis.call('pexpire', KEYS[1], ARGV[1]) end
return n`;

// 滑动窗口：有序集合里每个成员是一次请求，分数是请求时间（毫秒）
// 先删掉窗口之外的旧请求，再数窗口里还有几个；时间用 Redis 服务器的，多个实例的时钟不一致也没关系
const SLIDING_WINDOW_SCRIPT = `
local t = redis.call('time')
local now = tonumber(t[1]) * 1000 + math.floor(tonumber(t[2]) / 1000)
local window = tonumber(ARGV[1])
local limit = tonumber(ARGV[2])
redis.call('zremrangebyscore', KEYS[1], '-inf', now - window)
if redis.call('zcard', KEYS[1]) < limit then
  redis.call('zadd', KEYS[1], now, ARGV[3])
  redis.call('pexpire', KEYS[1], window)
  return 0
end
local oldest = redis.call('zrange', KEYS[1], 0, 0, 'WITHSCORES')
return tonumber(oldest[2]) + window - now`;

// 限流（D26）。Redis 出错时一律放行：限流是保护措施，Redis 挂了不能让登录、评论都不可用
@Injectable()
export class RateLimitService {
  private readonly logger = new Logger(RateLimitService.name);

  constructor(private readonly redis: RedisService) {}

  // 滑动窗口：任意 windowMs 毫秒内最多 limit 次，这一次算在里面
  async consume(key: string, limit: number, windowMs: number): Promise<RateLimitResult> {
    try {
      const waitMs = Number(await this.redis.eval(SLIDING_WINDOW_SCRIPT, 1, key, windowMs, limit, randomUUID()));
      return waitMs > 0
        ? { allowed: false, retryAfterSeconds: Math.max(1, Math.ceil(waitMs / 1000)) }
        : { allowed: true, retryAfterSeconds: 0 };
    } catch (error) {
      this.logger.warn(`限流检查失败，放行：${key}，${(error as Error).message}`);
      return { allowed: true, retryAfterSeconds: 0 };
    }
  }

  // 固定窗口计数（登录失败次数）：从第一次计数开始算 windowMs，到期后整体清零
  async count(key: string, windowMs: number): Promise<void> {
    try {
      await this.redis.eval(COUNT_SCRIPT, 1, key, windowMs);
    } catch (error) {
      this.logger.warn(`计数失败：${key}，${(error as Error).message}`);
    }
  }

  // 当前计数是否已经达到 limit；达到时返回还要等多少秒
  async check(key: string, limit: number): Promise<RateLimitResult> {
    try {
      const [count, ttlMs] = await Promise.all([this.redis.get(key), this.redis.pttl(key)]);
      return Number(count ?? 0) >= limit && ttlMs > 0
        ? { allowed: false, retryAfterSeconds: Math.max(1, Math.ceil(ttlMs / 1000)) }
        : { allowed: true, retryAfterSeconds: 0 };
    } catch (error) {
      this.logger.warn(`限流检查失败，放行：${key}，${(error as Error).message}`);
      return { allowed: true, retryAfterSeconds: 0 };
    }
  }

  async reset(key: string): Promise<void> {
    try {
      await this.redis.del(key);
    } catch (error) {
      this.logger.warn(`清除计数失败：${key}，${(error as Error).message}`);
    }
  }
}
```

import { HttpException, HttpStatus } from '@nestjs/common';

// 429 请求太频繁（D26）。全局异常过滤器会把 retryAfterSeconds 写进 Retry-After 响应头和 error.details
export class RateLimitedException extends HttpException {
  constructor(
    readonly retryAfterSeconds: number,
    message = '请求太频繁，请稍后再试',
  ) {
    super({ code: 'RATE_LIMITED', message }, HttpStatus.TOO_MANY_REQUESTS);
  }
}
