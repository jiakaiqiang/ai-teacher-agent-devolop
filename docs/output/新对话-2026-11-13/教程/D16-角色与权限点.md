## 零、闭卷回忆

客户管理员能代提单人验收吗？审批用户与审核知识由谁操作？业务角色与关联表权限各负责哪个判断？

## 一、知识点与推导

身份通过后由权限守卫查询获批用户的角色权限。未声明权限的入口默认拒绝。动作还需检查原始状态、操作者与工单之间的关系，下一天实现列表和详情的数据范围。

## 二、动手开发

先让普通提单人调用分类管理接口，预期 403。为四种角色列权限矩阵，再生成迁移；用关联表映射权限，用守卫处理接口声明。变更用户主角色时检查旧关联和新关联。

## 三、自检与提交

运行 test:e2e，并查询 customer_admin 的权限，结果只应包含 registration:approve 和 ticket:read。口述为什么看到部门数据不等于获得派单权限。

**apps/server/src/auth/permissions.service.ts**

```ts
import { Injectable } from '@nestjs/common';
import type { Permission } from '@ops-ticket/shared';
import { PrismaService } from '../prisma/prisma.service.js';

@Injectable()
export class PermissionsService {
  constructor(private readonly prisma: PrismaService) {}

  // 用户通过所有角色得到的权限点，去重、排序；用户不存在时返回 null（令牌还没过期，但账号已经删了）
  // 每个请求查一次数据库：改了角色立即生效，不用等令牌过期。W4 有了 Redis 后可以加缓存
  async permissionsOf(userId: string): Promise<Permission[] | null> {
    const user = await this.prisma.user.findFirst({
      where: { id: userId, registrationStatus: "approved" },
      select: {
        roles: { select: { role: { select: { permissions: { select: { permission: { select: { code: true } } } } } } } },
      },
    });
    if (!user) {
      return null;
    }
    const codes = user.roles.flatMap((ur) => ur.role.permissions.map((rp) => rp.permission.code as Permission));
    return [...new Set(codes)].sort();
  }
}
```

**apps/server/src/auth/permissions.guard.ts**

```ts
import { CanActivate, ExecutionContext, ForbiddenException, Injectable, Logger, UnauthorizedException } from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import type { Permission } from '@ops-ticket/shared';
import type { AuthenticatedRequest } from './auth.types.js';
import { LOGIN_ONLY, REQUIRED_PERMISSIONS } from './permissions.decorator.js';
import { PermissionsService } from './permissions.service.js';
import { IS_PUBLIC } from './public.decorator.js';

// 全局权限守卫（D16），在 JwtAuthGuard 之后执行（两个守卫在 AuthModule 里按这个顺序注册）
// 默认拒绝：接口必须声明 @Public()、@LoginOnly() 或 @RequirePermission() 之一，忘了写就是 403
@Injectable()
export class PermissionsGuard implements CanActivate {
  private readonly logger = new Logger(PermissionsGuard.name);

  constructor(
    private readonly reflector: Reflector,
    private readonly permissions: PermissionsService,
  ) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const targets = [context.getHandler(), context.getClass()];
    if (this.reflector.getAllAndOverride<boolean>(IS_PUBLIC, targets)) {
      return true;
    }

    const request = context.switchToHttp().getRequest<AuthenticatedRequest>();
    const user = request.user!; // JwtAuthGuard 已经保证登录
    const permissions = await this.permissions.permissionsOf(user.id);
    if (!permissions) {
      // 令牌有效，但用户已经被删除：按“未登录”处理，而不是“没有权限”
      throw new UnauthorizedException({ code: 'UNAUTHENTICATED', message: '用户不存在，请重新登录' });
    }
    // 查出来挂在请求上，Service 里需要判断权限时不用再查（D17）
    user.permissions = permissions;

    if (this.reflector.getAllAndOverride<boolean>(LOGIN_ONLY, targets)) {
      return true;
    }
    const required = this.reflector.getAllAndOverride<Permission[] | undefined>(REQUIRED_PERMISSIONS, targets);
    if (!required) {
      // 这是代码写漏了，不是用户的问题：写错误日志，方便在开发和测试阶段发现
      this.logger.error(`接口没有声明权限：${context.getClass().name}.${context.getHandler().name}`);
      throw new ForbiddenException({ code: 'PERMISSION_NOT_DECLARED', message: '接口没有声明所需权限' });
    }
    const missing = required.filter((p) => !user.permissions!.includes(p));
    if (missing.length > 0) {
      throw new ForbiddenException({ code: 'FORBIDDEN', message: `没有权限：${missing.join('、')}` });
    }
    return true;
  }
}
```

```ts
import { RequirePermission } from '../auth/permissions.decorator.js';
import { BadRequestException, Body, Controller, ForbiddenException, Param, Post, Req } from '@nestjs/common';
import { IsString, Length } from 'class-validator';
import { PrismaService } from '../prisma/prisma.service.js';
import { hashPassword } from './password.js';
import { Public } from './public.decorator.js';
import type { AuthenticatedRequest } from './auth.types.js';

class RegisterDto {
  @IsString() @Length(1,64) username!:string;
  @IsString() @Length(8,128) password!:string;
  @IsString() company!:string;
}
@Controller('registrations')
export class RegistrationsController {
  constructor(private readonly db:PrismaService) {}
  @Public() @Post()
  async register(@Body() dto:RegisterDto) {
    const company=await this.db.company.findUnique({where:{name:dto.company}});
    if (!company) throw new BadRequestException('客户公司不存在');
    return this.db.user.create({data:{
      username:dto.username,displayName:dto.username,passwordHash:await hashPassword(dto.password),
      companyId:company.id,role:'reporter',registrationStatus:'pending',
    },select:{id:true,registrationStatus:true}});
  }
  @RequirePermission('registration:approve')
@Post(':id/approve')
  async approve(@Req() req:AuthenticatedRequest,@Param('id') id:string) {
    const admin=await this.db.user.findFirst({where:{id:req.user!.id,role:'customer_admin',registrationStatus:'approved'}});
    if (!admin?.companyId) throw new ForbiddenException();
    const changed=await this.db.user.updateMany({where:{id,role:'reporter',companyId:admin.companyId,registrationStatus:'pending'},
      data:{registrationStatus:'approved'}});
    if (!changed.count) throw new BadRequestException('本公司没有这项待审申请');
    return {approved:true};
  }
}
```

```ts
import { LoginOnly } from '../auth/permissions.decorator.js';
import { RequirePermission } from '../auth/permissions.decorator.js';
import { BadRequestException, Body, Controller, Headers, Param, Post, Req } from '@nestjs/common';
import { IsArray, IsIn, IsInt, IsOptional, IsString, Min } from 'class-validator';
import { TicketCommands } from './ticket-commands.js';
import { SubmitTicket } from './submit-ticket.js';
import type { AuthenticatedRequest } from '../auth/auth.types.js';
import type { Action } from './workflow.js';

class ReportDto {
  @IsString() title!:string;
  @IsString() description!:string;
  @IsString() category!:string;
  @IsArray() @IsString({each:true}) logs!:string[];
}
class ActDto {
  @IsIn(['request_human','unresolved','assign','handle','approve','return','resolved']) action!:Action;
  @IsInt() @Min(0) version!:number;
  @IsOptional() @IsString() targetId?:string;
  @IsOptional() @IsString() comment?:string;
}
@Controller('tickets')
export class TicketWriteController {
  constructor(private readonly cmd:TicketCommands,
    private readonly reports:SubmitTicket) {}
  @RequirePermission('ticket:report')
  @Post()
  async submit(@Req() req:AuthenticatedRequest, @Headers('idempotency-key') key:string, @Body() dto:ReportDto) {
    if (!key) throw new BadRequestException('缺少请求键');
    const row=await this.reports.submit(req.user!.id,key,dto);
    return {ticketNo:row.ticketNo,status:row.status,version:row.version};
  }
  @LoginOnly()
  @Post(':no/actions')
  async act(@Param('no') no:string,@Req() req:AuthenticatedRequest,@Body() dto:ActDto) {
    return this.cmd.apply(no,{source:'user',id:req.user!.id},dto.action,dto.version,dto.targetId,dto.comment);
  }
}
```
