## 一、知识点与推导

保存配置、实际消耗、停止原因和已完成的工具步骤。

在两个浏览器会话分别以甲、乙公司提单人身份操作。先记下工单号、工单版本，再创建 run。按序号读取 events，按步骤关联 traces。把另一人的 runId 代入同一个接口，保存拒绝结果。

预算有三项：步数（`maxSteps`，默认 8）、挂钟时间（`maxMs`，默认 60 秒）、token（`maxTokens`，默认 20000）。每项都有硬上限（8 / 120 秒 / 60000），`parseBudget` 把非法值（0、超限、小数、NaN）直接拒成 `RUN_BUDGET_INVALID`，不做「悄悄截断到合法范围」——调用方传错了应该知道。判定只有一个纯函数 `exhausted(budget, spend)`：返回 `null` 表示还能走，否则返回该报的停止原因。顺序固定为**时间 → token → 步数**，三项同时用完时报最紧的那个，这样统计里不会出现同一场景有时报 A 有时报 B。

循环的形状是：**先检查预算 → 再走一步 → 记账 → 判断是否答出**。检查必须在开步**之前**，所以 `>=` 而不是 `>`：已经走了 8 步、上限 8 时就不能再开第 9 步。反证 A 把 `>=` 改成 `>`，立刻多走一步。时间同理：每次循环开头重新算 `elapsedMs`，超了就停，而不是等这一步跑完才发现。单步还有自己的时限 `stepDeadlineMs`：取「整个 Run 的剩余时间」和「单步上限 20 秒」中较小的那个，剩余时间为 0 时抛 `RUN_TIME_EXHAUSTED`——不能让一个步骤的超时比整个运行的预算还长（对照 D102 的分层超时：**外层比内层长，内层不能超出外层剩余**）。

`runLoop` 里 token 用 `Math.max(0, Math.trunc(...))` 累加，防止负数或小数把账算花。关键一条：**单步抛异常时也要先把步数加一**，再写失败步骤和事件，然后以 `model_failed` 结束。如果异常不记账，循环会以为「这一步没发生」，在反复失败的场景里原地打转——这正是文章里说的死循环。反证 B 删掉这两行，断言「异常也记一步」的用例失败。另外区分两种失败：`think` 返回 `failure` 字段是**工具失败**（`tool_failed`，这一步正常结束但结果是失败），`think` 抛异常是**模型或框架失败**（`model_failed`）。两者的运维含义不同。

```ts
import { BudgetError, budgetReasons, defaultBudget, exhausted, isBudgetReason, parseBudget,
  stepDeadlineMs } from './run-budget.js';
import { runLoop, type StepOutcome, type StepRequest } from './run-loop.js';

const spend = (overrides: Partial<{ steps: number; elapsedMs: number; tokens: number }> = {}) =>
  ({ steps: 0, elapsedMs: 0, tokens: 0, ...overrides });

describe('三种预算', () => {
  it('默认值与上限都有边界，非法输入直接拒绝', () => {
    expect(parseBudget()).toEqual(defaultBudget);
    expect(parseBudget({ maxSteps: 3 })).toEqual({ ...defaultBudget, maxSteps: 3 });
    for (const bad of [{ maxSteps: 0 }, { maxSteps: 9 }, { maxMs: 120_001 }, { maxTokens: 60_001 },
      { maxMs: 1.5 }, { maxTokens: Number.NaN }]) {
      expect(() => parseBudget(bad)).toThrow(BudgetError);
    }
  });

  it('任意一项用完都停，并给出对应原因', () => {
    expect(exhausted(defaultBudget, spend())).toBeNull();
    expect(exhausted(defaultBudget, spend({ steps: 7, elapsedMs: 59_999, tokens: 19_999 }))).toBeNull();
    expect(exhausted(defaultBudget, spend({ steps: 8 }))).toBe('step_limit');
    expect(exhausted(defaultBudget, spend({ elapsedMs: 60_000 }))).toBe('time_limit');
    expect(exhausted(defaultBudget, spend({ tokens: 20_000 }))).toBe('token_limit');
    // 同时用完时报最紧的一项：时间 > token > 步数
    expect(exhausted(defaultBudget, spend({ steps: 8, elapsedMs: 60_000, tokens: 20_000 }))).toBe('time_limit');
    expect(exhausted(defaultBudget, spend({ steps: 8, tokens: 20_000 }))).toBe('token_limit');
    expect(() => exhausted(defaultBudget, spend({ steps: -1 }))).toThrow('RUN_SPEND_INVALID');
  });

  it('单步时限不超过整个运行的剩余时间', () => {
    expect(stepDeadlineMs(defaultBudget, spend())).toBe(20_000);
    expect(stepDeadlineMs(defaultBudget, spend({ elapsedMs: 55_000 }))).toBe(5_000);
    expect(stepDeadlineMs(defaultBudget, spend({ elapsedMs: 30_000 }), 45_000)).toBe(30_000);
    expect(() => stepDeadlineMs(defaultBudget, spend({ elapsedMs: 60_000 }))).toThrow('RUN_TIME_EXHAUSTED');
  });

  it('预算类停止原因与故障类分开', () => {
    expect(budgetReasons).toEqual(['step_limit', 'time_limit', 'token_limit']);
    expect(isBudgetReason('tool_failed')).toBe(false);
    expect(isBudgetReason('user_cancelled')).toBe(false);
    expect(isBudgetReason('answered')).toBe(false);
  });
});

describe('推理-行动循环', () => {
  // 固定时钟：每次读时间都前进一个固定值，耗时可预测。
  function clock(stepMs: number) {
    let value = 1_000;
    return () => { const current = value; value += stepMs; return current; };
  }
  function recorder() {
    const steps: { request: StepRequest; outcome: StepOutcome }[] = [];
    return { steps, onStep: async (outcome: StepOutcome, request: StepRequest) => { steps.push({ request, outcome }); } };
  }

  it('给出答案就停，步数与 token 据实累计', async () => {
    const log = recorder();
    const result = await runLoop({
      budget: parseBudget(), now: clock(0), onStep: log.onStep,
      think: async request => request.ordinal < 3
        ? { phase: 'tool', summary: `查第 ${request.ordinal} 项`, tokens: 100 }
        : { phase: 'answer', summary: '给出结论', tokens: 150, answer: '重新绑定动态口令' },
    });
    expect(result).toMatchObject({ reason: 'answered', answer: '重新绑定动态口令' });
    expect(result.spend).toMatchObject({ steps: 3, tokens: 350 });
    expect(log.steps.map(item => item.request.ordinal)).toEqual([1, 2, 3]);
  });

  it('步数用完时停在上限，不会多走一步', async () => {
    const log = recorder();
    const result = await runLoop({
      budget: parseBudget({ maxSteps: 2 }), now: clock(0), onStep: log.onStep,
      think: async () => ({ phase: 'tool', summary: '继续查', tokens: 10 }),
    });
    expect(result.reason).toBe('step_limit');
    expect(result.spend.steps).toBe(2);
    expect(log.steps).toHaveLength(2);
  });

  it('时间与 token 用完时各报各的原因', async () => {
    const slow = await runLoop({
      budget: parseBudget({ maxMs: 2_000 }), now: clock(900), onStep: async () => {},
      think: async () => ({ phase: 'tool', summary: '慢工具', tokens: 1 }),
    });
    expect(slow.reason).toBe('time_limit');
    expect(slow.spend.steps).toBeLessThan(8);
    const greedy = await runLoop({
      budget: parseBudget({ maxTokens: 250 }), now: clock(0), onStep: async () => {},
      think: async () => ({ phase: 'plan', summary: '长提示词', tokens: 200 }),
    });
    expect(greedy.reason).toBe('token_limit');
    expect(greedy.spend.tokens).toBe(400);
  });

  it('单步时限随剩余时间收紧', async () => {
    const log = recorder();
    // 时间只在每一步执行时前进 8 秒，便于逐步核对剩余时间。
    let elapsed = 0;
    await runLoop({
      budget: parseBudget({ maxMs: 30_000 }), now: () => 1_000 + elapsed, onStep: log.onStep,
      think: async () => { elapsed += 8_000; return { phase: 'tool', summary: '查设备', tokens: 1 }; },
    });
    expect(log.steps.map(item => item.request.deadlineMs)).toEqual([20_000, 20_000, 14_000, 6_000]);
    expect(log.steps).toHaveLength(4);
  });

  it('单步异常与工具失败分别对应不同原因，并且都记一步', async () => {
    const log = recorder();
    const thrown = await runLoop({
      budget: parseBudget(), now: clock(0), onStep: log.onStep,
      think: async () => { throw new Error('MODEL_TRUNCATED'); },
    });
    expect(thrown.reason).toBe('model_failed');
    expect(thrown.spend.steps).toBe(1);
    expect(log.steps[0]!.outcome.failure).toBe('MODEL_TRUNCATED');
    const failed = await runLoop({
      budget: parseBudget(), now: clock(0), onStep: async () => {},
      think: async () => ({ phase: 'tool', summary: '工具超时', tokens: 20, failure: 'TOOL_TIMEOUT' }),
    });
    expect(failed).toMatchObject({ reason: 'tool_failed', answer: null });
    expect(failed.spend).toMatchObject({ steps: 1, tokens: 20 });
  });
});
```

```ts
import { stopReasons, type StopReason } from './run-state.js';

// 三种预算：步数、挂钟时间、token。任意一项用完就停，并给出对应的停止原因。
export interface RunBudget { maxSteps: number; maxMs: number; maxTokens: number; }
export interface RunSpend { steps: number; elapsedMs: number; tokens: number; }

export const defaultBudget: RunBudget = { maxSteps: 8, maxMs: 60_000, maxTokens: 20_000 };
const limits: RunBudget = { maxSteps: 8, maxMs: 120_000, maxTokens: 60_000 };

export class BudgetError extends Error {
  constructor(readonly code: string) { super(code); }
}

export function parseBudget(input: Partial<RunBudget> = {}): RunBudget {
  const budget = { ...defaultBudget, ...input };
  const bounded = (value: number, max: number) =>
    Number.isSafeInteger(value) && value >= 1 && value <= max;
  if (!bounded(budget.maxSteps, limits.maxSteps) || !bounded(budget.maxMs, limits.maxMs)
    || !bounded(budget.maxTokens, limits.maxTokens)) throw new BudgetError('RUN_BUDGET_INVALID');
  return budget;
}

// 开下一步之前问一次：还有余量吗？没有就返回该报的停止原因。
export function exhausted(budget: RunBudget, spend: RunSpend): StopReason | null {
  if (!Number.isSafeInteger(spend.steps) || spend.steps < 0 || !Number.isFinite(spend.elapsedMs)
    || spend.elapsedMs < 0 || !Number.isSafeInteger(spend.tokens) || spend.tokens < 0) {
    throw new BudgetError('RUN_SPEND_INVALID');
  }
  // 顺序固定：先报时间，再报 token，最后报步数。同时用完时报告最紧的那个。
  if (spend.elapsedMs >= budget.maxMs) return 'time_limit';
  if (spend.tokens >= budget.maxTokens) return 'token_limit';
  if (spend.steps >= budget.maxSteps) return 'step_limit';
  return null;
}

// 留给单步的时间：不能超过整个 Run 的剩余时间，也给单步一个上限。
export function stepDeadlineMs(budget: RunBudget, spend: RunSpend, perStepMs = 20_000) {
  const remaining = budget.maxMs - spend.elapsedMs;
  if (remaining <= 0) throw new BudgetError('RUN_TIME_EXHAUSTED');
  return Math.min(remaining, perStepMs);
}

export function isBudgetReason(reason: StopReason) {
  return reason === 'step_limit' || reason === 'time_limit' || reason === 'token_limit';
}

export const budgetReasons = stopReasons.filter(isBudgetReason);
```

```ts
import { Injectable } from '@nestjs/common';
import type { AuthUser } from '../auth/auth.types.js';
import { AgentRunsService } from './agent-runs.service.js';
import { AgentJournalService } from './agent-journal.service.js';
import { parseBudget, type RunBudget } from './run-budget.js';
import { runLoop, type StepOutcome, type StepRequest } from './run-loop.js';
import { RunFailure } from './run-errors.js';

export type Think = (request: StepRequest, context: { ticketNo: string; goal: string }) => Promise<StepOutcome>;

@Injectable()
export class RunExecutorService {
  constructor(private readonly runs: AgentRunsService, private readonly journal: AgentJournalService) {}

  // 认领 → 逐步执行 → 把停止原因写进 Run。每一步的状态与事件由 journal 落库。
  async execute(user: AuthUser, runId: string, think: Think, budget: Partial<RunBudget> = {}) {
    const limits = parseBudget(budget);
    const claimed = await this.runs.claim(user, runId, (await this.runs.get(user, runId)).version);
    await this.journal.record(claimed.id, 'run_started',
      { ticketNo: claimed.ticketNo, maxSteps: limits.maxSteps, maxMs: limits.maxMs, maxTokens: limits.maxTokens });
    const context = { ticketNo: claimed.ticketNo, goal: claimed.goal };
    const result = await runLoop({
      budget: limits,
      think: request => think(request, context),
      onStep: async (outcome, request) => {
        const { step } = await this.journal.beginStep(claimed.id, outcome.phase, outcome.summary);
        if (step.ordinal !== request.ordinal) throw new RunFailure('RUN_STEP_ORDINAL_MISMATCH');
        await this.journal.endStep(step.id, outcome.failure
          ? { status: 'failed', errorCode: outcome.failure }
          : { status: 'succeeded' });
      },
    });
    const current = await this.runs.get(user, runId);
    const finished = await this.runs.finish(current.id, current.version, result.reason);
    await this.journal.record(finished.id, 'run_finished',
      { stopReason: result.reason, steps: result.spend.steps, tokens: result.spend.tokens,
        elapsedMs: Math.trunc(result.spend.elapsedMs), answer: result.answer });
    return { run: finished, result };
  }
}
```

**apps/server/src/agent/run-loop.ts**

```ts
import { exhausted, stepDeadlineMs, type RunBudget, type RunSpend } from './run-budget.js';
import type { StepPhase } from './agent-journal.service.js';
import type { StopReason } from './run-state.js';

export interface StepRequest { ordinal: number; deadlineMs: number; spend: RunSpend; }
export interface StepOutcome {
  phase: StepPhase;
  summary: string;
  tokens: number;
  // answer 非空表示本步给出了最终答案，循环就此结束。
  answer?: string;
  failure?: string;
}
export interface LoopHooks {
  budget: RunBudget;
  now?: () => number;
  think: (request: StepRequest) => Promise<StepOutcome>;
  onStep: (outcome: StepOutcome, request: StepRequest) => Promise<void>;
}
export interface LoopResult { reason: StopReason; spend: RunSpend; answer: string | null; }

// 推理-行动循环：每一步之前检查三种预算，任何一项用完就停并给出原因。
// 这里不碰数据库，也不认识模型：think 由调用方注入（测试注入假实现，D111 注入真实链路）。
export async function runLoop(hooks: LoopHooks): Promise<LoopResult> {
  const now = hooks.now ?? (() => Date.now());
  const startedAt = now();
  const spend: RunSpend = { steps: 0, elapsedMs: 0, tokens: 0 };
  for (;;) {
    spend.elapsedMs = now() - startedAt;
    const stop = exhausted(hooks.budget, spend);
    if (stop) return { reason: stop, spend, answer: null };
    const request: StepRequest = {
      ordinal: spend.steps + 1,
      deadlineMs: stepDeadlineMs(hooks.budget, spend),
      spend: { ...spend },
    };
    let outcome: StepOutcome;
    try {
      outcome = await hooks.think(request);
    } catch (error) {
      // 单步异常也要先记账：这一步已经发生过，步数和耗时都算。
      spend.steps += 1;
      spend.elapsedMs = now() - startedAt;
      await hooks.onStep({ phase: 'plan', summary: '单步异常', tokens: 0,
        failure: error instanceof Error ? error.message : 'STEP_FAILED' }, request);
      return { reason: 'model_failed', spend, answer: null };
    }
    spend.steps += 1;
    spend.tokens += Math.max(0, Math.trunc(outcome.tokens));
    spend.elapsedMs = now() - startedAt;
    await hooks.onStep(outcome, request);
    if (outcome.failure) return { reason: 'tool_failed', spend, answer: null };
    if (outcome.answer) return { reason: 'answered', spend, answer: outcome.answer };
  }
}
```
