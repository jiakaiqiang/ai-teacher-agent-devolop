## 一、知识点与推导

### 1.1 卡片必须有自己的契约

tool/citation/result沿用version/runId/messageId/seq。cardId稳定；工具先running后completed/failed，不允许终态回到running，同ID不跨类型复用。schema限制标题、片段、数组大小，Generation最多12张卡。

**推导题：为什么卡片不能直接放任意组件名和props？**

<details><summary>答案</summary>

模型输入不能决定任意组件和事件处理器。受控type映射固定组件与文本字段，避免越过渲染和执行边界。

</details>

### 1.2 展示不代表真实执行

mockCards只在mock模式、done之前插入固定演示事件，所有卡片显示模拟标签；真实mode:model拒绝此类模拟卡片。没有设备请求、知识库查询或工具副作用。W7后端协议不变。

**推导题：工具卡片completed证明了什么？**

<details><summary>答案</summary>

只证明本地模拟状态转换完成，不证明设备健康、真实工具成功或资料可信。引用标题也不是自动生成的事实背书。

</details>

### 1.3 首字与速率计量

使用单调时钟记录开始、首个非空text、终止；空文本不触发首字。字符按Unicode码点计数，速率=总字符数/(终止-首字)，没有文本或时间跨度为0返回null。此TTFT测接收回调，不是首像素呈现，也不是模型纯推理延迟。

**推导题：为什么不能把字符/秒写成TPS？**

<details><summary>答案</summary>

tokens取决于分词器和供应商usage，中文、emoji与UTF16长度均不同。没有真实token数量就只能报告字符速率。

</details>

### 1.4 中断率的分母必须可解释

仅本页已结束且非主动停止的请求进入分母；error/interrupted为分子，completed/truncated不是中断；进行中不计，分母0返回null。终态只结算一次。每个attempt持有自己的timer，finally总清自己的timer，仅当前attempt可清共享控制器。

**推导题：截断算成功还是错误？**

<details><summary>答案</summary>

本UI单列truncated，不能说完整成功；在本章中断率中它不是网络/协议中断，但可另定义截断率。业务口径必须写明。

</details>

```ts
import { conversationEventSchema, type ConversationEvent } from '@ops-ticket/shared'

export type GenerationStatus = 'connecting' | 'streaming' | 'completed' | 'truncated' | 'stopped' | 'interrupted' | 'error'
export type AssistantCard = Extract<ConversationEvent, {type:'tool'|'citation'|'result'}>
export interface Generation {
  cards: AssistantCard[]
  runId: string
  messageId: string
  status: GenerationStatus
  mode: 'mock' | 'model' | null
  text: string
  error: string | null
  lastSeq: number
  seen: Record<number, string>
}
export class ConversationError extends Error {}
export const isGenerating = (run: Generation) => run.status === 'connecting' || run.status === 'streaming'
export function createGeneration(runId: string, messageId: string): Generation {
  return { cards: [], runId, messageId, status: 'connecting', mode: null, text: '', error: null, lastSeq: 0, seen: {} }
}
export function stopGeneration(run: Generation) { if (isGenerating(run)) run.status = 'stopped' }
export function failGeneration(run: Generation, code: string) {
  if (!isGenerating(run)) return
  run.status = run.text ? 'interrupted' : 'error'; run.error = code
}

// 校验全部通过后才提交状态；重复事件去重不等于请求幂等或断线恢复。
export function applyConversationEvent(run: Generation, input: unknown): 'applied' | 'duplicate' | 'stale' | 'terminal' {
  const parsed = conversationEventSchema.safeParse(input)
  if (!parsed.success) throw new ConversationError('EVENT_CONTRACT')
  const event = parsed.data
  if (event.runId !== run.runId || event.messageId !== run.messageId) return 'stale'
  const fingerprint = JSON.stringify(event)
  if (run.seen[event.seq] !== undefined) {
    if (run.seen[event.seq] !== fingerprint) throw new ConversationError('EVENT_CONFLICT')
    return 'duplicate'
  }
  if (!isGenerating(run)) return 'terminal'
  if (event.seq !== run.lastSeq + 1) throw new ConversationError('EVENT_SEQUENCE')
  if (event.type === 'start' ? run.lastSeq !== 0 : run.mode === null) throw new ConversationError('EVENT_ORDER')
  if (event.type === 'text' && run.text.length + event.text.length > 32_768) throw new ConversationError('OUTPUT_LIMIT')
  if (event.type === 'tool' || event.type === 'citation' || event.type === 'result') {
    if (run.mode !== 'mock') throw new ConversationError('MOCK_CARD_ONLY')
    const index = run.cards.findIndex(card => card.cardId === event.cardId)
    const previous = run.cards[index]
    if (previous) {
      if (previous.type !== 'tool' || event.type !== 'tool' || previous.state !== 'running' || event.state === 'running' || previous.label !== event.label) throw new ConversationError('CARD_ORDER')
    } else {
      if (run.cards.length >= 12) throw new ConversationError('CARD_LIMIT')
      if (event.type === 'tool' && event.state !== 'running') throw new ConversationError('CARD_ORDER')
    }
    if (index < 0) run.cards.push(event); else run.cards[index] = event
  }
  if (event.type === 'start') run.mode = event.mode
  if (event.type === 'text') { run.text += event.text; run.status = 'streaming' }
  if (event.type === 'finish') run.status = event.reason === 'length' ? 'truncated' : 'completed'
  if (event.type === 'failure') { run.error = event.code; run.status = run.text ? 'interrupted' : 'error' }
  run.lastSeq = event.seq; run.seen[event.seq] = fingerprint
  return 'applied'
}
```

```ts
import type { GenerationStatus } from './generation'
export interface GenerationMetrics { startedAt:number; firstTextAt:number|null; endedAt:number|null; chars:number; outcome:GenerationStatus|null }
export const startMetrics=(now:number):GenerationMetrics=>({startedAt:now,firstTextAt:null,endedAt:null,chars:0,outcome:null})
export function measureText(metric:GenerationMetrics,text:string,now:number) {
  if(metric.endedAt!==null || !text.length)return
  if(metric.firstTextAt===null)metric.firstTextAt=now
  metric.chars+=Array.from(text).length
}
export function finishMetrics(metric:GenerationMetrics,outcome:GenerationStatus,now:number) {
  if(metric.endedAt!==null || outcome==='connecting' || outcome==='streaming')return
  metric.endedAt=now;metric.outcome=outcome
}
export function metricsView(metric:GenerationMetrics) {
  const elapsed=metric.firstTextAt===null || metric.endedAt===null ? null : metric.endedAt-metric.firstTextAt
  return {ttftMs:metric.firstTextAt===null?null:Math.max(0,metric.firstTextAt-metric.startedAt),durationMs:metric.endedAt===null?null:Math.max(0,metric.endedAt-metric.startedAt),chars:metric.chars,charsPerSecond:elapsed!==null && elapsed>0 ? metric.chars*1000/elapsed:null,outcome:metric.outcome}
}
// 本页已结束且非主动停止的请求为分母；截断单列，不计为网络中断。
export function summarizeMetrics(metrics:GenerationMetrics[]) {
  const eligible=metrics.filter(metric=>metric.endedAt!==null && metric.outcome!=='stopped')
  const interrupted=eligible.filter(metric=>metric.outcome==='interrupted' || metric.outcome==='error').length
  return {eligible:eligible.length,interrupted,interruptionRate:eligible.length?interrupted/eligible.length:null}
}
```

```ts
      onAbort = () => reject(new Error('CANCELLED')); abort.signal.addEventListener('abort', onAbort, { once: true })
    })
    localDeadline = setTimeout(() => {
      if (current()) { failGeneration(run, 'MODEL_TIMEOUT'); abort.abort() }
    }, 45_000)
    deadline=localDeadline
    try {
      await Promise.race([transport(ticket, prompt, token, abort.signal, event => {
        if (!current()) return
        if (event.type === 'done') mockCards(run)
        const result=applyConversationEvent(run, adaptEvent(event, run, run.lastSeq+1))
        if(result==='applied' && event.type==='delta')measureText(turn.metrics,event.text,now())
        if(!isGenerating(run))finishMetrics(turn.metrics,run.status,now())
      }), cancelled])
      if (current() && isGenerating(run)) failGeneration(run, 'STREAM_INTERRUPTED')
    } catch {
      if (current()) failGeneration(run, 'GENERATION_FAILED')
    } finally {
      abort.signal.removeEventListener('abort', onAbort)
      clearTimeout(localDeadline)
      finishMetrics(turn.metrics,run.status,now())
      if (current()) { clearTimeout(deadline); deadline = undefined; controller = undefined; currentRun = undefined }
    }
    return true
  }
  function fork(id: string, question?: string) {
    const old = turns.value.find(turn => turn.id === id)
    return old ? generate(question ?? old.question, old.parentId) : Promise.resolve(false)
  }
  function select(id: string) {
    if (busy.value || !turns.value.some(turn => turn.id === id)) return false
    activeLeaf.value = id; return true
  }
  function dispose() { if (disposed) return; reset(); disposed = true; unwatch() }
  if (getCurrentScope()) onScopeDispose(dispose)
  return { turns, path, activeLeaf, busy, error, send: (question: string) => generate(question, activeLeaf.value), fork, select, stop, reset, dispose }
}
```

## 五、核心面试题

1. 如何设计工具、引用和结果卡片协议？ 要点：固定type与字段上限，统一version/run/message/seq；cardId定位状态更新，工具running只能进入一次终态。渲染器固定组件与文本字段，不接受模型提供组件路径或任意props。追问：引用卡片等于事实可信？不，资料来源和内容核实另有流程。
2. 如何定义本章TTFT和字符速率？ 要点：从生成动作建立本地记录，到首次非空文本回调为TTFT；字符按Unicode码点数，速率分母从首字到终止。无文本/零跨度为null。追问：为什么不同于用户感知TTFT？rAF、Markdown解析和绘制在回调之后，首像素需要另测。
3. 中断率怎样避免把用户停止算故障？ 要点：分母只取已结束且非stopped，分子为error/interrupted；进行中不进入，truncated另列，无样本为null。保留原outcome，便于以后换口径重算。追问：是否可以直接当线上SLA？本页样本短暂、无持久采样与上报，不能代表全量服务质量。

完整原题、答案、补充和网络来源见[D54面试题](D54-面试题.md)。

## 六、个人建议

面试讲指标先给口径与样本边界，再给数字。用“中😀文是3个码点、4个UTF16单元”解释为什么不叫tokens；讲卡片时明确UI状态与真实执行证据的区别。
