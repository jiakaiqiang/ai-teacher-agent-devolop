## 零、闭卷回忆

什么时候生成第一条事件？如何判断两条同毫秒的事件先后？为什么公司管理员不能审批共享知识？

## 一、知识点与推导

工单事件记录动作、执行人、前后状态、前后版本及说明。注册审批落到独立审计表，让工单事件只保存本单行为。事件按版本还原，不依赖客户端时钟。

## 二、动手开发

先要求七次动作具有连续的 resultingVersion 1–7，再给事件迁移增加版本对。把注册审批的授权更新和审计插入包在同一事务中。

## 三、自检与提交

运行 return-to-handler.ts 和 registration.e2e-spec.ts。检查一项失败的并发动作不会产生审计记录，当前注册审核的操作人和申请人出现在不同列。

**apps/server/src/tickets/ticket-commands.ts**

```ts
import { ConflictException, ForbiddenException, Injectable, NotFoundException } from '@nestjs/common';
import { ticketStatusSchema } from '@ops-ticket/shared';
import { PrismaService } from '../prisma/prisma.service.js';
import { transition, type Action } from './workflow.js';

type Caller = { source: 'agent' } | { source:'user'; id:string };

@Injectable()
export class TicketCommands {
  constructor(private readonly db: PrismaService) {}
  // Caller 由受信入口构造。Agent 调用只来自服务端诊断任务。
  async apply(ticketNo:string, caller:Caller, action:Action, expectedVersion:number, targetId?:string, comment='') {
    return this.db.$transaction(async tx => {
      const row = await tx.ticket.findUnique({where:{ticketNo}});
      if (!row) throw new NotFoundException('工单不存在');
      const actor = caller.source === 'agent' ? {id:'agent',role:'agent'} :
        await tx.user.findFirst({where:{id:caller.id,registrationStatus:'approved'}});
      if (!actor) throw new ForbiddenException('账号未获批');
      const target = targetId ? await tx.user.findFirst({where:{id:targetId,registrationStatus:'approved'}}) : null;
      const next=transition({...row,status:ticketStatusSchema.parse(row.status)},actor,action,target ?? undefined);
      const changed=await tx.ticket.updateMany({where:{id:row.id,version:expectedVersion,status:row.status},
        data:{status:next.status,assigneeId:next.assigneeId,version:{increment:1},
          completedAt:next.status==='completed' ? new Date() : null}});
      if (changed.count!==1) throw new ConflictException('工单已变化，请刷新');
      return tx.ticketEvent.create({data:{ticketId:row.id,actorId:caller.source==='user'?caller.id:null,
        previousVersion:row.version,resultingVersion:row.version+1,action,source:caller.source,fromStatus:row.status,toStatus:next.status,comment}});
    });
  }
}
```

**apps/server/src/auth/registrations.controller.ts**

```ts
import { RequirePermission } from '../auth/permissions.decorator.js';
import { BadRequestException, Body, Controller, ForbiddenException, Param, Post, Req } from '@nestjs/common';
import { IsString, Length } from 'class-validator';
import { PrismaService } from '../prisma/prisma.service.js';
import { hashPassword } from './password.js';
import { Public } from './public.decorator.js';
import type { AuthenticatedRequest } from './auth.types.js';

class RegisterDto {
  @IsString() @Length(1,64) username!:string;
  @IsString() @Length(8,128) password!:string;
  @IsString() company!:string;
}
@Controller('registrations')
export class RegistrationsController {
  constructor(private readonly db:PrismaService) {}
  @Public() @Post()
  async register(@Body() dto:RegisterDto) {
    const company=await this.db.company.findUnique({where:{name:dto.company}});
    if (!company) throw new BadRequestException('客户公司不存在');
    return this.db.user.create({data:{
      username:dto.username,displayName:dto.username,passwordHash:await hashPassword(dto.password),
      companyId:company.id,role:'reporter',registrationStatus:'pending',
    },select:{id:true,registrationStatus:true}});
  }
  @RequirePermission('registration:approve')
@Post(':id/approve')
  async approve(@Req() req:AuthenticatedRequest,@Param('id') id:string) {
    const admin=await this.db.user.findFirst({where:{id:req.user!.id,role:'customer_admin',registrationStatus:'approved'}});
    if (!admin?.companyId) throw new ForbiddenException();
    await this.db.$transaction(async tx=>{
    const changed=await tx.user.updateMany({where:{id,role:'reporter',companyId:admin.companyId,registrationStatus:'pending'},
      data:{registrationStatus:'approved'}});
    if (!changed.count) throw new BadRequestException('本公司没有这项待审申请');
      await tx.registrationAudit.create({data:{companyId:admin.companyId!,approverId:admin.id,applicantId:id}});
    });
    return {approved:true};
  }
}
```

```ts
import assert from 'node:assert/strict';
import { PrismaService } from '../src/prisma/prisma.service.js';
import { TicketCommands } from '../src/tickets/ticket-commands.js';

const db=new PrismaService();
try {
  const sample=await db.ticket.findUniqueOrThrow({where:{ticketNo:'T-001'}});
  const no='LOCK-'+crypto.randomUUID();
  const {id:_id,...fields}=sample;
  const ticket=await db.ticket.create({data:{...fields,ticketNo:no,status:'pending_assignment'}});
  const users=await Promise.all(['department_lead','operator','operator'].map((role,i)=>
    db.user.create({data:{username:no+i,displayName:role,role,departmentId:ticket.departmentId,registrationStatus:'approved'}})));
  try {
    const cmd=new TicketCommands(db);
    const lead={source:'user' as const,id:users[0].id};
    const operator={source:'user' as const,id:users[1].id};
    const reporter={source:'user' as const,id:sample.reporterId};
    await cmd.apply(no,lead,'assign',0,operator.id);
    await cmd.apply(no,operator,'handle',1);
    await cmd.apply(no,lead,'approve',2);
    await cmd.apply(no,reporter,'unresolved',3,undefined,'故障再次出现');
    const returned=await db.ticket.findUniqueOrThrow({where:{id:ticket.id}});
    assert.equal(returned.assigneeId,operator.id);
    assert.equal(returned.status,'human_processing');
    await cmd.apply(no,operator,'handle',4);
    await cmd.apply(no,lead,'approve',5);
    await cmd.apply(no,reporter,'resolved',6);
    assert.equal(await db.ticketEvent.count({where:{ticketId:ticket.id}}),7);
    const events=await db.ticketEvent.findMany({where:{ticketId:ticket.id},orderBy:{resultingVersion:'asc'}});
    assert.deepEqual(events.map(e=>e.resultingVersion),[1,2,3,4,5,6,7]);
    console.log('退回原处理人，再处理、复核和确认完成，共七个动作');
  } finally {
    await db.ticketEvent.deleteMany({where:{ticketId:ticket.id}});
    await db.ticket.delete({where:{id:ticket.id}});
    await db.user.deleteMany({where:{id:{in:users.map(u=>u.id)}}});
  }
} finally {await db.$disconnect();}
```
