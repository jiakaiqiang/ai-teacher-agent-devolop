## 一、知识点与推导

按操作者、工单和请求键查回执，同键异参返回冲突。

对同一运行分别提交只读批次与带请求键的私人笔记。记录参数、当前主体、工具允许的用途和回执。再用另一公司账号或变化后的参数重试，核对审计和写入次数。

两个进程都可能先查到“没有笔记”。联合唯一约束放在 user_id 和 request_key 上；createMany(skipDuplicates) 的返回数量告诉调用方是否实际插入。随后从同一事务按联合键取回回执。

把版本、工单内部 ID 和规范化后的正文序列化并计算 SHA-256。库里已有相同键时，摘要相同才视为重放；换了工单或正文都应报 TOOL_KEY_CONFLICT。

```ts
import { createHash } from 'node:crypto';
import type { TransactionClient } from '../prisma/prisma.service.js';
import { CallError, noteArgs, pageArgs } from './tool-catalog.js';
import { pageOf, shortText } from './ticket-tool-read.js';

// 调用方已在同一事务中核对工单可见范围。笔记只属于当前用户。
export async function saveNote(tx: TransactionClient, userId: string, ticketId: string, input: unknown) {
  const { requestKey, text } = noteArgs.parse(input);
  const hash = createHash('sha256').update(JSON.stringify([1, ticketId, text])).digest('hex');
  const inserted = await tx.agentToolNote.createMany({ data: [{ userId, ticketId, requestKey, hash, text }],
    skipDuplicates: true });
  const receipt = await tx.agentToolNote.findUniqueOrThrow({ where: { userId_requestKey: { userId, requestKey } },
    select: { id: true, hash: true } });
  if (receipt.hash !== hash) throw new CallError('TOOL_KEY_CONFLICT');
  return { noteId: receipt.id, replay: inserted.count === 0 };
}
export async function listNotes(tx: TransactionClient, userId: string, ticketId: string, input: unknown) {
  const { limit, after } = pageArgs.parse(input);
  const rows = await tx.agentToolNote.findMany({ where: { userId, ticketId, ...(after ? { id: { gt: after } } : {}) },
    orderBy: { id: 'asc' }, take: limit + 1, select: { id: true, text: true } });
  return pageOf(rows.map(r => ({ id: r.id, ...shortText(r.text) })), limit);
}
```

```ts
describe('带版本的工具契约', () => {
  it('只加载请求的主题与副作用等级', () => {
    expect(definitions('device').map(x => x.name)).toEqual(['ticket_category', 'ticket_log_counts', 'ticket_log_page']);
    expect(definitions('notes').map(x => x.name)).toEqual(['list_private_notes']);
    expect(definitions('notes', true)[0]?.version).toBe(1);
    expect(definitions('overview', true).some(x => x.name === 'publish_solution')).toBe(false);
  });
  it('服务端拒绝不支持的版本和分页越界', () => {
    const call = { name: 'ticket_history_page', version: 1, args: { limit: 5, after: null } };
    expect(parseCall(call).name).toBe('ticket_history_page');
    expect(() => parseCall({ ...call, version: 2 })).toThrow('TOOL_CONTRACT_INVALID');
    expect(() => parseCall({ ...call, args: { limit: 6, after: null } })).toThrow('TOOL_ARGUMENTS_INVALID');
    expect(() => parseCall({ ...call, args: { ...call.args, ticketNo: 'T1' } })).toThrow('TOOL_ARGUMENTS_INVALID');
    expect(() => parseCall({ ...call, name: 'publish_solution' })).toThrow('TOOL_APPROVAL_REQUIRED');
  });
```

```ts
  ticket_log_page: definition('device', 'read', '分页读取本单日志', pageArgs),
  ticket_history_page: definition('history', 'read', '按编号游标分页读取处理记录', pageArgs),
  list_private_notes: definition('notes', 'read', '读取本人保存的诊断笔记', pageArgs),
  save_private_note: definition('notes', 'write', '保存当前用户的诊断笔记', noteArgs),
  publish_solution: definition('overview', 'high', '发布共享方案，需要承办部门审核', empty),
};
export type Name = keyof typeof catalog;
export class CallError extends Error {
  constructor(readonly code: string) { super(code); }
}
export const callSchema = z.object({
  name: z.string().min(1).max(50), version: z.literal(1), args: z.unknown(),
}).strict();
export function parseCall(input: unknown) {
  const call = callSchema.safeParse(input);
  if (!call.success) throw new CallError('TOOL_CONTRACT_INVALID');
  if (!Object.hasOwn(catalog, call.data.name)) throw new CallError('TOOL_UNKNOWN');
  const name = call.data.name as Name;
  const entry = catalog[name];
  if (entry.effect === 'high') throw new CallError('TOOL_APPROVAL_REQUIRED');
  const args = entry.schema.safeParse(call.data.args);
  if (!args.success) throw new CallError('TOOL_ARGUMENTS_INVALID');
  return { name, args: args.data, effect: entry.effect };
}
export function definitions(topic: Topic, allowWrites = false) {
  return Object.entries(catalog).filter(([, e]) => e.topic === topic &&
    (e.effect === 'read' || (allowWrites && e.effect === 'write'))).map(([name, e]) => ({
    name, version: e.version, effect: e.effect, description: e.description, parameters: z.toJSONSchema(e.schema),
  }));
}
```
