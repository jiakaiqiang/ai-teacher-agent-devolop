## 一、知识点与推导

已输出正文后重放会产生重复内容。重试受错误种类、已发送内容和总期限约束。

知识点示例取自当天已通过测试，按TypeScript语法树排版。示例依赖同一测试文件中的导入与夹具，不能单独粘贴执行；先完成第四节对应文件，再在指定目录运行命令。数据库示例沿用隔离测试数据清理范围。

### 1.1 总预算包住尝试与退避

ModelPolicy为逻辑调用只建一个计时器，默认5000ms。两次尝试之间等待100—199ms抖动；只有一次重试，不声称实现多级指数退避。等待和请求共用signal，finally清定时器与父监听。

### 1.3 闭合、开路与半开

按受控别名维护一个进程内ModelPolicy。连续三次逻辑调用最终可用性失败才开路30秒；普通调用最多两次HTTP，半开只有一次。epoch阻止开路前的旧成功覆盖新状态。用户取消不计供应商失败。

**推导题：为什么只允许一个半开探针？**

<details><summary>答案</summary>

恢复未知时放入大量请求会再次压垮上游；探针成功才恢复，失败重新开路。非可重试错误或取消不证明健康，释放探针但不标为恢复。

</details>

```ts
import type { ModelConfig } from './model.config.js';
import { readLimitedJson, type ChatMessage } from './chat.client.js';
import { GatewayFailure, statusFailure } from './model-policy.js';

// 不复用流式HTTP错误折叠：这里需要区分是否值得重试。
export async function requestCompletion(config: ModelConfig, messages: ChatMessage[], signal: AbortSignal): Promise<unknown> {
  let response: Response;
  try {
    response = await fetch(config.endpoint, {
      method: 'POST', redirect: 'error', signal,
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${config.apiKey}` },
      body: JSON.stringify({ model: config.model, messages, stream: false, max_completion_tokens: 512, store: false }),
    });
  } catch {
    if (signal.aborted) throw signal.reason;
    throw new GatewayFailure('MODEL_UNAVAILABLE', true);
  }
  if (!response.ok) { void response.body?.cancel().catch(() => {}); throw statusFailure(response.status); }
  if (!response.headers.get('content-type')?.includes('application/json')) {
    void response.body?.cancel().catch(() => {}); throw new GatewayFailure('MODEL_PROTOCOL_ERROR');
  }
  try { return await readLimitedJson(response); }
  catch { if (signal.aborted) throw signal.reason; throw new GatewayFailure('MODEL_PROTOCOL_ERROR'); }
}
```

```ts
import { Injectable } from '@nestjs/common';
import { ChatClient, ModelError, type ChatMessage } from './chat.client.js';
import { readModelRegistry } from './model-registry.js';
import { mockStream } from './mock-stream.js';
import { ModelPolicy, GatewayFailure } from './model-policy.js';
import { requestCompletion } from './model-completion.js';

@Injectable()
export class ModelGateway {
  private readonly policies = new Map<string, ModelPolicy>();
  async complete(messages: ChatMessage[], signal?: AbortSignal, alias = 'default', budgetMs = 5000): Promise<unknown> {
    const registry = readModelRegistry();
    const config = registry.resolve(alias);
    if (!config) throw new GatewayFailure('MODEL_CONFIG_ERROR');
    // 按受控别名保存进程内状态；配置变更需重启，最多8个。
    for (const key of this.policies.keys()) if (!registry.aliases().includes(key)) this.policies.delete(key);
    let policy = this.policies.get(alias);
    if (!policy) { policy = new ModelPolicy(); this.policies.set(alias, policy); }
    return policy.run(child => requestCompletion(config, messages, child), signal, budgetMs);
  }
  async stream(messages: ChatMessage[], signal: AbortSignal, alias = 'default') {
    signal.throwIfAborted();
    const config = readModelRegistry().resolve(alias);
    if (!config) return { body: mockStream(signal), mode: 'mock' as const };
    const response = await new ChatClient(config).open(messages, true, signal);
    if (!response.headers.get('content-type')?.startsWith('text/event-stream') || !response.body) {
      await response.body?.cancel(); throw new ModelError('MODEL_PROTOCOL_ERROR');
    }
    // 本周流式保持W7预算与终止校验；不会在已发正文后重放请求。
    return { body: response.body, mode: 'model' as const };
  }
}
```

```ts
// 仅用于无副作用的非流式模型生成；计费仍可能发生，不能保证远端取消。
export type GatewayCode = 'MODEL_BUSY' | 'MODEL_UNAVAILABLE' | 'MODEL_TIMEOUT' | 'MODEL_CANCELLED' | 'MODEL_CONFIG_ERROR' | 'MODEL_REQUEST_ERROR' | 'MODEL_PROTOCOL_ERROR' | 'MODEL_CIRCUIT_OPEN';
export class GatewayFailure extends Error {
  constructor(readonly code: GatewayCode, readonly retryable = false) { super(code); }
}
export function statusFailure(status: number): GatewayFailure {
  if (status === 429) return new GatewayFailure('MODEL_BUSY', true);
  if (status === 408 || status === 524) return new GatewayFailure('MODEL_TIMEOUT', true);
  if (status >= 500) return new GatewayFailure('MODEL_UNAVAILABLE', true);
  return new GatewayFailure(status === 401 || status === 403 ? 'MODEL_CONFIG_ERROR' : 'MODEL_REQUEST_ERROR');
}

export function abortable<T>(work: Promise<T>, signal: AbortSignal): Promise<T> {
  return new Promise<T>((resolve, reject) => {
    const aborted = () => reject(signal.reason);
    if (signal.aborted) { work.catch(() => {}); reject(signal.reason); return; }
    signal.addEventListener('abort', aborted, { once: true });
    work.then(resolve, reject).finally(() => signal.removeEventListener('abort', aborted));
  });
}
async function delay(ms: number, signal: AbortSignal) {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { await abortable(new Promise<void>(resolve => { timer = setTimeout(resolve, ms); }), signal); }
  finally { clearTimeout(timer); }
}

export class ModelPolicy {
  private failures = 0;
  private openedAt: number | null = null;
  private probe = false;
  private epoch = 0;
  constructor(private readonly now = () => Date.now(), private readonly random = () => Math.random()) {}
  async run<T>(work: (signal: AbortSignal, attempt: number) => Promise<T>, parent?: AbortSignal, budgetMs = 5000): Promise<T> {
    if (parent?.aborted) throw new GatewayFailure('MODEL_CANCELLED');
    if (!Number.isFinite(budgetMs) || budgetMs <= 0 || budgetMs > 5000) throw new GatewayFailure('MODEL_TIMEOUT');
    if (this.openedAt !== null && (this.now() - this.openedAt < 30000 || this.probe)) throw new GatewayFailure('MODEL_CIRCUIT_OPEN');
    const halfOpen = this.openedAt !== null;
    if (halfOpen) this.probe = true;
    const epoch = this.epoch;
    const controller = new AbortController();
    const cancel = () => controller.abort(new GatewayFailure('MODEL_CANCELLED'));
    parent?.addEventListener('abort', cancel, { once: true });
    const timer = setTimeout(() => controller.abort(new GatewayFailure('MODEL_TIMEOUT', true)), budgetMs);
    try {
      for (let attempt = 1; ; attempt++) {
        controller.signal.throwIfAborted();
        try {
          const value = await abortable(Promise.resolve().then(() => work(controller.signal, attempt)), controller.signal);
          if (epoch === this.epoch) { this.failures = 0; this.openedAt = null; }
          return value;
        } catch (error) {
          if (controller.signal.aborted) throw controller.signal.reason;
          if (!(error instanceof GatewayFailure) || !error.retryable || attempt >= 2 || halfOpen) throw error;
          await delay(100 + Math.floor(this.random() * 100), controller.signal);
        }
      }
    } catch (error) {
      if (epoch === this.epoch && error instanceof GatewayFailure && error.retryable) {
        this.failures++;
        if (halfOpen || this.failures >= 3) { this.openedAt = this.now(); this.epoch++; }
      }
      throw error;
    } finally {
      clearTimeout(timer); parent?.removeEventListener('abort', cancel);
      if (halfOpen) this.probe = false;
    }
  }
}
```
