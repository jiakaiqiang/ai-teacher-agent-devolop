**推导题：为什么不能只靠tool_choice:none限制工具次数？**

### 1.2 工具事件代表可观察行为

SSE协议带protocol、runId和连续seq，依次发meta、tool-start/tool-result、answer、done或error。工具事件在实际调用前后发出，send等待背压；最多16帧、总事件JSON体积64KiB。最终回答是非流式模型完成后的正文，不宣称逐token输出，也不展示私有思维链。

### 1.3 持久化与额度顺序决定失败语义

服务端认领轮次后以该轮次ID预留额度，避免不同会话复用客户端键冲突。模型调用后结算，再提交完整成功记录，最后发answer/done。失败保留问题和错误，尽量结算；结算异常保留pending待核查。数据库与供应商费用没有跨系统原子事务，不自动退款或重放。

```ts
import { ConflictException, HttpException, Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service.js';
import { DAILY_UNITS, RESERVATION_UNITS, usageCharge, type UsageSummary } from './usage-meter.js';

@Injectable()
export class AiQuotaService {
  constructor(private readonly prisma:PrismaService){}
  async receipt(userId: string, requestKey: string): Promise<{ status: 'exact' | 'estimated'; units: number }> {
    const row = await this.prisma.aiUsage.findUnique({ where: { userId_requestKey: { userId, requestKey } } });
    if (!row || row.charged === null || !['exact','estimated'].includes(row.status)) throw new ConflictException({ code: 'AI_SETTLEMENT_PENDING', message: '用量结算待核对' });
    return { status: row.status === 'exact' ? 'exact' : 'estimated', units: row.charged };
  }
  async reserve(userId:string,requestKey:string,now=new Date()) {
    const period=now.toISOString().slice(0,10); // 明确使用UTC自然日，不随浏览器时区变化。
    return this.prisma.$transaction(async tx=>{
      await tx.aiQuota.createMany({data:[{userId,period,available:DAILY_UNITS}],skipDuplicates:true});
      const [reservation]=await tx.aiUsage.createManyAndReturn({data:[{userId,period,requestKey,reserved:RESERVATION_UNITS}],skipDuplicates:true,select:{id:true}});
      if(!reservation)throw new ConflictException({code:'AI_REQUEST_ALREADY_USED',message:'请求标识已使用，请勿重复生成'});
      const changed=await tx.aiQuota.updateMany({where:{userId,period,available:{gte:RESERVATION_UNITS}},data:{available:{decrement:RESERVATION_UNITS}}});
      if(changed.count!==1)throw new HttpException({code:'AI_QUOTA_EXCEEDED',message:'今日AI额度不足'},429);
      return reservation.id;
    });
  }
  async settle(id:string,summary:UsageSummary) {
    const charged=usageCharge(summary),status=summary.exact?'exact':'estimated';
    return this.prisma.$transaction(async tx=>{
      const row=await tx.aiUsage.findUniqueOrThrow({where:{id}});
      const changed=await tx.aiUsage.updateMany({where:{id,status:'pending'},data:{charged,status,inputTokens:summary.inputTokens,outputTokens:summary.outputTokens,settledAt:new Date()}});
      if(changed.count===0){
        const existing=await tx.aiUsage.findUniqueOrThrow({where:{id}});
        if(existing.charged!==charged||existing.status!==status||existing.inputTokens!==summary.inputTokens||existing.outputTokens!==summary.outputTokens)throw new ConflictException('AI_SETTLEMENT_CONFLICT');
        return existing;
      }
      // 实际消耗超过预留时允许余额为负，忠实记账并阻止新预留；不是供应商硬限额。
      await tx.aiQuota.update({where:{userId_period:{userId:row.userId,period:row.period}},data:{available:{increment:row.reserved-charged}}});
      return tx.aiUsage.findUniqueOrThrow({where:{id}});
    });
  }
}
```

**apps/server/src/assistant/model-completion.ts**

```ts
import type { ToolMessage } from './tool-protocol.js';
import type { ModelConfig } from './model.config.js';
import { readLimitedJson, type ChatMessage } from './chat.client.js';
import { GatewayFailure, statusFailure } from './model-policy.js';

// 不复用流式HTTP错误折叠：这里需要区分是否值得重试。
export async function requestCompletion(config: ModelConfig, messages: (ChatMessage | ToolMessage)[], signal: AbortSignal, responseFormat?: Record<string, unknown>, toolOptions?: { tools: unknown[]; tool_choice: 'auto' | 'none' }): Promise<unknown> {
  let response: Response;
  try {
    response = await fetch(config.endpoint, {
      method: 'POST', redirect: 'error', signal,
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${config.apiKey}` },
      body: JSON.stringify({ model: config.model, messages, stream: false, max_completion_tokens: 512, store: false, ...toolOptions, ...(responseFormat ? { response_format: responseFormat } : {}) }),
    });
  } catch {
    if (signal.aborted) throw signal.reason;
    throw new GatewayFailure('MODEL_UNAVAILABLE', true);
  }
  if (!response.ok) { void response.body?.cancel().catch(() => {}); throw statusFailure(response.status); }
  if (!response.headers.get('content-type')?.includes('application/json')) {
    void response.body?.cancel().catch(() => {}); throw new GatewayFailure('MODEL_PROTOCOL_ERROR');
  }
  try { return await readLimitedJson(response); }
  catch { if (signal.aborted) throw signal.reason; throw new GatewayFailure('MODEL_PROTOCOL_ERROR'); }
}
```

**apps/server/src/assistant/model-gateway.ts**

```ts
import type { ToolMessage } from './tool-protocol.js';
import type { UsageMeter } from './usage-meter.js';
import { Injectable } from '@nestjs/common';
import { ChatClient, ModelError, type ChatMessage } from './chat.client.js';
import { readModelRegistry } from './model-registry.js';
import { mockStream } from './mock-stream.js';
import { ModelPolicy, GatewayFailure } from './model-policy.js';
import { requestCompletion } from './model-completion.js';

@Injectable()
export class ModelGateway {
  private readonly policies = new Map<string, ModelPolicy>();
  async complete(messages: (ChatMessage | ToolMessage)[], signal?: AbortSignal, alias = 'default', budgetMs = 5000, responseFormat?: Record<string, unknown>, meter?: UsageMeter, toolOptions?: { tools: unknown[]; tool_choice: 'auto' | 'none' }): Promise<unknown> {
    const registry = readModelRegistry();
    const config = registry.resolve(alias);
    if (!config) throw new GatewayFailure('MODEL_CONFIG_ERROR');
    // 按受控别名保存进程内状态；配置变更需重启，最多8个。
    for (const key of this.policies.keys()) if (!registry.aliases().includes(key)) this.policies.delete(key);
    let policy = this.policies.get(alias);
    if (!policy) { policy = new ModelPolicy(); this.policies.set(alias, policy); }
    return policy.run(async child => { meter?.start(); const raw = await requestCompletion(config, messages, child, responseFormat, toolOptions); meter?.record(raw); return raw; }, signal, budgetMs);
  }
  async stream(messages: ChatMessage[], signal: AbortSignal, alias = 'default') {
    signal.throwIfAborted();
    const config = readModelRegistry().resolve(alias);
    if (!config) return { body: mockStream(signal), mode: 'mock' as const };
    const response = await new ChatClient(config).open(messages, true, signal);
    if (!response.headers.get('content-type')?.startsWith('text/event-stream') || !response.body) {
      await response.body?.cancel(); throw new ModelError('MODEL_PROTOCOL_ERROR');
    }
    // 本周流式保持W7预算与终止校验；不会在已发正文后重放请求。
    return { body: response.body, mode: 'model' as const };
  }
}
```

```ts
import { Injectable, type OnModuleDestroy } from '@nestjs/common';
import type { Response } from 'express';
import type { AssistantEvent, ToolSessionEvent } from '@ops-ticket/shared';
import { ModelError } from './chat.client.js';

export interface StreamPolicy { totalMs: number; idleMs: number; }
export class StreamLease {
  readonly controller = new AbortController();
  readonly signal = this.controller.signal;
  private readonly total: ReturnType<typeof setTimeout>;
  private idle: ReturnType<typeof setTimeout> | undefined;
  private disposed = false;
  private readonly closed = () => { if (!this.res.writableFinished) this.cancel('MODEL_CANCELLED'); };
  private readonly failed = () => this.cancel('MODEL_CANCELLED');
  constructor(private readonly res: Response, private readonly policy: StreamPolicy, private readonly release: () => void) {
    res.once('close', this.closed); res.once('error', this.failed);
    this.total = setTimeout(() => this.cancel('MODEL_TIMEOUT'), policy.totalMs);
    this.touch();
    if (res.destroyed) this.closed();
  }
  cancel(code: 'MODEL_TIMEOUT' | 'MODEL_CANCELLED') { if (!this.signal.aborted) this.controller.abort(new ModelError(code)); }
  touch() {
    if (this.disposed || this.signal.aborted) return;
    clearTimeout(this.idle);
    this.idle = setTimeout(() => this.cancel('MODEL_TIMEOUT'), this.policy.idleMs);
  }
  async send(event: AssistantEvent | ToolSessionEvent): Promise<void> {
    this.signal.throwIfAborted();
    if (this.res.destroyed) throw new ModelError('MODEL_CANCELLED');
    const accepted = this.res.write(`event: ${event.type}\ndata: ${JSON.stringify(event)}\n\n`);
    if (!accepted) {
      await new Promise<void>((resolve, reject) => {
        const cleanup = () => { this.res.off('drain', drained); this.signal.removeEventListener('abort', aborted); };
        const drained = () => { cleanup(); resolve(); };
        const aborted = () => { cleanup(); reject(this.signal.reason); };
        this.res.once('drain', drained); this.signal.addEventListener('abort', aborted, { once: true });
        if (this.signal.aborted) aborted();
      });
    }
    this.signal.throwIfAborted();
    this.touch(); // 应用事件进展重置空闲预算，注释心跳不算生成进展。
  }
  dispose() {
    if (this.disposed) return;
    this.disposed = true; clearTimeout(this.total); clearTimeout(this.idle);
    this.res.off('close', this.closed); this.res.off('error', this.failed);
    this.cancel('MODEL_CANCELLED'); this.release();
  }
}

@Injectable()
export class AssistantStreams implements OnModuleDestroy {
  private readonly leases = new Set<StreamLease>();
  private closing = false;
  policy(): StreamPolicy { return { totalMs: 30_000, idleMs: 10_000 }; }
  get activeCount() { return this.leases.size; }
  open(res: Response): StreamLease {
    if (this.closing) throw new ModelError('MODEL_CANCELLED');
    const lease = new StreamLease(res, this.policy(), () => this.leases.delete(lease));
    this.leases.add(lease); return lease;
  }
  beginShutdown() { this.closing = true; for (const lease of this.leases) lease.cancel('MODEL_CANCELLED'); }
  onModuleDestroy() { this.beginShutdown(); }
}
```

```ts
import { parseToolReply, ToolFailure, type ToolCall, type ToolMessage } from './tool-protocol.js';

export type ToolProgress = { kind: 'tool-start'; call: ToolCall }
  | { kind: 'tool-result'; call: ToolCall; content: string };
export interface ToolLoopOptions {
  messages: ToolMessage[];
  complete: (messages: ToolMessage[], signal: AbortSignal, toolsEnabled: boolean) => Promise<unknown>;
  execute: (call: ToolCall, signal: AbortSignal) => Promise<unknown>;
  signal?: AbortSignal;
  budgetMs?: number;
  onProgress?: (event: ToolProgress) => void | Promise<void>;
}

// 一轮最多三个只读工具，再给模型一次总结机会；不是无限Agent循环。
export async function runToolLoop(options: ToolLoopOptions) {
  const budget = options.budgetMs ?? 10000;
  if (!Number.isInteger(budget) || budget < 1 || budget > 10000) throw new ToolFailure('TOOL_BUDGET_INVALID');
  const controller = new AbortController();
  const cancel = () => controller.abort(new ToolFailure('TOOL_CANCELLED'));
  options.signal?.addEventListener('abort', cancel, { once: true });
  if (options.signal?.aborted) cancel();
  const timer = setTimeout(() => controller.abort(new ToolFailure('TOOL_TIMEOUT')), budget);
  const messages = structuredClone(options.messages);
  const appended: ToolMessage[] = [];
  let rejectAbort: () => void = () => {};
  const aborted = new Promise<never>((_, reject) => {
    rejectAbort = () => reject(controller.signal.reason);
    controller.signal.addEventListener('abort', rejectAbort, { once: true });
    if (controller.signal.aborted) rejectAbort();
  });
  const work = async () => {
    for (let step = 0; step < 2; step++) {
      controller.signal.throwIfAborted();
      const raw = await options.complete(messages, controller.signal, step === 0);
      controller.signal.throwIfAborted();
      const reply = parseToolReply(raw);
      if (reply.kind === 'answer') {
        appended.push({ role: 'assistant', content: reply.text });
        return { text: reply.text, appended };
      }
      if (step !== 0) throw new ToolFailure('TOOL_STEP_LIMIT');
      const assistant: ToolMessage = { role: 'assistant', content: reply.content, tool_calls: reply.calls };
      messages.push(assistant); appended.push(assistant);
      for (const call of reply.calls) {
        controller.signal.throwIfAborted();
        await options.onProgress?.({ kind: 'tool-start', call });
        const result = await options.execute(call, controller.signal);
        controller.signal.throwIfAborted();
        const content = JSON.stringify(result);
        if (typeof content !== 'string' || Buffer.byteLength(content, 'utf8') > 4096) throw new ToolFailure('TOOL_RESULT_LIMIT');
        const message: ToolMessage = { role: 'tool', tool_call_id: call.id, content };
        messages.push(message); appended.push(message);
        await options.onProgress?.({ kind: 'tool-result', call, content });
      }
    }
    throw new ToolFailure('TOOL_STEP_LIMIT');
  };
  try { return await Promise.race([work(), aborted]); }
  finally {
    clearTimeout(timer);
    options.signal?.removeEventListener('abort', cancel);
    controller.signal.removeEventListener('abort', rejectAbort);
  }
}
```

```ts
import { HttpException, Injectable } from '@nestjs/common';
import { toolSessionEventSchema, type ToolSessionEvent, type ToolSessionPayload } from '@ops-ticket/shared';
import { ConversationsService } from './conversations.service.js';
import { ReadToolsService, readToolDefinitions } from './read-tools.service.js';
import { ModelGateway } from './model-gateway.js';
import { AiQuotaService } from './ai-quota.service.js';
import { UsageMeter } from './usage-meter.js';
import { readModelRegistry } from './model-registry.js';
import { GatewayFailure } from './model-policy.js';
import { ToolFailure } from './tool-protocol.js';
import { buildConversationContext, parseStoredTranscript } from './conversation-context.js';
import { runToolLoop } from './tool-loop.js';
import { mockToolCompletion } from './mock-tools.js';

export function sessionFailureCode(error: unknown): string {
  if (error instanceof ToolFailure || error instanceof GatewayFailure) return error.code;
  if (error instanceof HttpException) {
    const body = error.getResponse();
    if (typeof body === 'object' && body && 'code' in body && typeof body.code === 'string' && /^[A-Z_]{1,64}$/.test(body.code)) return body.code;
  }
  return 'CONVERSATION_FAILED';
}

@Injectable()
export class ToolSessionService {
  constructor(private readonly store: ConversationsService, private readonly tools: ReadToolsService,
    private readonly gateway: ModelGateway, private readonly quota: AiQuotaService) {}
  async run(userId: string, conversationId: string, input: { requestKey: string; version: number; question: string }, signal: AbortSignal, emit: (event: ToolSessionEvent) => Promise<void>) {
    signal.throwIfAborted();
    const history = await this.store.get(userId, conversationId);
    const context = buildConversationContext(history.turns, input.question);
    const begun = await this.store.begin(userId, { ...input, conversationId });
    let seq = 0;
    const send = (payload: ToolSessionPayload) => emit(toolSessionEventSchema.parse({ ...payload, protocol: 1, runId: begun.turn.id, seq: ++seq }));
    if (begun.replay) {
      await send({ type: 'meta', mode: begun.turn.mode === 'model' ? 'model' : 'mock', replay: true });
      if (begun.turn.status === 'failed') { await send({ type: 'error', code: begun.turn.errorCode ?? 'CONVERSATION_FAILED' }); return; }
      const receipt = await this.quota.receipt(userId, begun.turn.id);
      await send({ type: 'answer', text: begun.turn.answer });
      await send({ type: 'done', conversationVersion: history.conversation.version, billing: receipt });
      return;
    }
    const meter = new UsageMeter();
    let reservation: string | undefined, settlementAttempted = false, persisted = false;
    let mode: 'mock' | 'model' = 'mock';
    try {
      signal.throwIfAborted();
      mode = readModelRegistry().resolve() ? 'model' : 'mock';
      // 用服务端轮次ID计量，避免不同会话或分诊共用客户端请求键。
      reservation = await this.quota.reserve(userId, begun.turn.id);
      await send({ type: 'meta', mode, replay: false });
      const deadline = Date.now() + 10000;
      const result = await runToolLoop({ messages: context.messages, signal,
        complete: (messages, child, enabled) => mode === 'mock' ? mockToolCompletion(messages, child, enabled) :
          this.gateway.complete(messages, child, 'default', Math.min(5000, Math.max(1, deadline - Date.now())), undefined, meter, { tools: readToolDefinitions, tool_choice: enabled ? 'auto' : 'none' }),
        execute: (call, child) => this.tools.execute(call, userId, child),
        onProgress: event => event.kind === 'tool-start' ? send({ type: 'tool-start', callId: event.call.id, name: event.call.function.name, arguments: event.call.function.arguments }) :
          send({ type: 'tool-result', callId: event.call.id, name: event.call.function.name, content: event.content }),
      });
      signal.throwIfAborted();
      // 生成前后各复核一次；不把读取时检查宣传为实时撤权中断。
      const previous = history.turns.filter(turn => turn.status === 'completed').flatMap(turn => parseStoredTranscript(turn.transcript));
      await this.tools.assertReferences(userId, [...previous, ...result.appended], history.conversation.scopeHash);
      settlementAttempted = true;
      const bill = await this.quota.settle(reservation, meter.summary());
      await this.store.finish(userId, begun.turn.id, { status: 'completed', mode, answer: result.text, transcript: result.appended, errorCode: null });
      persisted = true;
      const stored = await this.store.get(userId, conversationId);
      // 先提交，再发送最终正文和done；这是工具进度流，不是token级输出。
      await send({ type: 'answer', text: result.text });
      await send({ type: 'done', conversationVersion: stored.conversation.version, billing: { status: bill.status === 'exact' ? 'exact' : 'estimated', units: bill.charged! } });
    } catch (error) {
      if (persisted) throw error; // 断线不撤销已经提交的成功结果，重载可读取。
      let code = signal.aborted ? 'TOOL_CANCELLED' : sessionFailureCode(error);
      if (reservation && !settlementAttempted) {
        settlementAttempted = true;
        try { await this.quota.settle(reservation, meter.summary()); }
        catch { code = 'AI_SETTLEMENT_PENDING'; }
      }
      // 失败只保留问题与错误；中途工具卡片不伪装成完整成功轨迹。
      await this.store.finish(userId, begun.turn.id, { status: 'failed', mode, answer: '', transcript: [], errorCode: code });
      throw new ToolFailure(code);
    }
  }
}
```

```ts
function httpError(error: unknown): never {
  if (error instanceof HttpException) throw error;
  const code = sessionFailureCode(error);
  const status = code.endsWith('NOT_FOUND') ? 404 : code === 'TOOL_FORBIDDEN' ? 403 : /INVALID|ARGUMENT/.test(code) ? 400 : /CONFLICT|BUSY|CHANGED|LIMIT/.test(code) ? 409 : 503;
  throw new HttpException({ code, message: '会话请求未完成，请查看状态后再试' }, status);
}
function summary(row: { id: string; title: string; version: number; activeTurnId: string | null; updatedAt: Date }) {
  return sessionSummarySchema.parse({ id: row.id, title: row.title, version: row.version, activeTurnId: row.activeTurnId, updatedAt: row.updatedAt.toISOString() });
}

@Controller('assistant/conversations')
@RequirePermission('ticket:report')
export class ToolSessionsController {
  constructor(private readonly store: ConversationsService, private readonly sessions: ToolSessionService, private readonly streams: AssistantStreams) {}
  @Post() @HttpCode(201)
  async create(@Req() req: AuthenticatedRequest, @Body() input: unknown) {
    try { const parsed = sessionCreateSchema.safeParse(input); if (!parsed.success) throw new ToolFailure('CONVERSATION_INVALID'); return summary(await this.store.create(req.user!.id, parsed.data.title)); }
    catch (error) { httpError(error); }
  }
  @Get()
  async list(@Req() req: AuthenticatedRequest) {
    try { return sessionListSchema.parse((await this.store.list(req.user!.id)).map(summary)); } catch (error) { httpError(error); }
  }
  @Get(':id')
  async get(@Req() req: AuthenticatedRequest, @Param('id') id: string) {
    try {
      const { conversation, turns } = await this.store.get(req.user!.id, id);
      return sessionDetailSchema.parse({ conversation: summary(conversation), turns: turns.map(turn => {
        const messages = turn.status === 'completed' ? parseStoredTranscript(turn.transcript) : [];
        const calls = messages.flatMap(message => message.role === 'assistant' ? message.tool_calls ?? [] : []);
        return { id: turn.id, requestKey: turn.requestKey, ordinal: turn.ordinal, question: turn.question, status: turn.status, mode: turn.mode, answer: turn.answer, errorCode: turn.errorCode,
          tools: calls.map(call => ({ callId: call.id, name: call.function.name, arguments: call.function.arguments, content: messages.find(message => message.role === 'tool' && message.tool_call_id === call.id)?.content ?? '' })) };
      }) });
    } catch (error) { httpError(error); }
  }
  @Post(':id/turns') @HttpCode(200)
  async turn(@Req() req: AuthenticatedRequest, @Param('id') id: string, @Body() input: unknown, @Res() res: Response) {
    const parsed = sessionTurnInputSchema.safeParse(input);
    if (!parsed.success) httpError(new ToolFailure('CONVERSATION_INVALID'));
    const lease = this.streams.open(res);
    let last: ToolSessionEvent | undefined, bytes = 0;
    const send = async (event: ToolSessionEvent) => {
      bytes += Buffer.byteLength(JSON.stringify(event));
      if (bytes > 65536) throw new ToolFailure('TOOL_RESULT_LIMIT');
      if (!res.headersSent) { res.status(200).set({ 'Content-Type': 'text/event-stream; charset=utf-8', 'Cache-Control': 'no-store', 'X-Accel-Buffering': 'no' }); res.flushHeaders(); }
      last = event; await lease.send(event);
    };
    try { await this.sessions.run(req.user!.id, id, parsed.data, lease.signal, send); }
    catch (error) {
      if (!res.headersSent && !res.destroyed) httpError(error);
      if (last && !res.destroyed && !res.writableNeedDrain && last.type !== 'done' && last.seq < 16) {
        const event = { protocol: 1, runId: last.runId, seq: last.seq + 1, type: 'error', code: sessionFailureCode(error) };
        res.write(`event: error\ndata: ${JSON.stringify(event)}\n\n`);
      }
    } finally {
      lease.dispose();
      if (res.headersSent && !res.destroyed) { if (res.writableNeedDrain) res.destroy(); else res.end(); }
    }
  }
}
```
