**apps/server/test/evaluation/selection.spec.ts**

```ts
import { selection } from './selection.js';

const single = { success: 10, total: 12, violations: 0, inputTokens: 100, outputTokens: 10, p95Ms: 200,
  hits: 9, answered: 11, correctTools: 12, modelCalls: 13, steps: 26 };
describe('架构选择', () => {
  it('多角色的额外命中不能抵消用例违规', () => {
    const team = { ...single, success: 11, violations: 1 };
    expect(selection({ single, team, singleOnly: 0, teamOnly: 1 })).toContain('暂缓采用');
  });
  it('同样成功时同时查看延迟和两类 token', () => {
    const team = { ...single, outputTokens: 40, p95Ms: 300 };
    expect(selection({ single, team, singleOnly: 0, teamOnly: 0 })).toContain('保留单角色');
    expect(selection({ single, team: { ...team, p95Ms: 180 }, singleOnly: 0, teamOnly: 0 })).toContain('权衡');
  });
});
```

**apps/server/test/evaluation/selection.ts**

```ts
import type { compare } from './paired.js';

type Report = ReturnType<typeof compare>;
// 读数用于准备评审。只有在两侧均满足用例约束时才比较收益和运行成本。
export function selection(report: Report) {
  const { single, team } = report;
  if (single.violations && team.violations) return '两侧都有用例违规，先复核';
  if (team.violations) return '多角色有用例违规，暂缓采用';
  if (single.violations) return '单角色有用例违规，先修复对照';
  const tokens = (arm: typeof team) => arm.inputTokens + arm.outputTokens;
  if (team.success <= single.success && tokens(team) >= tokens(single) && team.p95Ms >= single.p95Ms) {
    return '样本未显示多角色收益，保留单角色';
  }
  if (team.success > single.success) return '多角色成功数更高，复核独有命中与代价';
  return '存在成本或延迟权衡，按场景复核';
}
```

**apps/server/test/evaluation/team-decision.ts**

```ts
import { readFileSync } from 'node:fs';
import { compare, type Arm } from './paired.js';
import { selection } from './selection.js';

const lines = readFileSync(process.argv[2], 'utf8').trim().split('\n').map(s => JSON.parse(s));
const live = lines.filter(x => x.kind === 'live' && x.edition === 'service-ticket-eval-v2');
if (live.length !== 1) throw new Error('需要一批独立的本机配对结果');
const report = compare(live[0].rows as Arm[], live[0].rounds);
console.log(JSON.stringify({ edition: live[0].edition, rounds: live[0].rounds, result: report, conclusion: selection(report) }));
```
