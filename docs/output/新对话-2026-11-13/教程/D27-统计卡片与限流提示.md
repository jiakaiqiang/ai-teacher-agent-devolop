## 一、知识点与推导

从响应读取统计总量；出错时提示统计不可用。倒计时按截止时刻计算，后台标签页回来后不会显示过期的等待时间。限流拒绝给出重试窗口。

在列表页点击“上报问题”，选择分类并输入描述和日志。提交后进入同一单号的详情页。相同材料重试沿用请求键，修改材料后生成新的键。用同一用户连续提交触发 429，记录 Retry-After 和按钮再次可用的时刻。

```ts
import { computed, getCurrentScope, onScopeDispose, ref } from 'vue'

// 倒计时（D27）：收到 429 后按 Retry-After 倒数，到 0 之前不让再提交
// 记的是截止时间而不是“还剩几秒”：浏览器把后台标签页的定时器放慢时，切回来显示的仍然是准确的剩余时间
export function useCountdown() {
  const remaining = ref(0)
  let deadline = 0
  let timer: ReturnType<typeof setInterval> | null = null

  function tick() {
    remaining.value = Math.max(0, Math.ceil((deadline - Date.now()) / 1000))
    if (remaining.value === 0) stop()
  }

  function stop() {
    if (timer) clearInterval(timer)
    timer = null
    remaining.value = 0
  }

  function start(seconds: number) {
    if (timer) clearInterval(timer)
    deadline = Date.now() + seconds * 1000
    timer = setInterval(tick, 1000)
    tick()
  }

  // 在组件里使用时，组件卸载就清掉定时器
  if (getCurrentScope()) onScopeDispose(stop)

  const active = computed(() => remaining.value > 0)
  return { remaining, active, start, stop }
}
```

```ts
import { afterEach, beforeEach, describe, expect, it, vi } from 'vitest'
import { useCountdown } from './useCountdown'

beforeEach(() => {
  vi.useFakeTimers()
})

afterEach(() => {
  vi.useRealTimers()

  it('从 Retry-After 的秒数开始倒数，到 0 后恢复可用', () => {

    c.start(3)
    expect(c.remaining.value).toBe(3)

    expect(c.remaining.value).toBe(1)
```

```ts
import { HttpException } from '@nestjs/common';
import { RateLimitService } from '../redis/rate-limit.service.js';
import { LoginOnly } from '../auth/permissions.decorator.js';
import { RequirePermission } from '../auth/permissions.decorator.js';
import { BadRequestException, Body, Controller, Headers, Param, Post, Req } from '@nestjs/common';
import { IsArray, IsIn, IsInt, IsOptional, IsString, Min } from 'class-validator';
import { TicketCommands } from './ticket-commands.js';
import { SubmitTicket } from './submit-ticket.js';
import type { AuthenticatedRequest } from '../auth/auth.types.js';
import type { Action } from './workflow.js';

class ReportDto {
  @IsString() title!:string;
  @IsString() description!:string;
  @IsString() category!:string;
  @IsArray() @IsString({each:true}) logs!:string[];
}
class ActDto {
  @IsIn(['request_human','unresolved','assign','handle','approve','return','resolved']) action!:Action;
  @IsInt() @Min(0) version!:number;
  @IsOptional() @IsString() targetId?:string;
  @IsOptional() @IsString() comment?:string;
}
@Controller('tickets')
export class TicketWriteController {
  constructor(private readonly cmd:TicketCommands,
    private readonly limits:RateLimitService, private readonly reports:SubmitTicket) {}
  @RequirePermission('ticket:report')
  @Post()
  async submit(@Req() req:AuthenticatedRequest, @Headers('idempotency-key') key:string, @Body() dto:ReportDto) {
    if (!key) throw new BadRequestException('缺少请求键');
    const admission=await this.limits.consume('report:'+req.user!.id,5,1000);
    if (!admission.allowed) throw new HttpException({code:'RATE_LIMITED',message:'提单过于频繁',retryAfterSeconds:admission.retryAfterSeconds},429);
    const row=await this.reports.submit(req.user!.id,key,dto);
    return {ticketNo:row.ticketNo,status:row.status,version:row.version};
  }
  @LoginOnly()
  @Post(':no/actions')
  async act(@Param('no') no:string,@Req() req:AuthenticatedRequest,@Body() dto:ActDto) {
    return this.cmd.apply(no,{source:'user',id:req.user!.id},dto.action,dto.version,dto.targetId,dto.comment);
  }
}
```

```ts
import { apiErrorSchema } from '@ops-ticket/shared'
import { z } from 'zod'
import { handleUnauthorized, session } from './session'

// 前端统一的请求错误：code 用来判断，message 用来展示
export class ApiError extends Error {
  // 前端开启了 erasableSyntaxOnly，不能用构造函数参数属性的写法，字段要单独声明
  readonly status: number
  retryAfterSeconds = 0
  readonly code: string

  constructor(status: number, code: string, message: string) {
    super(message)
    this.name = 'ApiError'
    this.status = status
    this.code = code
  }
}

const envelopeSchema = z.object({ data: z.unknown() })

// 所有请求都带上登录令牌（D15）
function headers(extra: Record<string, string> = {}): Record<string, string> {
  const token = session.value?.accessToken
  return { Accept: 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}), ...extra }
}

// 解析响应并按契约校验。后端返回的结构和 shared 里的 schema 对不上时，
// 在这里就报 CONTRACT_MISMATCH，而不是让页面拿着错误的数据渲染到一半出错
async function parseResponse<T extends z.ZodType>(response: Response, path: string, schema: T): Promise<z.infer<T>> {
  const body: unknown = await response.json().catch(() => null)

  if (!response.ok) {
    const parsed = apiErrorSchema.safeParse(body)
    // 未登录或令牌失效：清掉登录状态，回到登录页。登录接口本身的 401（密码错误）不算
    if (response.status === 401 && parsed.success && parsed.data.error.code === 'UNAUTHENTICATED') {
      handleUnauthorized()
    }
    if (parsed.success) {
      const failure = new ApiError(response.status, parsed.data.error.code, parsed.data.error.message)
      if (response.status === 429) failure.retryAfterSeconds = Number(response.headers?.get('Retry-After')) || 0
      throw failure
    }
    // 代理或网关返回的错误（例如后端没启动时的 502）没有统一结构
    throw new ApiError(response.status, 'HTTP_ERROR', `请求失败（HTTP ${response.status}）`)
  }

  // 先拆出 { data } 外壳，再按具体接口的 schema 校验 data
  const envelope = envelopeSchema.safeParse(body)
  const parsed = schema.safeParse(envelope.success ? envelope.data.data : undefined)
  if (!parsed.success) {
    // 开发时在控制台打印具体是哪个字段不符合契约
    console.error('接口响应不符合契约', path, parsed.error.issues)
    throw new ApiError(response.status, 'CONTRACT_MISMATCH', '接口返回的数据格式不正确')
  }
  return parsed.data
}

export async function getJson<T extends z.ZodType>(
  path: string,
  schema: T,
  signal?: AbortSignal,
): Promise<z.infer<T>> {
  const response = await fetch(path, { signal, headers: headers() })
  return parseResponse(response, path, schema)
}

export async function postJson<T extends z.ZodType>(path: string, body: unknown, schema: T, signal?: AbortSignal, extra:Record<string,string> = {}): Promise<z.infer<T>> {
  const response = await fetch(path, {
    method: 'POST', signal,
    headers: headers({ ...extra, 'Content-Type': 'application/json' }),
    body: JSON.stringify(body),
  })
  return parseResponse(response, path, schema)
}
```

**apps/server/src/common/http-exception.filter.ts**

```ts
import { redactForLog } from './sensitive.js';
import {
  ArgumentsHost,
  Catch,
  ExceptionFilter,
  HttpException,
  HttpStatus,
  Logger,
} from '@nestjs/common';
import type { Response } from 'express';
import { Prisma } from '../generated/prisma/client.js';

// 错误响应统一结构：code 给程序判断，message 给人看，details 放校验失败的字段
export interface ErrorBody {
  error: { code: string; message: string; details?: unknown };
}

interface MappedError {
  status: number;
  code: string;
  message: string;
}

// 把数据库约束错误翻译成 HTTP 错误；约束名等内部细节只写日志，不返回给客户端
function mapPrismaError(
  error: Prisma.PrismaClientKnownRequestError,
): MappedError | undefined {
  switch (error.code) {
    case 'P2002': // 唯一约束冲突
      return { status: HttpStatus.CONFLICT, code: 'DUPLICATE_RESOURCE', message: '数据已存在' };
    case 'P2003': // 外键约束：被引用的行不能删，或引用的行不存在
      return { status: HttpStatus.CONFLICT, code: 'RESOURCE_IN_USE', message: '数据仍被其他记录引用，不能删除' };
    case 'P2025': // 要更新或删除的记录不存在
      return { status: HttpStatus.NOT_FOUND, code: 'NOT_FOUND', message: '记录不存在' };
  }
  // CHECK 约束没有专门的 Prisma 错误码，要看 PostgreSQL 的原始错误码 23514
  const cause = (error.meta?.driverAdapterError as { cause?: { originalCode?: string } } | undefined)?.cause;
  if (cause?.originalCode === '23514') {
    return { status: HttpStatus.UNPROCESSABLE_ENTITY, code: 'CONSTRAINT_VIOLATION', message: '数据不满足业务约束' };
  }
  return undefined;
}

@Catch()
export class HttpExceptionFilter implements ExceptionFilter {
  private readonly logger = new Logger(HttpExceptionFilter.name);

  catch(exception: unknown, host: ArgumentsHost) {
    const response = host.switchToHttp().getResponse<Response>();

    if (exception instanceof HttpException) {
      const status = exception.getStatus();
      const raw = exception.getResponse();
      // ValidationPipe 抛出的 400，message 是字段错误数组，放进 details
      const message =
        typeof raw === 'object' && raw !== null && 'message' in raw
          ? (raw as { message: unknown }).message
          : exception.message;
      const body: ErrorBody = {
        error: {
          code: typeof raw === 'object' && raw !== null && 'code' in raw && typeof raw.code === 'string'
            ? raw.code : HttpStatus[status] ?? 'HTTP_ERROR',
          message: Array.isArray(message) ? '请求参数不合法' : String(message),
          ...(Array.isArray(message) ? { details: message } : {}),
        },
      };
      if (status === 429 && typeof raw === 'object' && 'retryAfterSeconds' in raw) {
        const delay = Number(raw.retryAfterSeconds);
        if (Number.isSafeInteger(delay) && delay > 0) response.setHeader('Retry-After', String(delay));
      }
      response.status(status).json(body);
      return;
    }

    if (exception instanceof Prisma.PrismaClientKnownRequestError) {
      const mapped = mapPrismaError(exception);
      if (mapped) {
        // 约束名这类细节写日志，方便排查
        this.logger.warn(`${exception.code}: ${exception.message.split('\n').filter(Boolean).pop()}`);
        const body: ErrorBody = { error: { code: mapped.code, message: mapped.message } };
        response.status(mapped.status).json(body);
        return;
      }
    }

    // 未知异常：完整堆栈只写日志，不返回给客户端，避免泄露内部信息
    this.logger.error(exception instanceof Error ? exception.stack ? redactForLog(exception.stack) : undefined : exception);
    const body: ErrorBody = {
      error: { code: 'INTERNAL_SERVER_ERROR', message: '服务器内部错误' },
    };
    response.status(HttpStatus.INTERNAL_SERVER_ERROR).json(body);
  }
}
```
