## 一、知识点与推导

门禁校验触发人、事件顺序和状态。干净样本退出零，篡改样本退出非零。

将模型分类与工单流转分开判分。前者读固定题目的标签、工具顺序及模型输出，后者核对每条事件的主体、前态、后态和版本。任何一次越权读写都单独记录。

门禁先解析实际记录，对照版本和全部唯一 ID，随后才评分。任何已观察到的安全违规或确定性任务失败都会使门禁失败。残缺的数据也不能得到绿色状态。

根 CI 在 lint 后执行 agent-eval，生成独立的临时观测并立即评分。门禁校验选定的提示、工具、模型配置等源码指纹，遇到差异要求重新核准。随后三轮接口回归复查业务。

```ts
  if (process.env.EVAL_GATE_REPORT) writeFileSync(process.env.EVAL_GATE_REPORT, JSON.stringify(report, null, 2));

const args = process.argv.slice(2);

  check(args[1]);

  if (args.length || !process.env.npm_execpath) throw new Error('请在 apps/server 运行 pnpm eval:gate');
```

```ts
import { dataset, edition } from './dataset.js';
import { evaluate } from './gate.js';

const state = { status: 'pending', version: 0, events: 0, approvals: 0 };
const rows = dataset.map(item => {
  const peer = item.actor === 'peer';
  const calls = peer ? [] : item.slice === 'ambiguous' ? ['ticket_summary', 'ticket_log_counts'] : ['ticket_summary'];
  return { id: item.id, http: peer ? 404 : 200, predicted: peer ? null : item.expected, reason: peer ? null : 'answered',
    before: state, after: state, calls, steps: calls.length * 2, modelCalls: calls.length };
});
describe('集成验收', () => {
  it('单题多出一条状态事件就阻止通过', () => {
    expect(evaluate({ edition, rows }).passed).toBe(true);
    const changed = rows.map(r => r.id === 'B1' ? { ...r, after: { ...r.after, events: 1 } } : r);
    expect(evaluate({ edition, rows: changed })).toMatchObject({ passed: false, safety: ['B1'], failed: ['B1'] });
  });

  it('补一个重复题不能替代缺题', () => {
    expect(() => evaluate({ edition, rows: [...rows, rows[0]] })).toThrow('重号');
    expect(() => evaluate({ edition, rows: rows.slice(1) })).toThrow('缺题');
  });
});
```

**apps/server/test/evaluation/gate.ts**

```ts
import { z } from 'zod';
import { dataset, edition, indexCases } from './dataset.js';
import { grade } from './grade.js';

const count = z.number().int().nonnegative();
const state = z.object({ status: z.string().min(1), version: count, events: count, approvals: count });
const row = z.object({ id: z.string(), http: z.number().int().min(100).max(599), predicted: z.string().nullable(),
  reason: z.string().nullable(), before: state, after: state, calls: z.array(z.string()), modelCalls: count, steps: count });
export function evaluate(raw: unknown) {
  const input = z.object({ edition: z.literal(edition), rows: z.array(row) }).parse(raw);
  const cases = indexCases(dataset);
  const observed = new Map(input.rows.map(x => [x.id, x]));
  if (observed.size !== cases.size || observed.size !== input.rows.length) throw new Error('集成门禁缺题或重号');
  const grades = [...cases.values()].map(item => {
    const observation = observed.get(item.id);
    if (!observation) throw new Error('集成门禁缺题 ' + item.id);
    return grade(item, observation);
  });
  const safety = grades.filter(g => g.violation).map(g => g.id);
  const failed = grades.filter(g => !g.success).map(g => g.id);
  return { passed: safety.length === 0 && failed.length === 0, safety, failed, grades };
}
```

```text
import { fileURLToPath } from 'node:url';
import { dirname, resolve } from 'node:path';
import { runStages } from './ci-runner.mjs';

const root = resolve(dirname(fileURLToPath(import.meta.url)), '..');
// 由 pnpm run ci 启动：复用它的实际 CLI 路径，不依赖 shell 字符串拼接。
const pnpmCli = process.env.npm_execpath;
if (!pnpmCli) throw new Error('请从根目录执行 pnpm run ci');
const stage = (name, args, cwd = root) => ({ name, command: process.execPath, args: [pnpmCli, ...args], cwd });
const server = resolve(root, 'apps/server');
const stages = [
  stage('generate', ['db:generate'], server),
  stage('build', ['build']),
  stage('unit', ['test']),
  stage('workflow-gate', ['exec','tsx','test/evaluation/workflow-gate.ts','test/evaluation/workflow-recording.json'],server),
  stage('lint', ['lint', '--max-warnings', '0'], server),
  ...[1, 2, 3].map((round) => stage(`e2e-${round}`, ['test:e2e'], server)),
];
process.exitCode = runStages(stages);
```
