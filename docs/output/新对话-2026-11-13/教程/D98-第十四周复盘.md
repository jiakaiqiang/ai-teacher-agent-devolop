```python
"""本地批处理练习：只生成标签摘要，不调用模型或执行工单操作。"""
import asyncio
from dataclasses import dataclass
from typing import Literal

from ops_bridge.bounded import run_bounded
from ops_bridge.contracts import parse_request
from ops_bridge.models import TicketSnapshot


@dataclass(frozen=True, slots=True)
class ReportItem:
    ticket_no: str
    text: str


@dataclass(frozen=True, slots=True)
class BatchReport:
    mode: Literal["local_mock"]
    count: int
    items: tuple[ReportItem, ...]


async def mock_worker(ticket: TicketSnapshot) -> str:
    await asyncio.sleep(0)  # 让出循环，展示异步接口；不是模型推理耗时。
    return f"{ticket.ticket_no}：{len(ticket.tags)}个标签"


async def build_report(value: object) -> BatchReport:
    request = parse_request(value)
    texts = await run_bounded(request.tickets, mock_worker,
                              concurrency=request.concurrency, timeout=2)
    items = tuple(ReportItem(ticket.ticket_no, text)
                  for ticket, text in zip(request.tickets, texts, strict=True))
    return BatchReport("local_mock", len(items), items)
```

**python/bridge/ops_bridge/contracts.py**

```python
"""未知输入先校验，再进入有类型的内部逻辑。"""
from dataclasses import dataclass
from typing import Protocol

from ops_bridge.models import TicketDraft, TicketSnapshot


class TicketWorker(Protocol):
    async def __call__(self, ticket: TicketSnapshot) -> str: ...


@dataclass(frozen=True, slots=True)
class BatchRequest:
    tickets: tuple[TicketSnapshot, ...]
    concurrency: int


def parse_request(value: object) -> BatchRequest:
    if not isinstance(value, dict) or set(value) != {"tickets", "concurrency"}:
        raise ValueError("请求字段必须为tickets和concurrency")
    concurrency = value["concurrency"]
    # bool是int的子类；业务整数明确排除bool。
    if type(concurrency) is not int or not 1 <= concurrency <= 8:
        raise ValueError("concurrency必须为1到8的整数")
    rows = value["tickets"]
    if not isinstance(rows, list) or not 1 <= len(rows) <= 100:
        raise ValueError("tickets必须为1到100项列表")
    tickets: list[TicketSnapshot] = []
    seen: set[str] = set()
    for row in rows:
        if not isinstance(row, dict) or set(row) != {"ticket_no", "tags"}:
            raise ValueError("工单字段不合法")
        ticket_no, tags = row["ticket_no"], row["tags"]
        if not isinstance(ticket_no, str) or not 1 <= len(ticket_no.strip()) <= 64:
            raise ValueError("工单号长度必须为1到64")
        ticket_no = ticket_no.strip()
        if ticket_no in seen:
            raise ValueError("工单号重复")
        seen.add(ticket_no)
        if not isinstance(tags, list) or len(tags) > 10:
            raise ValueError("tags必须为至多10项列表")
        draft = TicketDraft(ticket_no)
        for tag in tags:
            if not isinstance(tag, str) or not 1 <= len(tag.strip()) <= 64:
                raise ValueError("标签长度必须为1到64")
            draft.add_tag(tag)
        tickets.append(draft.snapshot())
    return BatchRequest(tuple(tickets), concurrency)
```

**python/bridge/ops_bridge/execution.py**

```python
"""区分整批失败与逐项保留结果；取消始终向上传播。"""
import asyncio
from dataclasses import dataclass
from typing import Literal

from ops_bridge.contracts import TicketWorker
from ops_bridge.models import TicketSnapshot


@dataclass(frozen=True, slots=True)
class ItemResult:
    ticket_no: str
    status: Literal["ok", "error"]
    text: str | None = None
    error: str | None = None


async def run_all(tickets: tuple[TicketSnapshot, ...], worker: TicketWorker) -> tuple[str, ...]:
    async with asyncio.TaskGroup() as group:
        tasks = [group.create_task(worker(ticket)) for ticket in tickets]
    return tuple(task.result() for task in tasks)


async def run_partial(
    tickets: tuple[TicketSnapshot, ...], worker: TicketWorker
) -> tuple[ItemResult, ...]:
    async def one(ticket: TicketSnapshot) -> ItemResult:
        try:
            text = await worker(ticket)
            return ItemResult(ticket.ticket_no, "ok", text=text)
        except Exception as error:
            # 只记录错误类名，不把可能含敏感数据的异常正文带入报告。
            return ItemResult(ticket.ticket_no, "error", error=type(error).__name__)

    async with asyncio.TaskGroup() as group:
        tasks = [group.create_task(one(ticket)) for ticket in tickets]
    return tuple(task.result() for task in tasks)
```

```python
import asyncio

import pytest

from ops_bridge.execution import run_all, run_partial
from ops_bridge.models import TicketSnapshot

TICKETS = (TicketSnapshot("A", ()), TicketSnapshot("B", ()))


def test_group_keeps_input_order() -> None:
    async def scenario() -> None:
        second = asyncio.Event()
        done: list[str] = []

        async def worker(ticket: TicketSnapshot) -> str:
            if ticket.ticket_no == "A":
                await second.wait()
            done.append(ticket.ticket_no)
            second.set()
            return ticket.ticket_no

        assert await run_all(TICKETS, worker) == ("A", "B")
        assert done == ["B", "A"]

    asyncio.run(scenario())


def test_group_failure_waits_for_sibling_cleanup() -> None:
    async def scenario() -> None:
        started = asyncio.Event()
        cleaned: list[str] = []

        async def worker(ticket: TicketSnapshot) -> str:
            if ticket.ticket_no == "B":
                await started.wait()
                raise ValueError("模拟失败")
            try:
                started.set()
                await asyncio.Event().wait()
                return "不可到达"
            finally:
                cleaned.append("A")

        with pytest.raises(ExceptionGroup) as caught:
            await run_all(TICKETS, worker)
        assert any(isinstance(error, ValueError) for error in caught.value.exceptions)
        assert cleaned == ["A"]

    asyncio.run(scenario())


def test_partial_retains_success_and_redacts_error_message() -> None:
    async def worker(ticket: TicketSnapshot) -> str:
        if ticket.ticket_no == "B":
            raise ValueError("此正文不应进入报告")
        return "成功"

    result = asyncio.run(run_partial(TICKETS, worker))
    assert [(row.ticket_no, row.status) for row in result] == [("A", "ok"), ("B", "error")]
    assert result[0].text == "成功"
    assert result[1].error == "ValueError"
    assert result[1].text is None


def test_worker_cancellation_is_not_item_error() -> None:
    async def worker(ticket: TicketSnapshot) -> str:
        raise asyncio.CancelledError()

    with pytest.raises(asyncio.CancelledError):
        asyncio.run(run_partial(TICKETS, worker))


def test_parent_cancellation_cleans_children() -> None:
    async def scenario() -> None:
        started = asyncio.Event()
        cleaned: list[str] = []

        async def worker(ticket: TicketSnapshot) -> str:
            try:
                started.set()
                await asyncio.Event().wait()
                return "不可到达"
            finally:
                cleaned.append(ticket.ticket_no)

        task = asyncio.create_task(run_partial(TICKETS, worker))
        await started.wait()
        task.cancel()
        with pytest.raises(asyncio.CancelledError):
            await task
        assert sorted(cleaned) == ["A", "B"]

    asyncio.run(scenario())


def test_empty_utility_does_not_call_worker() -> None:
    async def worker(ticket: TicketSnapshot) -> str:
        raise AssertionError("不应调用")

    assert asyncio.run(run_all((), worker)) == ()
    assert asyncio.run(run_partial((), worker)) == ()
```
