**apps/server/src/tickets/staff-actions.controller.ts**

```ts
import { Controller, Get, Param, Req, NotFoundException } from '@nestjs/common';
import type { AuthenticatedRequest } from '../auth/auth.types.js';
import { LoginOnly } from '../auth/permissions.decorator.js';
import { PrismaService } from '../prisma/prisma.service.js';
import { visibleTickets } from './scope.js';

@Controller('tickets/:no/staff-actions')
@LoginOnly()
export class StaffActionsController {
  constructor(private readonly db:PrismaService) {}
  @Get()
  async read(@Param('no') ticketNo:string,@Req() req:AuthenticatedRequest) {
    const actor=await this.db.user.findFirstOrThrow({where:{id:req.user!.id,registrationStatus:'approved'}});
    const ticket=await this.db.ticket.findFirst({where:{AND:[{ticketNo},visibleTickets(actor)]}});
    if(!ticket) throw new NotFoundException();
    const lead=actor.role==='department_lead' && actor.departmentId===ticket.departmentId;
    const handler=actor.role==='operator' && actor.id===ticket.assigneeId;
    const assign=lead && ticket.status==='pending_assignment';
    const operators=assign?await this.db.user.findMany({
      where:{departmentId:ticket.departmentId,role:'operator',registrationStatus:'approved'},
      select:{id:true,displayName:true},orderBy:{displayName:'asc'}
    }):[];
    return {actions:assign?['assign']:lead && ticket.status==='handled'?['approve','return']:
      handler && ticket.status==='human_processing'?['handle']:[],operators};
  }
}
```

**apps/web/src/components/StaffActions.vue**

```vue
<script setup lang="ts">
import { ref, watch } from 'vue'
import { z } from 'zod'
import { getJson, postJson } from '../api/http'
const props=defineProps<{ticketNo:string;version:number}>()
const emit=defineEmits<{reload:[]}>()
const labels={assign:'派单',handle:'提交处理结果',approve:'复核通过并发送',return:'退回处理人'}
const contract=z.object({actions:z.array(z.enum(['assign','handle','approve','return'])),
  operators:z.array(z.object({id:z.string(),displayName:z.string()}))})
const options=ref<z.infer<typeof contract>>()
const target=ref(''),comment=ref(''),error=ref(''),busy=ref(false)
watch(()=>[props.ticketNo,props.version],async(_,__,cleanup)=>{
  const ctrl=new AbortController(); cleanup(()=>ctrl.abort())
  options.value=undefined; target.value=''; comment.value=''; error.value=''
  try {const data=await getJson('/api/tickets/'+props.ticketNo+'/staff-actions',contract,ctrl.signal)
    if(!ctrl.signal.aborted) options.value=data
  }catch{if(!ctrl.signal.aborted) error.value='操作列表加载失败'}
},{immediate:true})
async function submit(action:keyof typeof labels) {
  if(busy.value) return
  busy.value=true
  try{
    await postJson('/api/tickets/'+props.ticketNo+'/actions',{action,version:props.version,
      ...(action==='assign'?{targetId:target.value}:{}),comment:comment.value},z.unknown())
    emit('reload')
  }catch(e){error.value=e instanceof Error?e.message:'操作失败，请刷新后重试'}
  finally{busy.value=false}
}
</script>
<template>
  <section v-if="options?.actions.length">
    <label v-if="options.actions.includes('assign')">处理人 <select v-model="target"><option value="" disabled>请选择</option>
      <option v-for="person in options.operators" :key="person.id" :value="person.id">{{ person.displayName }}</option></select></label>
    <label v-else>处理或复核说明 <textarea v-model="comment" /></label>
    <button v-for="act in options.actions" :key="act" :disabled="busy || (act==='assign' && !target) || (act==='handle' && !comment.trim())" @click="submit(act)">{{ labels[act] }}</button>
  </section><p v-if="error" role="alert">{{ error }}</p>
</template>
```

**apps/web/src/views/TicketDetailView.vue**

```vue
<script setup lang="ts">
import { ref, watch } from 'vue'
import ConversationPanel from '../components/ConversationPanel.vue'
import RecommendationPanel from '../components/RecommendationPanel.vue'
import StaffActions from '../components/StaffActions.vue'
import TicketFeedback from '../components/TicketFeedback.vue'
import { useRoute } from 'vue-router'
import type { TicketDetail } from '@ops-ticket/shared'
import { getTicket } from '../api/tickets'
const refresh = ref(0)
const route = useRoute()
const ticket = ref<TicketDetail>()
const error = ref('')
watch([() => String(route.params.ticketNo), refresh], async ([no], _, cleanup) => {
  const controller = new AbortController()
  cleanup(() => controller.abort())
  ticket.value = undefined
  error.value = ''
  try {
    const result = await getTicket(no, controller.signal)
    if (!controller.signal.aborted) ticket.value = result
  } catch {
    if (!controller.signal.aborted) error.value = '工单加载失败'
  }
}, { immediate: true })
</script>
<template>
  <RouterLink to="/">返回工单列表</RouterLink>
  <p v-if="error" role="alert">{{ error }}</p>

    <h3>处理记录</h3>

    <TicketFeedback :can-confirm="ticket.canConfirm" :ticket-no="ticket.ticketNo" :status="ticket.status" :version="ticket.version" @reload="refresh++"/>
  </article>
</template>
```
