### 1.2 bool与整数业务含义

Python的bool是int子类，isinstance(True,int)为真。并发数契约只接受实际int，并限制1—8，拒绝bool、浮点数、字符串以及越界数。这里不做隐式转换，错误输入不会悄悄变为合法值。

**python/bridge/ops_bridge/contracts.py**

```python
"""未知输入先校验，再进入有类型的内部逻辑。"""
from dataclasses import dataclass
from typing import Protocol

from ops_bridge.models import TicketDraft, TicketSnapshot


class TicketWorker(Protocol):
    async def __call__(self, ticket: TicketSnapshot) -> str: ...


@dataclass(frozen=True, slots=True)
class BatchRequest:
    tickets: tuple[TicketSnapshot, ...]
    concurrency: int


def parse_request(value: object) -> BatchRequest:
    if not isinstance(value, dict) or set(value) != {"tickets", "concurrency"}:
        raise ValueError("请求字段必须为tickets和concurrency")
    concurrency = value["concurrency"]
    # bool是int的子类；业务整数明确排除bool。
    if type(concurrency) is not int or not 1 <= concurrency <= 8:
        raise ValueError("concurrency必须为1到8的整数")
    rows = value["tickets"]
    if not isinstance(rows, list) or not 1 <= len(rows) <= 100:
        raise ValueError("tickets必须为1到100项列表")
    tickets: list[TicketSnapshot] = []
    seen: set[str] = set()
    for row in rows:
        if not isinstance(row, dict) or set(row) != {"ticket_no", "tags"}:
            raise ValueError("工单字段不合法")
        ticket_no, tags = row["ticket_no"], row["tags"]
        if not isinstance(ticket_no, str) or not 1 <= len(ticket_no.strip()) <= 64:
            raise ValueError("工单号长度必须为1到64")
        ticket_no = ticket_no.strip()
        if ticket_no in seen:
            raise ValueError("工单号重复")
        seen.add(ticket_no)
        if not isinstance(tags, list) or len(tags) > 10:
            raise ValueError("tags必须为至多10项列表")
        draft = TicketDraft(ticket_no)
        for tag in tags:
            if not isinstance(tag, str) or not 1 <= len(tag.strip()) <= 64:
                raise ValueError("标签长度必须为1到64")
            draft.add_tag(tag)
        tickets.append(draft.snapshot())
    return BatchRequest(tuple(tickets), concurrency)
```

```python
import pytest

from ops_bridge.contracts import parse_request


def payload() -> dict[str, object]:
    return {"tickets": [{"ticket_no": " A ", "tags": [" 网络 ", "网络"]}], "concurrency": 2}


def test_parse_normalizes_and_copies() -> None:
    rows = [{"ticket_no": " A ", "tags": [" 网络 ", "网络"]}]
    request = parse_request({"tickets": rows, "concurrency": 2})
    rows.clear()
    assert request.tickets[0].ticket_no == "A"
    assert request.tickets[0].tags == ("网络",)
    assert request.concurrency == 2


def test_bool_is_not_a_business_integer() -> None:
    with pytest.raises(ValueError, match="整数"):
        parse_request({**payload(), "concurrency": True})


def test_unknown_fields_are_rejected() -> None:
    with pytest.raises(ValueError, match="字段"):
        parse_request({**payload(), "admin": True})


def test_duplicate_ids_after_normalization_are_rejected() -> None:
    with pytest.raises(ValueError, match="重复"):
        parse_request({"tickets": [{"ticket_no": "A", "tags": []},
                                   {"ticket_no": " A ", "tags": []}], "concurrency": 1})


@pytest.mark.parametrize("value", [None, [], {}, {"tickets": [], "concurrency": 1},
                                  {"tickets": [{}], "concurrency": 1}])
def test_invalid_shapes_fail(value: object) -> None:
    with pytest.raises(ValueError):
        parse_request(value)


@pytest.mark.parametrize("limit", [0, 9, 1.5, "2"])
def test_invalid_concurrency_fails(limit: object) -> None:
    with pytest.raises(ValueError):
        parse_request({**payload(), "concurrency": limit})


def test_size_and_text_limits() -> None:
    for row in [{"ticket_no": "", "tags": []}, {"ticket_no": "A", "tags": [False]},
                {"ticket_no": "A", "tags": ["x"] * 11},
                {"ticket_no": "A", "tags": ["x" * 65]}]:
        with pytest.raises(ValueError):
            parse_request({"tickets": [row], "concurrency": 1})
    with pytest.raises(ValueError):
        parse_request({"tickets": [{"ticket_no": str(i), "tags": []} for i in range(101)],
                       "concurrency": 1})
```
