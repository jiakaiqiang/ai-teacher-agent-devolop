## 一、知识点与推导

UI 根据当前状态展示动作。服务器独立检查身份、版本、状态。409 后刷新只更新画面，不自动重发用户选择；重复请求用服务端状态机兜底。

## 二、动手开发

先对 POST /api/tickets 和 POST /api/tickets/:no/actions 写成功、跨用户、重复提交三种请求。补齐 Nest 模块依赖。给共享详情契约增加 version，重新构建共享包；让 Vue 的 TicketFeedback 只负责提交和触发重读，在详情页保持单一数据来源。

## 三、自检与提交

**apps/server/src/tickets/ticket-write.controller.ts**

```ts
import { BadRequestException, Body, Controller, Headers, Param, Post } from '@nestjs/common';
import { IsArray, IsIn, IsInt, IsOptional, IsString, Min } from 'class-validator';
import { TicketCommands } from './ticket-commands.js';
import { SubmitTicket } from './submit-ticket.js';
import { PrismaService } from '../prisma/prisma.service.js';
import type { Action } from './workflow.js';

class ReportDto {
  @IsString() title!:string;
  @IsString() description!:string;
  @IsString() category!:string;
  @IsArray() @IsString({each:true}) logs!:string[];
}
class ActDto {
  @IsIn(['request_human','unresolved','assign','handle','approve','return','resolved']) action!:Action;
  @IsInt() @Min(0) version!:number;
  @IsOptional() @IsString() targetId?:string;
  @IsOptional() @IsString() comment?:string;
}
@Controller('tickets')
export class TicketWriteController {
  constructor(private readonly db:PrismaService, private readonly cmd:TicketCommands,
    private readonly reports:SubmitTicket) {}
  // W2 的预设身份仅用于本地训练，W3 由经过登录校验的会话提供身份。
  private async trainingUser(username?:string) {
    const user = username ? await this.db.user.findUnique({where:{username}}) : null;
    if (!user) throw new BadRequestException('缺少练习身份');
    return user.id;
  }
  @Post()
  async submit(@Headers('x-user') username:string, @Headers('idempotency-key') key:string, @Body() dto:ReportDto) {
    if (!key) throw new BadRequestException('缺少请求键');
    const row=await this.reports.submit(await this.trainingUser(username),key,dto);
    return {ticketNo:row.ticketNo,status:row.status,version:row.version};
  }
  @Post(':no/actions')
  async act(@Param('no') no:string,@Headers('x-user') username:string,@Body() dto:ActDto) {
    return this.cmd.apply(no,{source:'user',id:await this.trainingUser(username)},dto.action,dto.version,dto.targetId,dto.comment);
  }
}
```

```ts
import { decodeCursor, encodeCursor } from '../common/cursor.js';
import type { ListTicketsQuery } from './dto/list-tickets.query.js';
import { Injectable, NotFoundException } from '@nestjs/common';
import type { Prisma } from '../generated/prisma/client.js';
import { PrismaService } from '../prisma/prisma.service.js';

const detailSelect = {
  version: true, ticketNo: true, title: true, description: true, status: true, createdAt: true,
  company: { select: { name: true } }, department: { select: { name: true } },
  category: { select: { name: true } }, reporter: { select: { displayName: true } },
  assignee: { select: { displayName: true } },
  materials: { select: { logText: true }, orderBy: { id: 'asc' } },
  events: { orderBy: [{ createdAt: 'asc' }, { id: 'asc' }],
    select: { action: true, toStatus: true, actor: { select: { displayName: true } } } },
} satisfies Prisma.TicketSelect;

export type TicketDetail = Prisma.TicketGetPayload<{ select: typeof detailSelect }>;

@Injectable()
export class TicketsService {
  constructor(private readonly prisma: PrismaService) {}
  async list({ company, department, cursor, limit }: ListTicketsQuery) {
    const after = cursor ? decodeCursor(cursor) : undefined;
    const rows = await this.prisma.ticket.findMany({
      where: {
        company: company ? { name: company } : undefined,
        department: department ? { name: department } : undefined,
        OR: after ? [
          { createdAt: { lt: after.createdAt } },
          { createdAt: after.createdAt, id: { lt: after.id } },
        ] : undefined,
      },
      orderBy: [{createdAt:'desc'}, {id:'desc'}], take: limit + 1,
      select: { id:true, ticketNo:true, title:true, status:true, createdAt:true },
    });
    const page = rows.slice(0,limit);
    return {
      items: page.map(({ id: _id, ...item }) => item),
      nextCursor: rows.length > limit ? encodeCursor(page[page.length - 1]) : null,
    };
  }

  async findDetail(ticketNo: string): Promise<TicketDetail> {
    const ticket = await this.prisma.ticket.findUnique({ where: { ticketNo }, select: detailSelect });
    if (!ticket) throw new NotFoundException('工单不存在');
    return ticket;
  }
}
```

**apps/web/src/components/TicketFeedback.vue**

```vue
<script setup lang="ts">
import { ref } from 'vue'
import type { TicketStatus } from '@ops-ticket/shared'
const props = defineProps<{ticketNo:string; version:number; status:TicketStatus; username:string}>()
const emit = defineEmits<{reload:[]}>()
const message = ref('')
const busy = ref(false)
async function act(action:string) {
  if (busy.value) return
  busy.value = true
  try {
    const res = await fetch('/api/tickets/'+encodeURIComponent(props.ticketNo)+'/actions',{
      method:'POST',headers:{'Content-Type':'application/json','x-user':props.username},
      body:JSON.stringify({action,version:props.version}),
    })
    if (res.status===409) message.value='工单已变化，请核对最新状态后操作'
    else if (!res.ok) {message.value='提交失败，请核对操作权限';return}
    else message.value='反馈已提交'
    emit('reload')
  } catch {message.value='网络中断，请刷新核对结果'}
  finally {busy.value=false}
}
</script>
<template>
  <div v-if="status==='handoff_offered'">
    <button :disabled="busy" @click="act('request_human')">人工接入</button>
  </div>
  <div v-if="status==='awaiting_feedback'||status==='reviewed'">
    <button :disabled="busy" @click="act('resolved')">已解决</button>
    <button :disabled="busy" @click="act('unresolved')">未解决</button>
  </div>
  <p v-if="message" role="status">{{message}}</p>
</template>
```

```vue
<script setup lang="ts">
import { ref, watch } from 'vue'
import TicketFeedback from '../components/TicketFeedback.vue'
import { useRoute } from 'vue-router'
import type { TicketDetail } from '@ops-ticket/shared'
import { getTicket } from '../api/tickets'
const refresh = ref(0)
const username = ref('lin')
const route = useRoute()
const ticket = ref<TicketDetail>()
const error = ref('')
watch([() => String(route.params.ticketNo), refresh], async ([no], _, cleanup) => {
  const controller = new AbortController()
  cleanup(() => controller.abort())
  ticket.value = undefined
  error.value = ''
  try {
    const result = await getTicket(no, controller.signal)
    if (!controller.signal.aborted) ticket.value = result
  } catch {
    if (!controller.signal.aborted) error.value = '工单加载失败'
  }
}, { immediate: true })
</script>
<template>
  <RouterLink to="/">返回工单列表</RouterLink>
  <p v-if="error" role="alert">{{ error }}</p>
  <article v-if="ticket"><h2>{{ ticket.ticketNo }} · {{ ticket.title }}</h2>
    <p>{{ ticket.company.name }} · {{ ticket.category.name }} · {{ ticket.department.name }}</p>
    <p>{{ ticket.description }}</p><h3>报障材料</h3>
    <pre v-for="(m, i) in ticket.materials" :key="i">{{ m.logText }}</pre>
  <label>本地练习账号 <input v-model="username"></label>
    <TicketFeedback :ticket-no="ticket.ticketNo" :status="ticket.status" :version="ticket.version" :username="username" @reload="refresh++"/>

</template>
```
