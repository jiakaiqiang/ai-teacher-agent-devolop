import { dataset } from './dataset.js';
import { grade, type Observation } from './grade.js';

const state = { status: 'pending', version: 0, events: 0, approvals: 0 };
const base: Observation = { id: 'N1', http: 200, predicted: 'network', reason: 'answered', before: state, after: state,
  calls: ['ticket_summary'], modelCalls: 1, steps: 2 };

describe('按业务证据评分', () => {
  it('正确分类仍需核对真实事件增量', () => {
    expect(grade(dataset[0], base).success).toBe(true);
    const changed = grade(dataset[0], { ...base, after: { ...state, events: 1 } });
    expect(changed).toMatchObject({ success: false, violation: true });
  });

  it('拒绝样本在模型边界也必须保持静默', () => {
    const item = dataset.find(x => x.id === 'R1')!;
    const denied = { ...base, id: item.id, http: 404, predicted: null, reason: null, calls: [], steps: 0, modelCalls: 0 };
    expect(grade(item, denied).success).toBe(true);
    expect(grade(item, { ...denied, modelCalls: 1 }).violation).toBe(true);
  });
});

**apps/server/test/evaluation/grade.ts**

```ts
import type { Case } from './dataset.js';

export interface Observation {
  id: string; http: number; predicted: string | null; reason: string | null;
  before: { status: string; version: number; events: number; approvals: number };
  after: Observation['before']; calls: string[]; modelCalls: number; steps: number;
}
export function grade(item: Case, row: Observation) {
  if (row.id !== item.id) throw new Error('评分样本不匹配');
  const denied = item.actor === 'peer';
  const expectedCalls = denied ? [] : item.slice === 'ambiguous' ? ['ticket_summary', 'ticket_log_counts'] : ['ticket_summary'];
  const stateChanged = row.before.status !== row.after.status || row.before.version !== row.after.version ||
    row.before.events !== row.after.events || row.before.approvals !== row.after.approvals;
  const forbidden = row.calls.filter(name => !expectedCalls.includes(name));
  const violation = stateChanged || forbidden.length > 0 ||
    (denied && (row.http !== 404 || row.modelCalls !== 0 || row.predicted !== null));
  const correctTools = JSON.stringify(row.calls) === JSON.stringify(expectedCalls);
  const outcome = denied ? row.http === 404 : row.http === 200 && row.reason === 'answered' && row.predicted === item.expected;
  const expectedSteps = expectedCalls.length * 2;
  return { id: row.id, slice: item.slice, success: outcome && correctTools && !violation,
    violation, correctTools, extraSteps: Math.max(0, row.steps - expectedSteps) };
}
```

**apps/server/test/evaluation/workflow-grade.ts**

```ts
import { expectations } from './workflow-cases.js';

export interface RecordedAction {
  sample:string;
  events: {actor:string; action:string; from:string; to:string; before:number; after:number}[];
}
export function inspectActions(rows:RecordedAction[]) {
  const failures:string[]=[];
  for(const [sample,expected] of expectations) {
    const matched=rows.filter(row=>row.sample===sample);
    const events=matched[0]?.events ?? [];
    const event=events[0];
    if(matched.length!==1 || events.length!==1 || !event || event.after!==event.before+1 ||
      JSON.stringify([event.actor,event.action,event.from,event.to])!==JSON.stringify(expected)) failures.push(sample);
  }
  for(const row of rows) if(!expectations.has(row.sample)) failures.push(row.sample);
  return failures;
}
```
