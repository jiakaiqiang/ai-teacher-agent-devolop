```ts
import { describe, expect, it } from 'vitest';
import { transition, type TicketSnapshot, type Actor } from './workflow.js';
const reporter: Actor = {id:'lin',role:'reporter',companyId:'A'};
const agent: Actor = {id:'agent',role:'agent'};
const lead: Actor = {id:'lead',role:'department_lead',departmentId:'network'};
const engineer: Actor = {id:'eng',role:'operator',departmentId:'network'};
const initial: TicketSnapshot = {status:'agent_processing',reporterId:'lin',companyId:'A',departmentId:'network',assigneeId:null};
describe('按用户意愿交接', () => {
  it('无法排查时先等用户点击，再进入负责人待派', () => {
    const offered=transition(initial,agent,'offer_handoff');
    expect(offered.status).toBe('handoff_offered');
    expect(() => transition(offered,lead,'assign',engineer)).toThrow();
    expect(transition(offered,reporter,'request_human').status).toBe('pending_assignment');
  });
  it('用户拒绝建议直接转人工；人工结果被退回时回到原人', () => {
    const advised=transition(initial,agent,'suggest');
    const pending=transition(advised,reporter,'unresolved');
    const assigned=transition(pending,lead,'assign',engineer);
    const handled=transition(assigned,engineer,'handle');
    const reviewed=transition(handled,lead,'approve');
    expect(transition(reviewed,reporter,'unresolved')).toMatchObject({status:'human_processing',assigneeId:'eng'});
    expect(transition(reviewed,reporter,'resolved').status).toBe('completed');
  });
  it('客户管理员不能验收，也不能被派单', () => {
    const admin: Actor={id:'admin',role:'customer_admin',companyId:'A'};
    expect(() => transition({...initial,status:'awaiting_feedback'},admin,'resolved')).toThrow();
    expect(() => transition({...initial,status:'pending_assignment'},lead,'assign',admin)).toThrow();
  });
});
```

**apps/server/src/tickets/workflow.ts**

```ts
import { ConflictException, ForbiddenException } from '@nestjs/common';
import type { TicketStatus } from '@ops-ticket/shared';

export interface Actor { id: string; role: string; companyId?: string | null; departmentId?: string | null }
export interface TicketSnapshot {
  status: TicketStatus; reporterId: string; companyId: string; departmentId: string; assigneeId: string | null;
}
export type Action = 'suggest' | 'offer_handoff' | 'request_human' | 'unresolved' | 'assign' | 'handle' | 'approve' | 'return' | 'resolved';
type Rule = readonly [TicketStatus, Action, 'agent' | 'reporter' | 'department_lead' | 'operator', TicketStatus];
const rules: Rule[] = [
  ['agent_processing','suggest','agent','awaiting_feedback'],
  ['agent_processing','offer_handoff','agent','handoff_offered'],
  ['handoff_offered','request_human','reporter','pending_assignment'],
  ['awaiting_feedback','unresolved','reporter','pending_assignment'],
  ['awaiting_feedback','resolved','reporter','completed'],
  ['pending_assignment','assign','department_lead','human_processing'],
  ['human_processing','handle','operator','handled'],
  ['handled','approve','department_lead','reviewed'],
  ['handled','return','department_lead','human_processing'],
  ['reviewed','unresolved','reporter','human_processing'],
  ['reviewed','resolved','reporter','completed'],
];

export function transition(ticket: TicketSnapshot, actor: Actor, action: Action, target?: Actor): TicketSnapshot {
  const rule = rules.find(([status,event]) => status === ticket.status && event === action);
  if (!rule) throw new ConflictException('当前状态不允许此操作');
  const [, , role, status] = rule;
  if (role !== actor.role) throw new ForbiddenException('需要对应角色');
  if (role === 'reporter' && (actor.id !== ticket.reporterId || actor.companyId !== ticket.companyId))
    throw new ForbiddenException('仅提单人可反馈');
  if ((role === 'operator' || role === 'department_lead') && actor.departmentId !== ticket.departmentId)
    throw new ForbiddenException('工单归属其他部门');
  if (role === 'operator' && actor.id !== ticket.assigneeId)
    throw new ForbiddenException('工单已派给其他人');
  if (action === 'assign') {
    if (target?.role !== 'operator' || target.departmentId !== ticket.departmentId)
      throw new ForbiddenException('指派对象必须是本部门内部处理人');
    return { ...ticket, status, assigneeId: target.id };
  }
  return { ...ticket, status };
}
```

**apps/server/src/tickets/ticket-commands.ts**

```ts
import { ConflictException, ForbiddenException, Injectable, NotFoundException } from '@nestjs/common';
import { ticketStatusSchema } from '@ops-ticket/shared';
import { PrismaService } from '../prisma/prisma.service.js';
import { transition, type Action } from './workflow.js';

type Caller = { source: 'agent' } | { source:'user'; id:string };

@Injectable()
export class TicketCommands {
  constructor(private readonly db: PrismaService) {}
  // Caller 由受信入口构造。Agent 调用只来自服务端诊断任务。
  async apply(ticketNo:string, caller:Caller, action:Action, expectedVersion:number, targetId?:string, comment='') {
    return this.db.$transaction(async tx => {
      const row = await tx.ticket.findUnique({where:{ticketNo}});
      if (!row) throw new NotFoundException('工单不存在');
      const actor = caller.source === 'agent' ? {id:'agent',role:'agent'} :
        await tx.user.findFirst({where:{id:caller.id,registrationStatus:'approved'}});
      if (!actor) throw new ForbiddenException('账号未获批');
      const target = targetId ? await tx.user.findFirst({where:{id:targetId,registrationStatus:'approved'}}) : null;
      const next=transition({...row,status:ticketStatusSchema.parse(row.status)},actor,action,target ?? undefined);
      const changed=await tx.ticket.updateMany({where:{id:row.id,version:expectedVersion,status:row.status},
        data:{status:next.status,assigneeId:next.assigneeId,version:{increment:1},
          completedAt:next.status==='completed' ? new Date() : null}});
      if (changed.count!==1) throw new ConflictException('工单已变化，请刷新');
      return tx.ticketEvent.create({data:{ticketId:row.id,actorId:caller.source==='user'?caller.id:null,
        action,source:caller.source,fromStatus:row.status,toStatus:next.status,comment}});
    });
  }
}
```
