```ts
// 敏感字段脱敏（D19）
// 规则集中在这里：哪些字段算敏感、怎么遮、日志里要擦掉什么

// 响应里叫这些名字的字段一律视为敏感，不管它出现在哪个接口、嵌套在第几层
const SENSITIVE_KEYS = new Set(['phone']);

// 13800000001 → 138****0001。保留前 3 位和后 4 位，够用户认出“是我的号码”，又不能拿去打电话
export function maskPhone(phone: string | null): string | null {
  if (phone === null) return null;
  if (phone.length < 7) return '*'.repeat(phone.length);
  return `${phone.slice(0, 3)}****${phone.slice(-4)}`;
}

// 递归遍历响应数据，把所有敏感字段换成脱敏后的值。返回新对象，不修改原来的
export function maskSensitive<T>(value: T): T {
  if (Array.isArray(value)) {
    return value.map((item) => maskSensitive(item)) as T;
  }
  if (value instanceof Date || value === null || typeof value !== 'object') {
    return value;
  }
  const result: Record<string, unknown> = {};
  for (const [key, field] of Object.entries(value)) {
    result[key] = SENSITIVE_KEYS.has(key) && (typeof field === 'string' || field === null)
      ? maskPhone(field)
      : maskSensitive(field);
  }
  return result as T;
}

// 写日志前擦掉文本里的敏感内容：手机号、Bearer 令牌、密码哈希
// 异常信息里可能带着请求参数（例如 Prisma 的校验错误会把整个查询参数打印出来）
export function redactForLog(text: string): string {
  return text
    .replace(/\b1\d{10}\b/g, (phone) => maskPhone(phone)!)
    .replace(/Bearer\s+[\w-]+\.[\w-]+\.[\w-]+/g, 'Bearer [已隐藏]')
    .replace(/scrypt\$[^\s"'`]+/g, 'scrypt$[已隐藏]');
}
```

**apps/server/src/common/sensitive.spec.ts**

```ts
import { maskPhone, maskSensitive, redactForLog } from './sensitive.js';

describe('敏感字段脱敏', () => {
  it('手机号保留前 3 位和后 4 位', () => {
    expect(maskPhone('13800000001')).toBe('138****0001');
    expect(maskPhone(null)).toBeNull();
    expect(maskPhone('12345')).toBe('*****');
  });

  it('不管嵌套在第几层、在数组里还是对象里，叫 phone 的字段都被脱敏；其他字段和日期原样保留', () => {
    const createdAt = new Date('2026-11-15T00:00:00.000Z');
    const input = {
      ticketNo: 'T1',
      createdAt,
      reporter: { username: 'alice', phone: '13800000001' },
      assignee: null,
      events: [{ actor: { displayName: '李四', phone: '13800000002' } }],
    };
    const output = maskSensitive(input);
    expect(output).toEqual({
      ticketNo: 'T1',
      createdAt,
      reporter: { username: 'alice', phone: '138****0001' },
      assignee: null,
      events: [{ actor: { displayName: '李四', phone: '138****0002' } }],
    });
    // 不修改原来的对象
    expect(input.reporter.phone).toBe('13800000001');
  });

  it('日志里的手机号、令牌、密码哈希被擦掉', () => {
    const text =
      'Invalid `prisma.user.update()` invocation: { phone: "13800000001" } Authorization: Bearer aaa.bbb.ccc hash scrypt$32768$8$1$c2FsdA==$aGFzaA==';
    expect(redactForLog(text)).toBe(
      'Invalid `prisma.user.update()` invocation: { phone: "138****0001" } Authorization: Bearer [已隐藏] hash scrypt$[已隐藏]',
    );
  });
});
```

```ts
import { redactForLog } from '../common/sensitive.js';
import { visibleTickets } from './scope.js';
import { decodeCursor, encodeCursor } from '../common/cursor.js';
import type { ListTicketsQuery } from './dto/list-tickets.query.js';
import { Injectable, NotFoundException } from '@nestjs/common';
import type { Prisma } from '../generated/prisma/client.js';
import { PrismaService } from '../prisma/prisma.service.js';

const detailSelect = {
  version: true, ticketNo: true, title: true, description: true, status: true, createdAt: true,
  company: { select: { name: true } }, department: { select: { name: true } },
  category: { select: { name: true } }, reporter: { select: { displayName: true } },
  assignee: { select: { displayName: true } },
  materials: { select: { logText: true }, orderBy: { id: 'asc' } },
  events: { orderBy: [{ createdAt: 'asc' }, { id: 'asc' }],
    select: { action: true, toStatus: true, actor: { select: { displayName: true } } } },
} satisfies Prisma.TicketSelect;

export type TicketDetail = Prisma.TicketGetPayload<{ select: typeof detailSelect }>;

@Injectable()
export class TicketsService {
  constructor(private readonly prisma: PrismaService) {}
  private async scope(id: string) {
    return visibleTickets(await this.prisma.user.findUniqueOrThrow({where:{id}}));
  }
  async list(userId:string, { company, department, cursor, limit }: ListTicketsQuery) {
    const after = cursor ? decodeCursor(cursor) : undefined;
    const rows = await this.prisma.ticket.findMany({
      where: {
        AND: [await this.scope(userId)],
        company: company ? { name: company } : undefined,
        department: department ? { name: department } : undefined,
        OR: after ? [
          { createdAt: { lt: after.createdAt } },
          { createdAt: after.createdAt, id: { lt: after.id } },
        ] : undefined,
      },
      orderBy: [{createdAt:'desc'}, {id:'desc'}], take: limit + 1,
      select: { id:true, ticketNo:true, title:true, status:true, createdAt:true },
    });
    const page = rows.slice(0,limit);
    return {
      items: page.map(({ id: _id, ...item }) => item),
      nextCursor: rows.length > limit ? encodeCursor(page[page.length - 1]) : null,
    };
  }

  async findDetail(userId:string, ticketNo: string): Promise<TicketDetail> {
    const ticket = await this.prisma.ticket.findFirst({ where: { AND:[await this.scope(userId),{ ticketNo }] }, select: detailSelect });
    if (!ticket) throw new NotFoundException('工单不存在');
    return { ...ticket, materials:ticket.materials.map(m=>({logText:redactForLog(m.logText)})) };
  }
}
```

**apps/server/src/common/http-exception.filter.ts**

```ts
import { redactForLog } from './sensitive.js';
import {
  ArgumentsHost,
  Catch,
  ExceptionFilter,
  HttpException,
  HttpStatus,
  Logger,
} from '@nestjs/common';
import type { Response } from 'express';
import { Prisma } from '../generated/prisma/client.js';

// 错误响应统一结构：code 给程序判断，message 给人看，details 放校验失败的字段
export interface ErrorBody {
  error: { code: string; message: string; details?: unknown };
}

interface MappedError {
  status: number;
  code: string;
  message: string;
}

// 把数据库约束错误翻译成 HTTP 错误；约束名等内部细节只写日志，不返回给客户端
function mapPrismaError(
  error: Prisma.PrismaClientKnownRequestError,
): MappedError | undefined {
  switch (error.code) {
    case 'P2002': // 唯一约束冲突
      return { status: HttpStatus.CONFLICT, code: 'DUPLICATE_RESOURCE', message: '数据已存在' };
    case 'P2003': // 外键约束：被引用的行不能删，或引用的行不存在
      return { status: HttpStatus.CONFLICT, code: 'RESOURCE_IN_USE', message: '数据仍被其他记录引用，不能删除' };
    case 'P2025': // 要更新或删除的记录不存在
      return { status: HttpStatus.NOT_FOUND, code: 'NOT_FOUND', message: '记录不存在' };
  }
  // CHECK 约束没有专门的 Prisma 错误码，要看 PostgreSQL 的原始错误码 23514
  const cause = (error.meta?.driverAdapterError as { cause?: { originalCode?: string } } | undefined)?.cause;
  if (cause?.originalCode === '23514') {
    return { status: HttpStatus.UNPROCESSABLE_ENTITY, code: 'CONSTRAINT_VIOLATION', message: '数据不满足业务约束' };
  }
  return undefined;
}

@Catch()
export class HttpExceptionFilter implements ExceptionFilter {
  private readonly logger = new Logger(HttpExceptionFilter.name);

  catch(exception: unknown, host: ArgumentsHost) {
    const response = host.switchToHttp().getResponse<Response>();

    if (exception instanceof HttpException) {
      const status = exception.getStatus();
      const raw = exception.getResponse();
      // ValidationPipe 抛出的 400，message 是字段错误数组，放进 details
      const message =
        typeof raw === 'object' && raw !== null && 'message' in raw
          ? (raw as { message: unknown }).message
          : exception.message;
      const body: ErrorBody = {
        error: {
          code: HttpStatus[status] ?? 'HTTP_ERROR',
          message: Array.isArray(message) ? '请求参数不合法' : String(message),
          ...(Array.isArray(message) ? { details: message } : {}),
        },
      };
      response.status(status).json(body);
      return;
    }

    if (exception instanceof Prisma.PrismaClientKnownRequestError) {
      const mapped = mapPrismaError(exception);
      if (mapped) {
        // 约束名这类细节写日志，方便排查
        this.logger.warn(`${exception.code}: ${exception.message.split('\n').filter(Boolean).pop()}`);
        const body: ErrorBody = { error: { code: mapped.code, message: mapped.message } };
        response.status(mapped.status).json(body);
        return;
      }
    }

    // 未知异常：完整堆栈只写日志，不返回给客户端，避免泄露内部信息
    this.logger.error(exception instanceof Error ? exception.stack ? redactForLog(exception.stack) : undefined : exception);
    const body: ErrorBody = {
      error: { code: 'INTERNAL_SERVER_ERROR', message: '服务器内部错误' },
    };
    response.status(HttpStatus.INTERNAL_SERVER_ERROR).json(body);
  }
}
```
