## 一、知识点与推导

穿透来自反复读不存在的对象，击穿来自同一热点键的并发未命中，雪崩来自大量键同时失效。重建锁需有超时，等待者能恢复。

**apps/server/src/redis/cache.service.ts**

```ts
import { Injectable, Logger } from '@nestjs/common';
import { setTimeout as sleep } from 'node:timers/promises';
import { RedisService } from './redis.service.js';

// 三种缓存问题的防护（D24），按需打开
export interface CacheOptions {
  // 雪崩：过期时间再加 0 到 jitterSeconds 秒的随机数，同一批写进去的键不会在同一秒过期
  jitterSeconds?: number;
  // 穿透：数据库里也没有时，缓存一个空值这么多秒；不传就不缓存空值
  nullTtlSeconds?: number;
  // 击穿：未命中时只让一个请求查数据库，其余请求等它把缓存写好
  rebuildLock?: boolean;
}

// 重建锁最多占用 5 秒：查数据库的请求挂了，锁也会自己过期
const LOCK_TTL_MS = 5000;
// 没拿到锁的请求最多等 1 秒，每 20 毫秒看一次缓存；等不到就自己查，宁可多查一次也不让请求一直等
const LOCK_WAIT_MS = 1000;
const LOCK_POLL_MS = 20;

type CacheRead<T> = { hit: true; value: T } | { hit: false };

// 旁路缓存（Cache-Aside）的读路径（D22）：先读缓存，没有再读数据库，并把结果写回缓存
// 缓存只是加速，任何缓存错误都不能让请求失败：出错就当没命中，直接走数据库
@Injectable()
export class CacheService {
  private readonly logger = new Logger(CacheService.name);
  // 命中、未命中（也就是查了数据库）的次数，D27 用来算命中率
  readonly stats = { hits: 0, misses: 0, errors: 0 };

  constructor(private readonly redis: RedisService) {}

  // 写路径（D23）：数据库提交成功之后调用，删掉相关的缓存，下一次读会从数据库重新加载
  // 删失败只记日志：数据库已经改好了，不能因为缓存让写请求失败；最坏等到过期时间自动恢复
  async invalidate(...keys: string[]): Promise<void> {
    try {
      await this.redis.del(...keys);
    } catch (error) {
      this.stats.errors++;
      this.logger.warn(`删缓存失败，等过期后恢复：${keys.join(', ')}，${(error as Error).message}`);
    }
  }

  async getOrLoad<T>(key: string, ttlSeconds: number, loader: () => Promise<T>, options: CacheOptions = {}): Promise<T> {
    const cached = await this.read<T>(key);
    if (cached.hit) {
      return cached.value;
    }
    return options.rebuildLock
      ? this.loadWithLock(key, ttlSeconds, loader, options)
      : this.loadAndFill(key, ttlSeconds, loader, options);
  }

  // 缓存里的空值存成字符串 'null'，读出来是 null，也算命中
  private async read<T>(key: string): Promise<CacheRead<T>> {
    try {
      const raw = await this.redis.get(key);
      if (raw !== null) {
        const value = JSON.parse(raw) as T;
        this.stats.hits++;
        return { hit: true, value };
      }
    } catch (error) {
      // Redis 不可用、超时、缓存里的内容不是合法的 JSON：都按未命中处理
      this.stats.errors++;
      this.logger.warn(`读缓存失败，回退到数据库：${key}，${(error as Error).message}`);
    }
    return { hit: false };
  }

  private async loadAndFill<T>(key: string, ttlSeconds: number, loader: () => Promise<T>, options: CacheOptions): Promise<T> {
    this.stats.misses++;
    const value = await loader();
    const empty = value === null || value === undefined;
    const jitter = Math.floor(Math.random() * ((options.jitterSeconds ?? 0) + 1));
    const seconds = empty ? options.nullTtlSeconds : ttlSeconds + jitter;
    if (seconds) {
      try {
        await this.redis.set(key, JSON.stringify(empty ? null : value), 'EX', seconds);
      } catch (error) {
        this.stats.errors++;
        this.logger.warn(`写缓存失败：${key}，${(error as Error).message}`);
      }
    }
    return value;
  }

  private async loadWithLock<T>(key: string, ttlSeconds: number, loader: () => Promise<T>, options: CacheOptions): Promise<T> {
    const lockKey = `lock:${key}`;
    let locked: boolean;
    try {
      locked = (await this.redis.set(lockKey, '1', 'PX', LOCK_TTL_MS, 'NX')) === 'OK';
    } catch (error) {
      // Redis 不可用：没法加锁，直接查数据库
      this.stats.errors++;
      this.logger.warn(`加重建锁失败，直接查数据库：${key}，${(error as Error).message}`);
      return this.loadAndFill(key, ttlSeconds, loader, options);
    }

    if (locked) {
      try {
        return await this.loadAndFill(key, ttlSeconds, loader, options);
      } finally {
        // 这里直接删锁：最坏情况是多一个请求去查数据库。D25 讲这样释放锁为什么可能删掉别人的锁
        await this.redis.del(lockKey).catch(() => 0);
      }
    }

    const deadline = Date.now() + LOCK_WAIT_MS;
    while (Date.now() < deadline) {
      await sleep(LOCK_POLL_MS);
      const cached = await this.read<T>(key);
      if (cached.hit) {
        return cached.value;
      }
    }
    return this.loadAndFill(key, ttlSeconds, loader, options);
  }
}
```

**apps/server/src/redis/cache.service.spec.ts**

```ts
import { Logger } from '@nestjs/common';
import { CacheService } from './cache.service.js';
import type { RedisService } from './redis.service.js';

// 用一个假的 Redis：只实现 get、set、del，可以控制它返回什么、是否报错
function fakeRedis(store = new Map<string, string>(), broken = false) {
  return {
    store,
    get: vi.fn(async (key: string) => {
      if (broken) throw new Error('Connection is closed.');
      return store.get(key) ?? null;
    }),
    // 支持 SET key value PX 5000 NX：键已经存在时不写、返回 null（D24 重建锁）
    set: vi.fn(async (key: string, value: string, ...args: (string | number)[]) => {
      if (broken) throw new Error('Connection is closed.');
      if (args.includes('NX') && store.has(key)) return null;
      store.set(key, value);
      return 'OK';
    }),
    del: vi.fn(async (...keys: string[]) => {
      if (broken) throw new Error('Connection is closed.');
      return keys.filter((key) => store.delete(key)).length;
    }),
  };
}

describe('CacheService.getOrLoad（D22）', () => {
  beforeEach(() => {
    vi.spyOn(Logger.prototype, 'warn').mockImplementation(() => {});
  });

  it('第一次未命中：调用 loader 并写回缓存；第二次命中：不再调用 loader', async () => {
    const redis = fakeRedis();
    const cache = new CacheService(redis as unknown as RedisService);
    const loader = vi.fn().mockResolvedValue({ code: 'srv-001', name: '订单服务器' });

    expect(await cache.getOrLoad('device:srv-001', 300, loader)).toEqual({ code: 'srv-001', name: '订单服务器' });
    expect(await cache.getOrLoad('device:srv-001', 300, loader)).toEqual({ code: 'srv-001', name: '订单服务器' });

    expect(loader).toHaveBeenCalledTimes(1);
    expect(redis.set).toHaveBeenCalledWith('device:srv-001', '{"code":"srv-001","name":"订单服务器"}', 'EX', 300);
    expect(cache.stats).toEqual({ hits: 1, misses: 1, errors: 0 });
  });

  it('Redis 不可用：每次都直接走数据库，请求照样成功', async () => {
    const cache = new CacheService(fakeRedis(new Map(), true) as unknown as RedisService);
    const loader = vi.fn().mockResolvedValue({ code: 'srv-001' });

    expect(await cache.getOrLoad('device:srv-001', 300, loader)).toEqual({ code: 'srv-001' });
    expect(loader).toHaveBeenCalledTimes(1);
    expect(cache.stats.errors).toBe(2); // 读失败一次、写失败一次
  });

  it('缓存里的内容不是合法 JSON：当作未命中，用数据库的结果覆盖', async () => {
    const redis = fakeRedis(new Map([['device:srv-001', '{坏掉的']]));
    const cache = new CacheService(redis as unknown as RedisService);
    const loader = vi.fn().mockResolvedValue({ code: 'srv-001' });

    expect(await cache.getOrLoad('device:srv-001', 300, loader)).toEqual({ code: 'srv-001' });
    expect(redis.store.get('device:srv-001')).toBe('{"code":"srv-001"}');
  });

  it('数据库里也没有（null）：不写缓存', async () => {
    const redis = fakeRedis();
    const cache = new CacheService(redis as unknown as RedisService);
    expect(await cache.getOrLoad('device:nope', 300, async () => null)).toBeNull();
    expect(redis.set).not.toHaveBeenCalled();
  });
});

describe('CacheService.invalidate（D23）', () => {
  beforeEach(() => {
    vi.spyOn(Logger.prototype, 'warn').mockImplementation(() => {});
  });

  it('删掉以后下一次读重新调用 loader，拿到数据库里的新值', async () => {
    const redis = fakeRedis();
    const cache = new CacheService(redis as unknown as RedisService);
    await cache.getOrLoad('device:srv-001', 300, async () => ({ name: '旧名字' }));

    await cache.invalidate('device:srv-001');
    expect(await cache.getOrLoad('device:srv-001', 300, async () => ({ name: '新名字' }))).toEqual({ name: '新名字' });
  });

  it('Redis 不可用：删缓存失败不抛错，只记一次错误', async () => {
    const cache = new CacheService(fakeRedis(new Map(), true) as unknown as RedisService);
    await expect(cache.invalidate('device:srv-001')).resolves.toBeUndefined();
    expect(cache.stats.errors).toBe(1);
  });
});

describe('CacheService 的三种防护（D24）', () => {
  beforeEach(() => {
    vi.spyOn(Logger.prototype, 'warn').mockImplementation(() => {});
  });

  it('防穿透：数据库里没有，也缓存一个空值；第二次命中空值，不再调用 loader', async () => {
    const redis = fakeRedis();
    const cache = new CacheService(redis as unknown as RedisService);
    const loader = vi.fn().mockResolvedValue(null);

    expect(await cache.getOrLoad('device:nope', 300, loader, { nullTtlSeconds: 30 })).toBeNull();
    expect(await cache.getOrLoad('device:nope', 300, loader, { nullTtlSeconds: 30 })).toBeNull();

    expect(loader).toHaveBeenCalledTimes(1);
    expect(redis.set).toHaveBeenCalledWith('device:nope', 'null', 'EX', 30);
    expect(cache.stats).toEqual({ hits: 1, misses: 1, errors: 0 });
  });

  it('防雪崩：过期时间在 60 到 70 秒之间，并且不是都一样', async () => {
    const redis = fakeRedis();
    const cache = new CacheService(redis as unknown as RedisService);
    for (let i = 0; i < 200; i++) {
      await cache.getOrLoad(`k${i}`, 60, async () => i, { jitterSeconds: 10 });
    }
    const ttls = redis.set.mock.calls.map((call) => call[3] as number);
    expect(Math.min(...ttls)).toBeGreaterThanOrEqual(60);
    expect(Math.max(...ttls)).toBeLessThanOrEqual(70);
    expect(new Set(ttls).size).toBeGreaterThan(1);
  });

  it('防击穿：20 个请求同时未命中，只有一个调用 loader，其余等它写好缓存', async () => {
    const cache = new CacheService(fakeRedis() as unknown as RedisService);
    const loader = vi.fn(async () => {
      await new Promise((r) => setTimeout(r, 50)); // 模拟一个慢查询
      return { total: 42 };
    });

    const results = await Promise.all(
      Array.from({ length: 20 }, () => cache.getOrLoad('ticket-stats:all', 60, loader, { rebuildLock: true })),
    );

    expect(loader).toHaveBeenCalledTimes(1);
    expect(results.every((r) => r.total === 42)).toBe(true);
  });

  it('等不到别人写好缓存（锁被占着、缓存一直是空的）：等 1 秒后自己查', async () => {
    const redis = fakeRedis(new Map([['lock:ticket-stats:all', '1']]));
    const cache = new CacheService(redis as unknown as RedisService);
    const loader = vi.fn().mockResolvedValue({ total: 1 });

    const start = Date.now();
    expect(await cache.getOrLoad('ticket-stats:all', 60, loader, { rebuildLock: true })).toEqual({ total: 1 });
    expect(Date.now() - start).toBeGreaterThanOrEqual(1000);
    expect(loader).toHaveBeenCalledTimes(1);
  });

  it('Redis 不可用时加不了锁：直接查数据库，请求照样成功', async () => {
    const cache = new CacheService(fakeRedis(new Map(), true) as unknown as RedisService);
    const loader = vi.fn().mockResolvedValue({ total: 1 });
    expect(await cache.getOrLoad('ticket-stats:all', 60, loader, { rebuildLock: true })).toEqual({ total: 1 });
    expect(loader).toHaveBeenCalledTimes(1);
  });
});
```

```ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service.js';
import { CacheService } from '../redis/cache.service.js';
import { visibleTickets } from './scope.js';
@Injectable()
export class WorklistStats {
  constructor(private db:PrismaService,private cache:CacheService) {}
  async count(userId:string) {
    const user=await this.db.user.findUniqueOrThrow({where:{id:userId}});
    const scope=visibleTickets(user);
    const key='worklist:'+JSON.stringify(scope);
    return this.cache.getOrLoad(key,20,()=>this.db.ticket.count({where:scope}),{rebuildLock:true,jitterSeconds:5});
  }
}
```
