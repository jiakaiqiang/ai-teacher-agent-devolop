## 一、知识点与推导

描述只公开契约，调用时用真实主体重新授权并保留审计。

先从已登录用户取得自己的 runId。用工具目录查看允许的参数，发起一次 MCP 调用，再改用另一客户身份请求同一 runId。把工具回执与直接访问 HTTP 的结果并列。

客户端初始化后用 tools/list 读取名称、说明和 JSON Schema，再用 tools/call 提交参数。tool-catalog 已约定笔记文本长度和请求键格式。保留 Zod 对象的具体类型，SDK 就能从同一个定义生成协议描述并校验调用。

只读会话只注册 ticket_summary。工具参数中没有 runId、身份或权限标记，这些值将由建立连接的一方绑定。调用失败时把可公开的业务错误码写进工具结果，客户端仍然拥有协议连接。

**apps/server/src/mcp/server.spec.ts**

```ts
import { Client } from '@modelcontextprotocol/sdk/client/index.js';
import { InMemoryTransport } from '@modelcontextprotocol/sdk/inMemory.js';
import { ticketServer } from './server.js';

async function session(write = false) {
  const [left, right] = InMemoryTransport.createLinkedPair();
  const dispatch = vi.fn().mockResolvedValue({ ticketNo: 'DEMO-7' });
  const server = ticketServer(dispatch, write);
  const client = new Client({ name: 'lesson', version: '1' });
  await server.connect(right);
  await client.connect(left);
  return { client, dispatch, close: async () => { await client.close(); await server.close(); } };
}

describe('工单 MCP 契约', () => {
  it('只读目录仅列出工单摘要', async () => {
    const s = await session();
    try {
      expect((await s.client.listTools()).tools.map(x => x.name)).toEqual(['ticket_summary']);
      expect((await s.client.callTool({ name: 'ticket_summary', arguments: {} })).content)
        .toEqual([{ type: 'text', text: '{"ticketNo":"DEMO-7"}' }]);
    } finally { await s.close(); }
  });

  it('获准写笔记时发布原有参数结构', async () => {
    const s = await session(true);
    try {
      const note = (await s.client.listTools()).tools.find(x => x.name === 'save_private_note');
      expect(note?.inputSchema.required).toEqual(expect.arrayContaining(['text', 'requestKey']));
      const bad = await s.client.callTool({ name: 'save_private_note', arguments: {} });
      expect(bad.isError).toBe(true);
      expect(s.dispatch).not.toHaveBeenCalled();
    } finally { await s.close(); }
  });
});
```

**apps/server/src/mcp/server.ts**

```ts
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { catalog } from '../agent/tool-catalog.js';

export type TicketTool = 'ticket_summary' | 'save_private_note';
export type Dispatch = (name: TicketTool, args: Record<string, unknown>) => Promise<unknown>;

// 权限由服务端上下文传入，客户端提供的工具参数只能描述业务数据。
export function ticketServer(dispatch: Dispatch, writeNotes = false) {
  const server = new McpServer({ name: 'ops-ticket', version: '1.0.0' });
  function register(name: TicketTool) {
    const entry = catalog[name];
    server.registerTool(name, { description: entry.description, inputSchema: entry.schema }, async (args: unknown) => {
      try {
        const data = await dispatch(name, entry.schema.parse(args) as Record<string, unknown>);
        return { content: [{ type: 'text' as const, text: JSON.stringify(data) }] };
      } catch (error) {
        const code = error instanceof TicketToolFailure ? error.code : 'TICKET_UNAVAILABLE';
        return { isError: true, content: [{ type: 'text' as const, text: code }] };
      }
    });
  }
  register('ticket_summary');
  if (writeNotes) register('save_private_note');
  return server;
}

export class TicketToolFailure extends Error {
  constructor(readonly code: string) { super(code); }
}
```
