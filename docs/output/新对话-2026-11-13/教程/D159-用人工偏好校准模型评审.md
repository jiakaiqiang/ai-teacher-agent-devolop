## 一、知识点与推导

裁判偏好单列，权限、引用和业务事件仍由规则与人工证据判定。

将模型分类与工单流转分开判分。前者读固定题目的标签、工具顺序及模型输出，后者核对每条事件的主体、前态、后态和版本。任何一次越权读写都单独记录。

HH-RLHF helpful-base 测试集为同一问题提供人类偏好的回答和另一回答。选择六个简短日常问题，冻结文件校验和与行号。第一次把偏好答案放 A，第二次放 B。请求只包含问题、A、B。

用 JSON schema 约束裁判输出，逐次记录模型版本、位置和答案。无效票仍计入十二次呈现；裁判与人工一致率低时，只宜把分歧用于人工排查。

import { agreement, blind } from './judge.js';

describe('人工偏好的位置对照', () => {
  it('交换位置并从裁判输入拿掉金标', () => {
    const pair = { line: 4, question: '合成问题', chosen: '甲', rejected: '乙' };
    expect(blind(pair, true)).toEqual({ question: '合成问题', A: '乙', B: '甲' });
    expect(blind(pair, false)).toEqual({ question: '合成问题', A: '甲', B: '乙' });
  });

  it('固定选首项只能获得一半逐次一致', () => {
    const votes = [{ line: 4, reverse: false, vote: 'A' as const }, { line: 4, reverse: true, vote: 'A' as const }];
    expect(agreement([4], votes)).toMatchObject({ presentations: 2, matched: 1, bothRight: 0, inconsistent: 1 });
    expect(() => agreement([4], [votes[0], votes[0]])).toThrow('不完整');
  });
});

**apps/server/test/evaluation/judge.ts**

```ts
export type Vote = 'A' | 'B' | 'invalid';
export interface Pair { line: number; question: string; chosen: string; rejected: string }
export interface Ballot { line: number; reverse: boolean; vote: Vote }

export function blind(pair: Pair, reverse: boolean) {
  return { question: pair.question, A: reverse ? pair.rejected : pair.chosen, B: reverse ? pair.chosen : pair.rejected };
}

export function agreement(lines: number[], votes: Ballot[]) {
  if (!lines.length || new Set(lines).size !== lines.length || votes.length !== 2 * lines.length) throw new Error('配对记录不完整');
  let matched = 0, bothRight = 0, inconsistent = 0, invalid = 0;
  for (const line of lines) {
    const group = votes.filter(v => v.line === line);
    const normal = group.find(v => !v.reverse), swapped = group.find(v => v.reverse);
    if (group.length !== 2 || !normal || !swapped) throw new Error('配对记录不完整');
    matched += Number(normal.vote === 'A') + Number(swapped.vote === 'B');
    bothRight += Number(normal.vote === 'A' && swapped.vote === 'B');
    invalid += group.filter(v => v.vote === 'invalid').length;
    if (normal.vote !== 'invalid' && swapped.vote !== 'invalid' && normal.vote === swapped.vote) inconsistent++;
  }
  return { presentations: votes.length, matched, bothRight, inconsistent, invalid, pairs: lines.length };
}
```

**apps/server/test/evaluation/run-judge.ts**

```ts
import { readFileSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { gunzipSync } from 'node:zlib';
import { agreement, blind, type Ballot } from './judge.js';

// helpful-base/test.jsonl.gz，以下为零起始行号。只将候选文本送给本机裁判。
const lines = [31, 709, 801, 1231, 1485, 2109];
const revision = '09be8c5bbc57cb3887f3a9732ad6aa7ec602a1fa';
const packed = readFileSync(process.env.EVAL_HH_FILE ?? 'helpful-base-test.jsonl.gz');
if (createHash('sha256').update(packed).digest('hex') !== '8be3fc1a13b27901631696f2be6f184c799f1baeaf145f53ac5db24960adc37b') throw new Error('语料版本不符');
const source = gunzipSync(packed).toString('utf8').trimEnd().split('\n');
const host = 'http://127.0.0.1:11434', model = 'tinyllama:latest';
const tags = await (await fetch(host + '/api/tags')).json();
const digest = tags.models.find((x: { name: string }) => x.name === model)?.digest;
if (!digest) throw new Error('本机裁判模型不可用');
const ballots: Ballot[] = [];
for (const line of lines) {
  const row = JSON.parse(source[line]);
  const [question, chosen] = row.chosen.split('\n\nAssistant:');
  const [same, rejected] = row.rejected.split('\n\nAssistant:');
  if (same !== question || !chosen || !rejected) throw new Error('问题和候选未配对');
  for (const reverse of [false, true]) {
    const input = blind({ line, question, chosen, rejected }, reverse);
    const response = await fetch(host + '/api/chat', { method: 'POST', signal: AbortSignal.timeout(30_000),
      headers: { 'content-type': 'application/json' }, body: JSON.stringify({ model, stream: false,
        messages: [{ role: 'system', content: 'Select the more helpful and accurate answer to the question. Return your choice in the vote field.' },
          { role: 'user', content: JSON.stringify(input) }], options: { seed: 23, temperature: 0, num_predict: 32 },
        format: { type: 'object', properties: { vote: { type: 'string', enum: ['A', 'B'] } }, required: ['vote'] } }) });
    if (!response.ok) throw new Error('裁判调用失败 ' + response.status);
    const result = await response.json();
    let vote: Ballot['vote'] = 'invalid';
    try {
      const answer = JSON.parse(result.message.content);
      if (answer.vote === 'A' || answer.vote === 'B') vote = answer.vote;
    } catch { /* 无效票也保留在分母。 */ }
    ballots.push({ line, reverse, vote });
  }
}
const report = { model, digest, revision, lines, ballots, ...agreement(lines, ballots) };
console.log(JSON.stringify(report));
if (process.env.EVAL_JUDGE_REPORT) writeFileSync(process.env.EVAL_JUDGE_REPORT, JSON.stringify(report, null, 2));
```
