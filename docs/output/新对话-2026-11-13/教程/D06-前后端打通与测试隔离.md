## 一、知识点与推导

静态类型无法检查 HTTP 返回的 JSON。前端先校验统一响应，再校验 data。列表需要 items 和 nextCursor，详情必须区分 company、department、materials 和 events。Vitest 在启动工作进程前准备独立库、执行迁移和种子，并把测试 URL 显式传给服务。

```ts
// 测试库的连接串：在开发库的库名后面加 _test，例如 ops_ticket → ops_ticket_test
// 已经是 _test 结尾的就原样返回，多次调用结果不变
export function testDatabaseUrl(devUrl: string | undefined): string {
  if (!devUrl) {
    throw new Error('缺少 DATABASE_URL，请检查 apps/server/.env');
  }
  const url = new URL(devUrl);
  const name = url.pathname.slice(1);
  if (!name.endsWith('_test')) {
    url.pathname = `/${name}_test`;
  }
  return url.toString();
}

// 测试会删除并重建这个库，所以只允许操作名字以 _test 结尾的库，防止误删开发库
export function assertTestDatabase(url: string): string {
  const name = new URL(url).pathname.slice(1);
  if (!name.endsWith('_test')) {
    throw new Error(`拒绝操作数据库 ${name}：测试只能使用名字以 _test 结尾的库`);
  }
  return name;
}

// 同一台 PostgreSQL 上的维护库 postgres，用来执行建库和删库（不能连着要删的库去删它自己）
export function maintenanceDatabaseUrl(url: string): string {
  const maintenance = new URL(url);
  maintenance.pathname = '/postgres';
  return maintenance.toString();
}
```

```ts
// 接口测试开始前执行一次：删除并重建测试库，执行全部迁移，放入种子数据
// 这样每次测试都从同一份已知的数据开始，不受开发库里的批量数据和手工改动影响
import 'dotenv/config';
import { execSync } from 'node:child_process';
import pg from 'pg';
import {
  assertTestDatabase,
  maintenanceDatabaseUrl,
  testDatabaseUrl,
} from './test-database.js';

export default async function setup() {
  const testUrl = testDatabaseUrl(process.env.DATABASE_URL);
  const testDbName = assertTestDatabase(testUrl);
  if (!/^[A-Za-z0-9_]+$/.test(testDbName)) throw new Error("测试库名包含无效字符");

  const admin = new pg.Client({ connectionString: maintenanceDatabaseUrl(testUrl) });
  await admin.connect();
  try {
    // WITH (FORCE) 会断开还连着测试库的会话，例如上次测试异常退出留下的连接
    await admin.query(`DROP DATABASE IF EXISTS "${testDbName}" WITH (FORCE)`);
    await admin.query(`CREATE DATABASE "${testDbName}"`);
  } finally {
    await admin.end();
  }

  // 用测试库的连接串执行迁移和种子
  const env = { ...process.env, DATABASE_URL: testUrl };
  execSync('pnpm exec prisma migrate deploy', { env, stdio: 'pipe' });
  execSync('pnpm exec prisma db seed', { env, stdio: 'pipe' });
}
```

```ts
import { apiErrorSchema } from '@ops-ticket/shared'
import { z } from 'zod'

// 前端统一的请求错误：code 用来判断，message 用来展示
export class ApiError extends Error {
  // 前端开启了 erasableSyntaxOnly，不能用构造函数参数属性的写法，字段要单独声明
  readonly status: number
  readonly code: string

  constructor(status: number, code: string, message: string) {
    super(message)
    this.name = 'ApiError'
    this.status = status
    this.code = code
  }
}

const envelopeSchema = z.object({ data: z.unknown() })

// 发请求并按契约校验响应。后端返回的结构和 shared 里的 schema 对不上时，
// 在这里就报 CONTRACT_MISMATCH，而不是让页面拿着错误的数据渲染到一半出错
export async function getJson<T extends z.ZodType>(
  path: string,
  schema: T,
  signal?: AbortSignal,
): Promise<z.infer<T>> {
  const response = await fetch(path, { signal, headers: { Accept: 'application/json' } })
  const body: unknown = await response.json().catch(() => null)

  if (!response.ok) {
    const parsed = apiErrorSchema.safeParse(body)
    if (parsed.success) {
      throw new ApiError(response.status, parsed.data.error.code, parsed.data.error.message)
    }
    // 代理或网关返回的错误（例如后端没启动时的 502）没有统一结构
    throw new ApiError(response.status, 'HTTP_ERROR', `请求失败（HTTP ${response.status}）`)
  }

  // 先拆出 { data } 外壳，再按具体接口的 schema 校验 data
  const envelope = envelopeSchema.safeParse(body)
  const parsed = schema.safeParse(envelope.success ? envelope.data.data : undefined)
  if (!parsed.success) {
    // 开发时在控制台打印具体是哪个字段不符合契约
    console.error('接口响应不符合契约', path, parsed.error.issues)
    throw new ApiError(response.status, 'CONTRACT_MISMATCH', '接口返回的数据格式不正确')
  }
  return parsed.data
}
```

```ts
import {
  ticketDetailSchema,
  ticketPageSchema,
  type TicketDetail,
  type TicketPage,
} from '@ops-ticket/shared'
import { getJson } from './http'

export interface ListTicketsParams {
  company?: string
  cursor?: string | null
  limit?: number
}

export function listTickets(params: ListTicketsParams, signal?: AbortSignal): Promise<TicketPage> {
  const query = new URLSearchParams()
  if (params.company) query.set('company', params.company)
  if (params.cursor) query.set('cursor', params.cursor)
  query.set('limit', String(params.limit ?? 20))
  return getJson(`/api/tickets?${query}`, ticketPageSchema, signal)
}

export function getTicket(ticketNo: string, signal?: AbortSignal): Promise<TicketDetail> {
  return getJson(`/api/tickets/${encodeURIComponent(ticketNo)}`, ticketDetailSchema, signal)
}
```

**apps/web/src/composables/useTicketList.ts**

```ts
import type { TicketListItem } from '@ops-ticket/shared'
import { computed, ref, shallowRef, watch, type Ref } from 'vue'
import { ApiError } from '../api/http'
import { listTickets } from '../api/tickets'

// 工单列表的游标分页状态：切换筛选条件时清空重来，触底时加载下一页
export function useTicketList(company: Ref<string | undefined>, pageSize = 20) {
  const items = shallowRef<TicketListItem[]>([])
  const nextCursor = ref<string | null>(null)
  const loading = ref(false)
  const error = ref<string | null>(null)
  const started = ref(false)

  // 还没加载过，或者上一页给出了游标，才有下一页
  const hasMore = computed(() => !started.value || nextCursor.value !== null)

  let controller: AbortController | null = null

  async function loadMore() {
    // 正在加载或已经到底时不重复请求
    if (loading.value || !hasMore.value) return
    loading.value = true
    error.value = null
    controller = new AbortController()
    const signal = controller.signal
    try {
      const page = await listTickets(
        { company: company.value, cursor: nextCursor.value, limit: pageSize },
        signal,
      )
      if (signal.aborted) return
      items.value = [...items.value, ...page.items]
      nextCursor.value = page.nextCursor
      started.value = true
    } catch (e) {
      // 被 reset 取消的请求不算错误
      if (signal.aborted) return
      error.value = e instanceof ApiError ? e.message : '网络异常，请稍后重试'
    } finally {
      if (!signal.aborted) loading.value = false
    }
  }

  // 筛选条件变了：取消进行中的请求，清空列表和游标，从第一页开始（游标不能跨筛选条件使用）
  function reset() {
    controller?.abort()
    controller = null
    items.value = []
    nextCursor.value = null
    started.value = false
    loading.value = false
    error.value = null
    void loadMore()
  }

  watch(company, reset)

  const stop = () => controller?.abort()
  return { stop, items, loading, error, hasMore, loadMore, reset }
}
```
