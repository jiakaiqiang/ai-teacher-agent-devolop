import { embeddingUsage, quoteUsage } from './cost-quote.js';
import { UsageMeter } from './usage-meter.js';

describe('演示报价与用量核对', () => {
  it('缺失用量不是零费用', () => {
    const meter = new UsageMeter();
    meter.start();
    expect(quoteUsage(meter.summary())).toMatchObject({
      status: 'unknown', amount: null, knownAmount: '0.000000',
    });
  });
  it('输入输出分价且不把额度当货币', () => {
    expect(quoteUsage({ inputTokens: 10, outputTokens: 3, exact: true, attempts: 1 }))
      .toMatchObject({ amount: '0.000038', tariffKind: 'synthetic', tariffId: 'demo-linear-v1' });
  });
  it('本机embedding计输入token而非维度', () => {
    const usage = embeddingUsage({ prompt_eval_count: 12, embeddings: [Array(1024).fill(0)] });
    expect(usage).toEqual({ inputTokens: 12, outputTokens: 0, exact: true, attempts: 1 });
    expect(quoteUsage(usage).amount).toBe('0.000024');
  });
  it('重试已知部分不能伪装完整账单', () => {
    const meter = new UsageMeter();
    meter.start(); // 第一次失败且没有usage。
    meter.start();
    meter.record({ usage: { prompt_tokens: 10, completion_tokens: 3, total_tokens: 13 } });
    expect(quoteUsage(meter.summary())).toMatchObject({
      attempts: 2, amount: null, knownAmount: '0.000038',
    });
  });
  it('整数微元累计不提前截断小额', () => {
    expect(quoteUsage({ inputTokens: 1, outputTokens: 1, exact: true, attempts: 1 }).amount).toBe('0.000008');
    expect(quoteUsage({ inputTokens: 1_000_000, outputTokens: 1_000_000, exact: true, attempts: 2 }).amount).toBe('8.000000');
  });
  it('非法计数与不可能的零尝试被拒绝', () => {
    for (const count of [NaN, -1, 0.5, 1_000_001]) {
      expect(embeddingUsage({ prompt_eval_count: count }).exact).toBe(false);
    }
    expect(embeddingUsage({}).exact).toBe(false);
    expect(() => quoteUsage({ inputTokens: 1, outputTokens: 0, exact: true, attempts: 0 })).toThrow('USAGE_INVALID');
  });
});

**apps/server/src/assistant/cost-quote.ts**

```ts
import type { UsageSummary } from './usage-meter.js';

// 演示价表，不是供应商报价，更不代表本机实际成本。
export const demoTariff = Object.freeze({
  id: 'demo-linear-v1', currency: 'CNY', inputMicroPerToken: 2n, outputMicroPerToken: 6n,
});

function validate(summary: UsageSummary) {
  const counts = [summary.inputTokens, summary.outputTokens];
  if (counts.some(v => !Number.isSafeInteger(v) || v < 0 || v > 4_000_000) ||
      !Number.isSafeInteger(summary.attempts) || summary.attempts < 0 || summary.attempts > 4 ||
      typeof summary.exact !== 'boolean' ||
      (summary.attempts === 0 && counts.some(v => v !== 0))) throw new Error('USAGE_INVALID');
}

function formatMicro(value: bigint) {
  return (value / 1_000_000n).toString() + '.' + (value % 1_000_000n).toString().padStart(6, '0');
}

export function quoteUsage(summary: UsageSummary) {
  validate(summary);
  const micro = BigInt(summary.inputTokens) * demoTariff.inputMicroPerToken +
    BigInt(summary.outputTokens) * demoTariff.outputMicroPerToken;
  const knownAmount = formatMicro(micro);
  return {
    tariffId: demoTariff.id, currency: demoTariff.currency, tariffKind: 'synthetic' as const,
    status: summary.exact ? 'complete-usage' as const : 'unknown' as const,
    amount: summary.exact ? knownAmount : null, knownAmount,
    attempts: summary.attempts,
  };
}

// embed只有输入token；向量维度不属于completion token。
export function embeddingUsage(raw: unknown): UsageSummary {
  const count = raw && typeof raw === 'object' && 'prompt_eval_count' in raw ? raw.prompt_eval_count : undefined;
  const exact = typeof count === 'number' && Number.isSafeInteger(count) && count >= 0 && count <= 1_000_000;
  return { inputTokens: exact ? count : 0, outputTokens: 0, exact, attempts: 1 };
}
```

**scripts/observe-embedding-usage.mjs**

```text
import assert from 'node:assert/strict';
import { model, dimension } from '../apps/server/dist/retrieval/encoding.js';
import { embeddingUsage, quoteUsage } from '../apps/server/dist/assistant/cost-quote.js';

// 固定合成问题；只输出用量元数据，不打印向量或环境变量。
const start = performance.now();
const response = await fetch('http://127.0.0.1:11434/api/embed', {
  method: 'POST', redirect: 'error', signal: AbortSignal.timeout(20000),
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ model, input: ['VPN登录失败，如何核对动态口令？'], truncate: false }),
});
assert.equal(response.status, 200);
const raw = await response.json();
assert.equal(raw.model, model);
assert.equal(raw.embeddings.length, 1);
assert.equal(raw.embeddings[0].length, dimension);
assert.ok(raw.embeddings[0].every(Number.isFinite));
const usage = embeddingUsage(raw);
assert.ok(usage.exact);
assert.ok(Number.isSafeInteger(raw.total_duration) && raw.total_duration >= 0);
console.log(JSON.stringify({
  source: 'local-ollama-embedding', model, dimension, usage,
  clientTotalMs: performance.now() - start, providerTotalMs: raw.total_duration / 1e6,
  demoQuote: quoteUsage(usage), actualCost: null,
}));
```
