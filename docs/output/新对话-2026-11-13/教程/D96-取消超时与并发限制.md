### 1.1 并发上限与任务数量

async with semaphore限制同时进入worker的数量。等待许可的任务仍存在，所以本章同时限制批次至多100项。它不是QPS限速器，也不能让CPU计算自动并行。共享active/peak计数器测同一批任务，避免每个任务各计各的。

**可运行示例：test_peak_concurrency_and_order**

来源：`python/bridge/tests/test_bounded.py`。

```python
def test_peak_concurrency_and_order() -> None:
    async def scenario() -> None:
        ready, release = asyncio.Event(), asyncio.Event()
        active = peak = 0

        async def worker(ticket: TicketSnapshot) -> str:
            nonlocal active, peak
            active += 1
            peak = max(peak, active)
            if active == 2:
                ready.set()
            try:
                await release.wait()
                return ticket.ticket_no
            finally:
                active -= 1

        task = asyncio.create_task(run_bounded(TICKETS, worker, concurrency=2, timeout=2))
        await ready.wait()
        release.set()
        result = await task
        assert peak == 2
        assert active == 0
        assert result == ("0", "1", "2", "3", "4")

    asyncio.run(scenario())
```

### 1.2 超时包含排队和执行

外层asyncio.timeout包裹任务组，预算从组开始前计算。超时取消当前执行路径，任务组清理子任务，离开timeout上下文才得到TimeoutError。本章测排队工单尚未进入worker，正在执行的任务完成finally。

**可运行示例：test_timeout_includes_queue_and_cleans_active_worker**

来源：`python/bridge/tests/test_bounded.py`。

```python
def test_timeout_includes_queue_and_cleans_active_worker() -> None:
    async def scenario() -> None:
        started: list[str] = []
        cleaned: list[str] = []

        async def worker(ticket: TicketSnapshot) -> str:
            started.append(ticket.ticket_no)
            try:
                await asyncio.Event().wait()
                return "不可到达"
            finally:
                cleaned.append(ticket.ticket_no)

        with pytest.raises(TimeoutError):
            await run_bounded(TICKETS, worker, concurrency=1, timeout=0.02)

        assert cleaned == ["0"]
```

### 1.3 外部取消保留原意

调用者cancel不改写成TimeoutError或普通业务失败。只在finally做清理，允许CancelledError继续传播。创建任务前验证预算，防止0并发等待或无效超时拖入运行过程。依据：[取消与超时](https://docs.python.org/3.12/library/asyncio-task.html#timeouts)、[Semaphore](https://docs.python.org/3.12/library/asyncio-sync.html#semaphore)。

**可运行示例：test_external_cancellation_is_not_timeout**

来源：`python/bridge/tests/test_bounded.py`。

```python
def test_external_cancellation_is_not_timeout() -> None:
    async def scenario() -> None:
        started = asyncio.Event()
        cleaned: list[str] = []

        async def worker(ticket: TicketSnapshot) -> str:
            try:
                started.set()
                await asyncio.Event().wait()
                return "不可到达"
            finally:
                cleaned.append(ticket.ticket_no)

        task = asyncio.create_task(run_bounded(TICKETS, worker, concurrency=1, timeout=2))
        await started.wait()
        task.cancel()
        with pytest.raises(asyncio.CancelledError):
            await task
        assert cleaned == ["0"]

    asyncio.run(scenario())
```

## 四、本章完整源码

**python/bridge/ops_bridge/bounded.py**

```python
"""有限批次、有限并发、整批截止时间；不强行终止不合作的代码。"""
import asyncio
import math

from ops_bridge.contracts import TicketWorker
from ops_bridge.models import TicketSnapshot


async def run_bounded(
    tickets: tuple[TicketSnapshot, ...], worker: TicketWorker,
    *, concurrency: int, timeout: float,
) -> tuple[str, ...]:
    # 创建协程前先拒绝无效预算，避免未等待协程和无限等待。
    if type(concurrency) is not int or not 1 <= concurrency <= 8:
        raise ValueError("并发必须为1到8的整数")
    if isinstance(timeout, bool) or not math.isfinite(timeout) or timeout <= 0:
        raise ValueError("超时必须为有限正数")
    if len(tickets) > 100:
        raise ValueError("批次不能超过100项")
    semaphore = asyncio.Semaphore(concurrency)

    async def one(ticket: TicketSnapshot) -> str:
        async with semaphore:
            return await worker(ticket)

    # 超时包含排队与执行；退出TaskGroup时等待子任务的finally完成。
    async with asyncio.timeout(timeout):
        async with asyncio.TaskGroup() as group:
            tasks = [group.create_task(one(ticket)) for ticket in tickets]
    return tuple(task.result() for task in tasks)
```
