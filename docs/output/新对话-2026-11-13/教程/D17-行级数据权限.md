把本人、本公司、本部门和分配给本人的数据范围编译成 Prisma 查询条件。列表、详情和材料同时执行这些条件。

## 零、闭卷回忆

接口权限与数据范围分别回答什么？客户端传入乙公司的筛选值能改变登录人的公司吗？用户确认后是否能公开原日志？

## 一、知识点与推导

角色条件和用户筛选取交集。详情使用 findFirst 把范围与工单号一起下推，越界与不存在按同一方式返回。部门负责人可跨客户承办，普通处理人还要满足 assigneeId。

## 二、动手开发

先用 lin 身份按乙公司筛选，并直接请求 T-002。接口应返回空列表与 404。在 scope.ts 为四类用户计算范围，由 controller 传受信用户 id，service 在查询里组合范围。

## 三、自检与提交

用网络部负责人读两家公司的单，再检查普通提单人只有自己的单。修改 URL 中的公司或工单号无法放宽范围；详情内嵌材料受同一条件约束。

```ts
import { ForbiddenException } from '@nestjs/common';
import type { Prisma } from '../generated/prisma/client.js';

export interface ScopeUser {id:string;role:string;companyId:string|null;departmentId:string|null}
export function visibleTickets(user:ScopeUser):Prisma.TicketWhereInput {
  switch(user.role) {
    case 'reporter':
      if (user.companyId) return {companyId:user.companyId,reporterId:user.id};
      break;
    case 'customer_admin':
      if (user.companyId) return {companyId:user.companyId};
      break;
    case 'department_lead':
      if (user.departmentId) return {departmentId:user.departmentId};
      break;
    case 'operator':
      if (user.departmentId) return {departmentId:user.departmentId,assigneeId:user.id};
      break;
  }
  throw new ForbiddenException('缺少有效的数据范围');
}
```

```ts
import type { AuthenticatedRequest } from '../auth/auth.types.js';
import { RequirePermission } from '../auth/permissions.decorator.js';
import { ListTicketsQuery } from './dto/list-tickets.query.js';
import { Controller, Get, Param, Query, Req } from '@nestjs/common';
import { TicketsService, type TicketDetail } from './tickets.service.js';

@RequirePermission('ticket:read')
@Controller('tickets')
export class TicketsController {
  constructor(private readonly ticketsService: TicketsService) {}

  @Get()
  list(@Req() req:AuthenticatedRequest, @Query() query: ListTicketsQuery) {
    return this.ticketsService.list(req.user!.id,query);
  }

  // 用给人看的工单编号查询，不在 URL 里暴露内部主键
  @Get(':ticketNo')
  findDetail(@Req() req:AuthenticatedRequest, @Param('ticketNo') ticketNo: string): Promise<TicketDetail> {
    return this.ticketsService.findDetail(req.user!.id,ticketNo);
  }
}
```

**apps/server/src/tickets/tickets.service.ts**

```ts
import { visibleTickets } from './scope.js';
import { decodeCursor, encodeCursor } from '../common/cursor.js';
import type { ListTicketsQuery } from './dto/list-tickets.query.js';
import { Injectable, NotFoundException } from '@nestjs/common';
import type { Prisma } from '../generated/prisma/client.js';
import { PrismaService } from '../prisma/prisma.service.js';

const detailSelect = {
  version: true, ticketNo: true, title: true, description: true, status: true, createdAt: true,
  company: { select: { name: true } }, department: { select: { name: true } },
  category: { select: { name: true } }, reporter: { select: { displayName: true } },
  assignee: { select: { displayName: true } },
  materials: { select: { logText: true }, orderBy: { id: 'asc' } },
  events: { orderBy: [{ createdAt: 'asc' }, { id: 'asc' }],
    select: { action: true, toStatus: true, actor: { select: { displayName: true } } } },
} satisfies Prisma.TicketSelect;

export type TicketDetail = Prisma.TicketGetPayload<{ select: typeof detailSelect }>;

@Injectable()
export class TicketsService {
  constructor(private readonly prisma: PrismaService) {}
  private async scope(id: string) {
    return visibleTickets(await this.prisma.user.findUniqueOrThrow({where:{id}}));
  }
  async list(userId:string, { company, department, cursor, limit }: ListTicketsQuery) {
    const after = cursor ? decodeCursor(cursor) : undefined;
    const rows = await this.prisma.ticket.findMany({
      where: {
        AND: [await this.scope(userId)],
        company: company ? { name: company } : undefined,
        department: department ? { name: department } : undefined,
        OR: after ? [
          { createdAt: { lt: after.createdAt } },
          { createdAt: after.createdAt, id: { lt: after.id } },
        ] : undefined,
      },
      orderBy: [{createdAt:'desc'}, {id:'desc'}], take: limit + 1,
      select: { id:true, ticketNo:true, title:true, status:true, createdAt:true },
    });
    const page = rows.slice(0,limit);
    return {
      items: page.map(({ id: _id, ...item }) => item),
      nextCursor: rows.length > limit ? encodeCursor(page[page.length - 1]) : null,
    };
  }

  async findDetail(userId:string, ticketNo: string): Promise<TicketDetail> {
    const ticket = await this.prisma.ticket.findFirst({ where: { AND:[await this.scope(userId),{ ticketNo }] }, select: detailSelect });
    if (!ticket) throw new NotFoundException('工单不存在');
    return ticket;
  }
}
```

```ts
import { PrismaService } from '../src/prisma/prisma.service.js';
import { bearer } from './login-fixtures.js';
import { Test } from '@nestjs/testing';
import type { INestApplication } from '@nestjs/common';
import request from 'supertest';
import { afterAll, beforeAll, expect, it } from 'vitest';
import { AppModule } from '../src/app.module.js';
import { configureApp } from '../src/app.setup.js';

let app: INestApplication;
let lead: string;
beforeAll(async () => {
  const mod = await Test.createTestingModule({ imports: [AppModule] }).compile();
  app = mod.createNestApplication();
  configureApp(app);
  await app.init();
  const db=app.get(PrismaService);
  const dept=await db.department.findUniqueOrThrow({where:{name:'网络部'}});
  lead=(await db.user.create({data:{username:'lead-'+crypto.randomUUID(),displayName:'主管',role:'department_lead',departmentId:dept.id,registrationStatus:'approved'}})).username;
});
afterAll(async () => { if (app && lead) await app.get(PrismaService).user.deleteMany({where:{username:lead}}); await app?.close(); });

it('客户和部门是两个独立的筛选维度', async () => {
  const customer = await request(app.getHttpServer()).get('/api/tickets').set('Authorization',await bearer(app,lead)).query({ company: '甲公司' }).expect(200);
  expect(customer.body.data.items.map((r: { ticketNo: string }) => r.ticketNo)).toEqual(['T-001']);
  const department = await request(app.getHttpServer()).get('/api/tickets').set('Authorization',await bearer(app,lead)).query({ department: '网络部' }).expect(200);
  expect(department.body.data.items).toHaveLength(2);
});

it('同一时间的两张工单能逐页读出并拒绝无效游标', async () => {
  const http = request(app.getHttpServer());
  const first = (await http.get('/api/tickets?limit=1').set('Authorization',await bearer(app,lead)).expect(200)).body.data;
  expect(first.items).toHaveLength(1);
  expect(first.nextCursor).toEqual(expect.any(String));
  const second = (await http.get('/api/tickets').set('Authorization',await bearer(app,lead)).query({ limit: 1, cursor: first.nextCursor }).expect(200)).body.data;
  expect(second.items).toHaveLength(1);
  expect(new Set([first.items[0].ticketNo, second.items[0].ticketNo]).size).toBe(2);
  expect(second.nextCursor).toBeNull();
  await http.get('/api/tickets?cursor=broken').set('Authorization',await bearer(app,lead)).expect(400);
});

it('甲公司提单人直接读乙公司详情与材料得到 404',async()=>{
  await request(app.getHttpServer()).get('/api/tickets/T-002').set('Authorization',await bearer(app)).expect(404);
  const r=await request(app.getHttpServer()).get('/api/tickets').set('Authorization',await bearer(app)).query({company:'乙公司'}).expect(200);
  expect(r.body.data.items).toHaveLength(0);
});
```
