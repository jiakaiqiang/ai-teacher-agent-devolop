## 一、知识点与推导

按运行与序号去重，补缺口，单独判定终止事件。

在两个浏览器会话分别以甲、乙公司提单人身份操作。先记下工单号、工单版本，再创建 run。按序号读取 events，按步骤关联 traces。把另一人的 runId 代入同一个接口，保存拒绝结果。

数据库保存事件；进程内广播只传「有变化」的信号。先订阅再读库，读取中又收到唤醒就再读一轮。终态更新和最后一条业务事件必须一起提交，否则前端可能看到已结束却收不到最终答案。

浏览器保留最高业务 seq。只取 seq 大于 after 的行；心跳和 closed 控制帧不计入业务进度。原生 EventSource 的构造参数不能传授权头，用登录接口换一次性短票据，断线后重签。

```ts
import { Injectable } from '@nestjs/common';
import { z } from 'zod';
import { PrismaService, type TransactionClient } from '../prisma/prisma.service.js';
import { RunLiveHub } from './run-live.hub.js';
import { RunFailure } from './run-errors.js';

export const stepPhases = ['plan', 'tool', 'answer'] as const;
export type StepPhase = (typeof stepPhases)[number];
export const eventTypes = ['run_started', 'step_started', 'step_finished', 'run_finished'] as const;
export type EventType = (typeof eventTypes)[number];
export const STEP_LIMIT = 8;

const payloadSchema = z.record(z.string().max(32), z.union([z.string().max(500), z.number().finite(),
  z.boolean(), z.null()])).refine(value => Object.keys(value).length <= 8, 'PAYLOAD_TOO_WIDE');
export type EventPayload = z.infer<typeof payloadSchema>;

type Journal = Pick<TransactionClient, 'agentRun' | 'agentStep' | 'agentEvent' | '$queryRaw'>;

@Injectable()
export class AgentJournalService {
  constructor(private readonly db: PrismaService, private readonly live: RunLiveHub) {}

  // 事件序号由 Run 行上的计数器分配：同一事务内 UPDATE ... RETURNING 保证不重号，
  // 并发写第二个事务会等锁，拿到的一定是递增后的值。
  private async append(tx: Journal, runId: string, type: EventType, stepId: string | null,
    payload: EventPayload) {
    const parsed = payloadSchema.safeParse(payload);
    if (!parsed.success) throw new RunFailure('RUN_EVENT_INVALID');
    const [row] = await tx.$queryRaw<{ event_seq: number }[]>`
      UPDATE agent_runs SET event_seq = event_seq + 1 WHERE id = ${runId}::uuid RETURNING event_seq`;
    if (!row) throw new RunFailure('RUN_NOT_FOUND');
    return tx.agentEvent.create({
      data: { runId, stepId, seq: row.event_seq, type, payload: parsed.data },
    });
  }

  // 终态与最后一帧共用提交点。
  async finishRun<T>(runId: string, payload: EventPayload, finish: (tx: TransactionClient) => Promise<T>) {
    const result = await this.db.$transaction(async tx => {
      const run = await finish(tx);
      await this.append(tx, runId, 'run_finished', null, payload);
      return run;
    });
    this.live.publish(runId);
    return result;
  }

  async record(runId: string, type: EventType, payload: EventPayload = {}, stepId: string | null = null) {
    const event = await this.db.$transaction(tx => this.append(tx, runId, type, stepId, payload));
    // 事务提交之后再广播：订阅者醒来时一定能读到这条事件。
    this.live.publish(runId);
    return event;
  }

  // 开一步：序号由库里的现有步数决定，并和 step_started 事件写在同一个事务里。
  async beginStep(runId: string, phase: StepPhase, summary = '') {
    if (!stepPhases.includes(phase)) throw new RunFailure('RUN_STEP_PHASE_INVALID');
    return this.db.serializable(async tx => {
      const run = await tx.agentRun.findUnique({ where: { id: runId }, select: { status: true, stepCount: true } });
      if (!run) throw new RunFailure('RUN_NOT_FOUND');
      if (run.status !== 'running') throw new RunFailure('RUN_NOT_RUNNING');
      if (run.stepCount >= STEP_LIMIT) throw new RunFailure('RUN_STEP_LIMIT');
      const ordinal = run.stepCount + 1;
      const step = await tx.agentStep.create({
        data: { runId, ordinal, phase, summary: summary.slice(0, 500) },
      });
      await tx.agentRun.update({ where: { id: runId }, data: { stepCount: { increment: 1 } } });
      const event = await this.append(tx, runId, 'step_started', step.id, { ordinal, phase });
      return { step, event };
    }).then(result => { this.live.publish(runId); return result; });
  }

  // 收一步：只允许 running → succeeded/failed，失败才带错误码。
  async endStep(stepId: string, outcome: { status: 'succeeded' | 'failed'; summary?: string; errorCode?: string }) {
    return this.db.$transaction(async tx => {
      const changed = await tx.agentStep.updateMany({
        where: { id: stepId, status: 'running' },
        data: {
          status: outcome.status, endedAt: new Date(),
          errorCode: outcome.status === 'failed' ? outcome.errorCode ?? 'STEP_FAILED' : null,
          ...(outcome.summary === undefined ? {} : { summary: outcome.summary.slice(0, 500) }),
        },
      });
      if (changed.count !== 1) throw new RunFailure('RUN_STEP_STATE_CONFLICT');
      const step = await tx.agentStep.findUniqueOrThrow({ where: { id: stepId } });
      const event = await this.append(tx, step.runId, 'step_finished', step.id,
        { ordinal: step.ordinal, phase: step.phase, status: step.status, errorCode: step.errorCode });
      return { step, event };
    }).then(result => { this.live.publish(result.step.runId); return result; });
  }

  async steps(runId: string) {
    return this.db.agentStep.findMany({ where: { runId }, orderBy: { ordinal: 'asc' }, take: STEP_LIMIT });
  }

  // 读事件：按序号升序，支持 after 增量拉取（D110 的事件流用同一个口径）。
  async events(runId: string, after = 0, take = 100) {
    if (!Number.isInteger(after) || after < 0 || !Number.isInteger(take) || take < 1 || take > 200) {
      throw new RunFailure('RUN_INPUT_INVALID');
    }
    return this.db.agentEvent.findMany({
      where: { runId, seq: { gt: after } }, orderBy: { seq: 'asc' }, take,
    });
  }
}
```

```ts
// 输入问题是 400，看不到是 404，权限变化是 403；状态类问题（转移不允许或被别人改过）都是 409。
export function runHttpStatus(code: string, stateError = false) {
  if (stateError) return 409;
  if (code.endsWith('NOT_FOUND')) return 404;
  if (code === 'RUN_INPUT_INVALID') return 400;
  if (code === 'RUN_SCOPE_CHANGED') return 403;
  return 409;
}

function httpError(error: unknown): never {
  if (error instanceof HttpException) throw error;
  const known = error instanceof RunFailure || error instanceof RunStateError;
  const code = known ? error.code : 'RUN_FAILED';
  throw new HttpException({ code, message: '运行请求未完成，请查看运行状态后再试' },
    known ? runHttpStatus(code, error instanceof RunStateError) : 503);
}
```

```ts
import { Injectable } from '@nestjs/common';
import { z } from 'zod';
import { PrismaService, type TransactionClient } from '../prisma/prisma.service.js';
import { visibleTickets } from '../tickets/scope.js';
import type { AuthUser } from '../auth/auth.types.js';
import { ReadToolsService } from '../assistant/read-tools.service.js';
import { assertTransition, statusForReason, type RunStatus, type StopReason } from './run-state.js';
import { RunFailure } from './run-errors.js';

const startSchema = z.object({
  requestKey: z.uuid(),
  ticketNo: z.string().trim().min(1).max(64),
  goal: z.string().trim().min(1).max(500),
}).strict();
export type StartInput = z.infer<typeof startSchema>;
export const RUN_LIMIT = 20;

@Injectable()
export class AgentRunsService {
  constructor(private readonly db: PrismaService, private readonly tools: ReadToolsService) {}

  // 开跑前先确认工单在当前用户的数据权限范围内；范围由 W3 的 groupScope 决定。
  private async target(user: AuthUser, ticketNo: string) {
    const ticket = await this.db.ticket.findFirst({
      where: { ...visibleTickets((await this.tools.access(user.id)).user), ticketNo }, select: { ticketNo: true },
    });
    if (!ticket) throw new RunFailure('RUN_TICKET_NOT_FOUND');
    return ticket.ticketNo;
  }

  async start(user: AuthUser, input: unknown) {
    const parsed = startSchema.safeParse(input);
    if (!parsed.success) throw new RunFailure('RUN_INPUT_INVALID');
    const { requestKey, goal } = parsed.data;
    const ticketNo = await this.target(user, parsed.data.ticketNo);
    const { fingerprint } = await this.tools.access(user.id);
    // 同一用户同一请求键只建一次：重试拿回原来的 Run，不会重复开跑。
    const existing = await this.db.agentRun.findUnique({ where: { userId_requestKey: { userId: user.id, requestKey } } });
    if (existing) {
      if (existing.ticketNo !== ticketNo || existing.goal !== goal) throw new RunFailure('RUN_REQUEST_KEY_CONFLICT');
      return { run: existing, replay: true };
    }
    return this.db.serializable(async tx => {
      const live = await tx.agentRun.count({ where: { userId: user.id, status: { in: ['queued', 'running'] } } });
      if (live >= RUN_LIMIT) throw new RunFailure('RUN_LIMIT');
      const run = await tx.agentRun.create({
        data: { userId: user.id, requestKey, ticketNo, goal, scopeHash: fingerprint },
      });
      return { run, replay: false };
    });
  }

  async get(user: AuthUser, id: string) {
    if (!z.uuid().safeParse(id).success) throw new RunFailure('RUN_INPUT_INVALID');
    const run = await this.db.agentRun.findFirst({ where: { id, userId: user.id } });
    if (!run) throw new RunFailure('RUN_NOT_FOUND');
    await this.target(user,run.ticketNo);
    return run;
  }

  async list(user: AuthUser) {
    return this.db.agentRun.findMany({
      where: { userId: user.id }, orderBy: [{ createdAt: 'desc' }, { id: 'desc' }], take: 30,
    });
  }

  // 状态推进统一走条件更新：WHERE 带上旧状态和版本，影响 0 行就是被别人改过。
  private async move(id: string, from: RunStatus, to: RunStatus, version: number,
    reason?: StopReason, extra: { startedAt?: Date; endedAt?: Date } = {}, db: TransactionClient | PrismaService = this.db) {
    assertTransition(from, to, reason);
    const result = await db.agentRun.updateMany({
      where: { id, status: from, version },
      data: { status: to, version: { increment: 1 }, stopReason: reason ?? null, ...extra },
    });
    if (result.count !== 1) throw new RunFailure('RUN_STATE_CONFLICT');
    return db.agentRun.findUniqueOrThrow({ where: { id } });
  }

  // 认领：把 queued 变成 running。多个工作者同时认领只有一个成功。
  async claim(user: AuthUser, id: string, version: number) {
    const run = await this.get(user, id);
    if (run.scopeHash !== (await this.tools.access(user.id)).fingerprint) throw new RunFailure('RUN_SCOPE_CHANGED');
    return this.move(run.id, run.status as RunStatus, 'running', version, undefined, { startedAt: new Date() });
  }

  // 收尾：停止原因决定终态，不让调用方自己挑状态。
  async finish(id: string, version: number, reason: StopReason, from: RunStatus = 'running', tx?: TransactionClient) {
    return this.move(id, from, statusForReason(reason), version, reason, { endedAt: new Date() }, tx);
  }

  async cancel(user: AuthUser, id: string, version: number) {
    const run = await this.get(user, id);
    return this.finish(run.id, version, 'user_cancelled', run.status as RunStatus);
  }
}
```

```ts
import { runLoop, type StepOutcome, type StepRequest } from './run-loop.js';
import { RunFailure } from './run-errors.js';

export type Think = (request: StepRequest, context: { ticketNo: string; goal: string }) => Promise<StepOutcome>;

@Injectable()
export class RunExecutorService {
  constructor(private readonly runs: AgentRunsService, private readonly journal: AgentJournalService,
    private readonly traces: RunTraceService, 
    ) {}

  // 认领 → 逐步执行 → 把停止原因写进 Run。每一步的状态与事件由 journal 落库。
  async execute(user: AuthUser, runId: string, think: Think, budget: Partial<RunBudget> = {}) {
    const limits = parseBudget(budget);
    const claimed = await this.runs.claim(user, runId, (await this.runs.get(user, runId)).version);
    await this.journal.record(claimed.id, 'run_started',
      { ticketNo: claimed.ticketNo, maxSteps: limits.maxSteps, maxMs: limits.maxMs, maxTokens: limits.maxTokens });
    const context = { ticketNo: claimed.ticketNo, goal: claimed.goal };
    const result = await runLoop({
      budget: limits,
      think: request => think(request, context),
      onStep: async (outcome, request) => {
        const startedAt = Date.now();
        const { step } = await this.journal.beginStep(claimed.id, outcome.phase, outcome.summary);
        if (step.ordinal !== request.ordinal) throw new RunFailure('RUN_STEP_ORDINAL_MISMATCH');
        if (outcome.trace) {
          // Trace 与步骤一一对应；内容先脱敏再截断，正文不进事件。
          await this.traces.record(claimed.id, {
            stepId: step.id, ordinal: step.ordinal, kind: outcome.trace.kind,
            input: outcome.trace.input, output: outcome.trace.output,
            inputTokens: outcome.trace.inputTokens, outputTokens: outcome.trace.outputTokens,
            durationMs: Date.now() - startedAt,
          });
        }
        await this.journal.endStep(step.id, outcome.failure
          ? { status: 'failed', errorCode: outcome.failure }
          : { status: 'succeeded' });
      },
    });
    const current = await this.runs.get(user, runId);
    const finished = await this.journal.finishRun(current.id, {
      stopReason: result.reason, steps: result.spend.steps, tokens: result.spend.tokens,
      elapsedMs: Math.trunc(result.spend.elapsedMs), answer: result.answer?.slice(0, 500) ?? null,
    }, tx => this.runs.finish(current.id, current.version, result.reason, 'running', tx));
    const usage = await this.traces.usage(finished.id);
    return { run: finished, result, usage };
  }
}
```

```ts
// 进程内广播：同实例产生的事件立刻唤醒对应的流，不用等轮询。
@Injectable()
export class RunLiveHub {
  private readonly listeners = new Map<string, Set<() => void>>();

  subscribe(runId: string, wake: () => void) {
    const set = this.listeners.get(runId) ?? new Set();
    set.add(wake);
    this.listeners.set(runId, set);
    return () => {
      set.delete(wake);
      if (!set.size) this.listeners.delete(runId);
    };
  }

  publish(runId: string) {
    for (const wake of this.listeners.get(runId) ?? []) wake();
  }

  get watched() { return this.listeners.size; }
}
```

```ts
import { Inject, Injectable, type MessageEvent } from '@nestjs/common';
import { Observable } from 'rxjs';
import { AgentJournalService } from './agent-journal.service.js';
import { PrismaService } from '../prisma/prisma.service.js';

export const RUN_STREAM_OPTIONS = 'run-stream-options';
export interface RunStreamOptions { heartbeatMs: number; pollMs: number; }
export const DEFAULT_RUN_STREAM_OPTIONS: RunStreamOptions = { heartbeatMs: 15_000, pollMs: 3_000 };
export const runCursor = (seq: number) => 'r-' + seq;

import { RunLiveHub } from './run-live.hub.js';

@Injectable()
export class RunStreamService {
  private readonly connections = new Set<symbol>();
  private readonly closers = new Set<() => void>();
  private stopping = false;
  get activeConnections() { return this.connections.size; }

  constructor(private readonly journal: AgentJournalService, private readonly live: RunLiveHub,
    private readonly db: PrismaService,
    @Inject(RUN_STREAM_OPTIONS) private readonly options: RunStreamOptions) {}

  // 关闭前主动通知客户端重连，不是悄悄断开（沿用 D33 的做法）。
  beginShutdown() {
    this.stopping = true;
    for (const close of this.closers) close();
  }

  open(runId: string, after: number): Observable<MessageEvent> {
    return new Observable<MessageEvent>(observer => {
      if (this.stopping) { observer.complete(); return; }
      const connection = Symbol();
      this.connections.add(connection);
      let disposed = false;
      let running = false;
      let dirty = true;
      let cursor = after;
      const close = () => {
        observer.next({ id: runCursor(cursor), type: 'reauth', data: { reason: 'server-shutdown' } });
        observer.complete();
      };
      this.closers.add(close);

      const drain = async () => {
        dirty = true;
        if (running || disposed) return;
        running = true;
        try {
          while (dirty && !disposed) {
            dirty = false;
            // 按 seq 续传：客户端带着已收到的最大序号，不会重复也不会漏。
            const events = await this.journal.events(runId, cursor, 100);
            for (const event of events) {
              if (disposed) return;
              cursor = event.seq;
              observer.next({ id: runCursor(cursor), type: event.type,
                data: { seq: event.seq, type: event.type, payload: event.payload ?? {},
                  createdAt: event.createdAt.toISOString() } });
            }
            if (events.length === 100) dirty = true;
            // 运行已结束且事件全部发完，主动结束这条流，不让客户端一直挂着。
            if (!dirty) {
              const run = await this.db.agentRun.findUnique({ where: { id: runId }, select: { status: true, eventSeq: true } });
              if (!run) { observer.complete(); return; }
              const terminal = run.status !== 'queued' && run.status !== 'running';
              if (terminal && cursor >= run.eventSeq) {
                observer.next({ id: runCursor(cursor), type: 'closed', data: { status: run.status } });
                observer.complete();
                return;
              }
            }
          }
        } catch (error) {
          observer.error(error);
        } finally {
          running = false;
        }
      };

      const unsubscribe = this.live.subscribe(runId, () => { void drain(); });
      const poll = setInterval(() => { void drain(); }, this.options.pollMs);
      const heartbeat = setInterval(() => {
        // 心跳是命名事件，不推进业务游标。
        observer.next({ type: 'heartbeat', data: { at: new Date().toISOString() } });
      }, this.options.heartbeatMs);
      void drain();

      return () => {
        disposed = true;
        clearInterval(poll);
        clearInterval(heartbeat);
        unsubscribe();
        this.closers.delete(close);
        this.connections.delete(connection);
      };
    });
  }
}
```

```ts
import { BadRequestException, CanActivate, ExecutionContext, Injectable, ServiceUnavailableException,
  UnauthorizedException } from '@nestjs/common';
import type { Request } from 'express';
import { createHash, randomBytes } from 'node:crypto';
import { RedisService } from '../redis/redis.service.js';

export const RUN_TICKET_TTL_SECONDS = 30;
export const runTicketKey = (ticket: string) =>
  'agent-run-ticket:' + createHash('sha256').update(ticket).digest('hex');

const invalidTicket = () =>
  new UnauthorizedException({ code: 'RUN_TICKET_INVALID', message: '连接票据无效或已过期，请重新获取' });

// 浏览器的 EventSource 不能带 Authorization 头（D33 的结论），所以先用已登录的接口换一张
// 短期、一次性、绑定到这个 Run 的票据，再用它建立 SSE 连接。
@Injectable()
export class RunTicketService {
  constructor(private readonly redis: RedisService) {}

  async issue(userId: string, runId: string, after: number) {
    const ticket = randomBytes(32).toString('hex');
    try {
      await this.redis.set(runTicketKey(ticket), JSON.stringify({ userId, runId, after }),
        'EX', RUN_TICKET_TTL_SECONDS);
    } catch {
      throw new ServiceUnavailableException({ code: 'RUN_STREAM_UNAVAILABLE', message: '运行事件流暂不可用，请稍后重试' });
    }
    return { ticket, expiresAt: new Date(Date.now() + RUN_TICKET_TTL_SECONDS * 1000).toISOString() };
  }

  async consume(ticket: unknown, runId: string, lastEventId?: string) {
    if (typeof ticket !== 'string' || !/^[a-f0-9]{64}$/.test(ticket)) throw invalidTicket();
    let raw: string | null;
    try {
      // 原子读取并删除：同一张票据只能换一条连接。
      raw = await this.redis.getdel(runTicketKey(ticket));
    } catch {
      throw new ServiceUnavailableException({ code: 'RUN_STREAM_UNAVAILABLE', message: '运行事件流暂不可用，请稍后重试' });
    }
    if (raw === null) throw invalidTicket();
    let saved: { userId?: unknown; runId?: unknown; after?: unknown };
    try {
      saved = JSON.parse(raw) as typeof saved;
    } catch {
      throw invalidTicket();
    }
    if (typeof saved.userId !== 'string' || typeof saved.runId !== 'string'
      || typeof saved.after !== 'number') throw invalidTicket();
    // 票据绑定 Run：拿着另一个运行的票据来连，拒绝。
    if (saved.runId !== runId) throw invalidTicket();
    return { userId: saved.userId, runId: saved.runId, after: parseRunCursor(lastEventId) ?? saved.after };
  }
}

export function parseRunCursor(value: string | undefined) {
  if (value === undefined || value === '') return null;
  if (value.length > 12 || !/^r-(0|[1-9]\d*)$/.test(value)) {
    throw new BadRequestException({ code: 'RUN_CURSOR_INVALID', message: '运行事件游标无效' });
  }
  return Number(value.slice(2));
}

export interface RunStreamRequest extends Request {
  runStream: { userId: string; runId: string; after: number };
}

@Injectable()
export class RunStreamGuard implements CanActivate {
  constructor(private readonly tickets: RunTicketService) {}

  // 在写出 SSE 响应头之前鉴权：失败返回标准 HTTP 错误，不建立一条错误的长连接。
  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest<RunStreamRequest>();
    const runId = String(request.params.id ?? '');
    request.runStream = await this.tickets.consume(request.query.ticket, runId,
      request.header('last-event-id'));
    return true;
  }
}
```

```ts
import { RunStreamService } from '../agent/run-stream.service.js';
import { AssistantStreams } from '../assistant/stream-lifecycle.js';
import type { INestApplication } from '@nestjs/common';
import type { NextFunction, Request, Response } from 'express';
import { logEvent } from '../common/structured-log.js';
import { ShutdownState } from './shutdown-state.js';

export function drainingMiddleware(state: ShutdownState) {
  return (req: Request, res: Response, next: NextFunction) => {
    const healthProbe = ['/api/health/live', '/api/health/ready'].includes(req.path);
    if (state.draining && !healthProbe) {
      res.setHeader('Connection', 'close');
      res.status(503).json({ error: { code: 'SHUTTING_DOWN', message: '服务正在停止，请稍后重试' } });
      return;
    }
    if (!healthProbe) {
      const finished = state.trackResponse();
      res.once('finish', finished);
      res.once('close', finished);
    }
    next();
  };
}

const closing = new WeakMap<INestApplication, Promise<void>>();
// 先摘流量、结束 SSE、等待在途任务，再关闭 HTTP 与依赖；重复调用共用同一 Promise。
export function drainAndClose(app: INestApplication): Promise<void> {
  const previous = closing.get(app);
  if (previous) return previous;
  const task = (async () => {
    app.get(ShutdownState).begin();
    logEvent('shutdown.started');
    app.get(RunStreamService).beginShutdown();
    app.get(AssistantStreams).beginShutdown();
    // 保持队列生产者可用，已进入的 HTTP 仍可能要完成入队。
    await app.get(ShutdownState).waitForResponses();
    await app.close();
    logEvent('shutdown.completed');
  })();
  closing.set(app, task);
  return task;
}
```
