## 一、知识点与推导

模型角色只处理诊断运行。工单的业务复核由运维部门负责人执行。

两侧使用同一个数据集、工具授权和模型配置。逐题保存 mode、round、实际工具序列、业务状态、耗时和 token 数；按题号和轮次做配对。先核对违规，再讨论调用代价。

允许模型直接写交接摘要，容易把推断当成原文。本例由模型给出片段位置和补查意图，宿主从已授权的片段中取值。合法的整数也可能超出这张工单实际拥有的片段数。

## 四、本章完整源码

**apps/server/src/agent/team/handoff.spec.ts**

```ts
import { receive, selected } from './handoff.js';

const id = '00000000-0000-4000-8000-000000000001';
describe('交接契约', () => {
  it('只允许选择实际存在的证据片段', () => {
    expect(selected(id, ['已见掉线', '仍有丢包'], { index: 1, needCounts: true }))
      .toEqual({ runId: id, quote: '仍有丢包', needCounts: true });
    expect(() => selected(id, ['仅一项'], { index: 1, needCounts: false })).toThrow();
    expect(() => selected(id, ['原文'], { index: 0, needCounts: false, quote: '新文字' })).toThrow();
  });
  it('拒绝跨运行交接', () => {
    const packet = selected(id, ['已见掉线'], { index: 0, needCounts: false });
    expect(receive(id, packet)).toEqual(packet);
    expect(() => receive('00000000-0000-4000-8000-000000000002', packet)).toThrow('运行不一致');
  });
});
```

**apps/server/src/agent/team/handoff.ts**

```ts
import { z } from 'zod';

export const category = z.enum(['network', 'storage', 'power', 'unknown']);
export type Category = z.infer<typeof category>;
export const selection = z.object({ index: z.number().int().min(0).max(3), needCounts: z.boolean() }).strict();
export const proposal = z.object({ category }).strict();
export const review = z.object({ verdict: z.enum(['accept', 'insufficient']) }).strict();
export const handoff = z.object({ runId: z.uuid(), quote: z.string().min(1).max(180), needCounts: z.boolean() }).strict();
export type Handoff = z.infer<typeof handoff>;

// 由宿主绑定运行并从可见片段取原文，模型只选择索引。
export function selected(runId: string, excerpts: readonly string[], raw: unknown): Handoff {
  const choice = selection.parse(raw);
  return handoff.parse({ runId, quote: excerpts[choice.index], needCounts: choice.needCounts });
}
export function receive(runId: string, envelope: unknown) {
  const packet = handoff.parse(envelope);
  if (packet.runId !== runId) throw new Error('交接运行不一致');
  return packet;
}
```
