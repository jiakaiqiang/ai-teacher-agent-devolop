## 一、知识点与推导

持久化原参数及其摘要，恢复时使用已确认的动作和版本。

将请求键、工单版本、负责人、动作参数和接收端回执画在同一时间轴。用两个进程复现确认前中断，检查 operation_tasks、ticket_events 和 preview_receipts 中留下的行。

网页表单可以先后填写原因与目标，JSON 的字段插入顺序会不同。先用联合 schema 抽取类型正确的业务字段并去掉原因两端空白，再对固定次序的元组计算 SHA-256。元组第一位是契约版本，便于以后并行支持旧命令。

改派只能指定新处理人；重启只能选择登记的 diagnosis-preview。批准有绝对截止时间，比较时刻直接来自服务端。测试注入两个时间点便可核对等号边界。

```ts
import { commandDigest, commandSchema, stillValid, approvalWindowMs } from './durable-command.js';

describe('持久命令契约', () => {
  it('同一意图生成稳定摘要并区分目标与版本', () => {
    const input = commandSchema.parse({ action: 'reassign', version: 3, reason: ' 夜班交接 ', assignee: 'bob' });
    const hash = (cmd: typeof input) => commandDigest('运行甲', 'T100', '甲', 'east', cmd);
    expect(hash(input)).toMatch(/^[a-f0-9]{64}$/);
    expect(hash({ ...input, reason: '夜班交接' })).toBe(hash(input));
    expect(hash({ ...input, version: 4 })).not.toBe(hash(input));
    expect(hash({ action: 'restart', version: 3, service: 'diagnosis-preview', reason: '夜班交接' })).not.toBe(hash(input));
  });
  it('只允许登记的服务并在有效期边界停止', () => {
    const cmd = { action: 'restart', version: 0, service: 'diagnosis-preview', reason: '验证恢复' };
    expect(commandSchema.safeParse(cmd).success).toBe(true);
    expect(commandSchema.safeParse({ ...cmd, service: 'unregistered' }).success).toBe(false);
    expect(approvalWindowMs).toBe(86_400_000);
    expect(stillValid(new Date(100), new Date(99))).toBe(true);
    expect(stillValid(new Date(100), new Date(100))).toBe(false);
  });
});
```

**apps/server/src/agent/durable-command.ts**

```ts
import { createHash } from 'node:crypto';
import { z } from 'zod';

const base = { version: z.number().int().min(0), reason: z.string().trim().min(1).max(200) };
// 只保存声明式动作。服务目标是教学环境中登记的固定名称。
export const commandSchema = z.discriminatedUnion('action', [
  z.object({ action: z.literal('reassign'), assignee: z.string().trim().min(1).max(64), ...base }).strict(),
  z.object({ action: z.literal('restart'), service: z.literal('diagnosis-preview'), ...base }).strict(),
]);
export type DurableCommand = z.infer<typeof commandSchema>;

export function commandDigest(runId: string, ticketNo: string, requesterId: string, tenant: string, input: DurableCommand) {
  const command = commandSchema.parse(input);
  const target = command.action === 'reassign' ? command.assignee : command.service;
  const fields = [1, runId, ticketNo, requesterId, tenant, command.action, target, command.version, command.reason];
  return createHash('sha256').update(JSON.stringify(fields)).digest('hex');
}
export const approvalWindowMs = 24 * 60 * 60 * 1000;
export function stillValid(expiresAt: Date, now = new Date()) {
  return now.getTime() < expiresAt.getTime();
}
```
