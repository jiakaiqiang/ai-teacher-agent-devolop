## 一、知识点与推导

可能重复建议无效步骤。用事实清单核对已尝试动作、结果和待查材料。

取一段包含用户尝试、未解决原因和已批准通用步骤的材料。先标明各段来源，再分别测量完整历史和预算内历史。用最后一轮回答反查哪一项事实曾被压缩掉。

每接受一对问题和回答，水位加一。第五对到达时移出第一对，将旧问题的前 48 个 Unicode 码点写入摘要。这样内存有上限，也知道哪一段已被归档；摘要只保存一个最近的归档摘录。

按上一轮水位加一接收事件，重复和跳号都先报错。同事或模型一句“批准了”只进入问答；审批事实由 W19 业务记录提供。两类数据的更新时机不同。

**apps/server/src/agent/handover-history.spec.ts**

```ts
import { describe, expect, it } from 'vitest';
import { advance, emptyMemory, historyBlocks } from './handover-history.js';

describe('滚动交接记录', () => {
  it('五十轮后保留最近四轮和连续摘要水位', () => {
    let memory = emptyMemory();
    for (let ordinal = 1; ordinal <= 50; ordinal++) memory = advance(memory, { ordinal, question: '巡检' + ordinal, answer: '记录' });
    expect(memory.recent.map(row => row.ordinal)).toEqual([47, 48, 49, 50]);
    expect(JSON.parse(historyBlocks(memory).summary)).toEqual({ through: 46, lastQuestion: '巡检46' });
  });

  it('重复和跳号都不能推进交接水位', () => {
    const first = { ordinal: 1, question: '审批通过了吗', answer: '文字称通过' };
    const saved = advance(emptyMemory(), first);
    expect(() => advance(saved, first)).toThrow('CONTEXT_TURN_OUT_OF_ORDER');
    expect(() => advance(saved, { ...first, ordinal: 3 })).toThrow('CONTEXT_TURN_OUT_OF_ORDER');
    expect(saved.through).toBe(1);
  });
});
```

**apps/server/src/agent/handover-history.ts**

```ts
export interface TicketFacts { ticketNo: string; decision: string; approval: string }
export type HandoverEvent = { ordinal: number; question: string; answer: string };
export interface Memory {
  through: number;
  archived: number;
  lastArchived: string;
  recent: HandoverEvent[];
}
export const emptyMemory = (): Memory => ({ through: 0, archived: 0, lastArchived: '', recent: [] });

// 只把问答记作历史。决定与审批来自调用方持有的业务快照，不从模型文本提取。
export function advance(memory: Memory, turn: HandoverEvent): Memory {
  if (turn.ordinal !== memory.through + 1) throw new Error('CONTEXT_TURN_OUT_OF_ORDER');
  const recent = [...memory.recent, { ...turn }];
  const old = recent.length > 4 ? recent.shift() : undefined;
  return { through: turn.ordinal, recent, archived: memory.archived + Number(Boolean(old)),
    lastArchived: old ? Array.from(old.question).slice(0, 48).join('') : memory.lastArchived };
}

export function historyBlocks(memory: Memory) {
  return { history: memory.recent.map(row => JSON.stringify([row.ordinal, row.question, row.answer])),
    summary: memory.archived ? JSON.stringify({ through: memory.archived, lastQuestion: memory.lastArchived }) : '' };
}
```
